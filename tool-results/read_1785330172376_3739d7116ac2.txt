    51→    51→// ============================================================================
    52→    52→// A4Editor — WYSIWYG editor that mimics an A4 page
    53→    53→// - 210mm × 297mm aspect ratio
    54→    54→// - Standard academic margins (25mm top/sides, 30mm bottom — clearly visualized)
    55→    55→// - Justified serif body text, sans-serif headings
    56→    56→// - Two-row rich toolbar (NO scroll, all tools visible):
    57→    57→//     Row 1: undo/redo | font family | font size | bold/italic/underline/strike |
    58→    58→//            subscript/superscript | case (upper/lower/title) | highlight
    59→    59→//     Row 2: paragraph/H1/H2/H3 | lists/quote | alignment | watermark | select all/clear | erase
    60→    60→// - Per-button active state indicators (voyant lumineux) tracked via
    61→    61→//   document.queryCommandState + selectionchange event.
    62→    62→// - Optional watermark (filigrane) rendered behind the page content.
    63→    63→// - Persists HTML to the store.
    64→    64→// ============================================================================
    65→    65→
    66→    66→export interface A4EditorHandle {
    67→    67→  insertHtml: (html: string) => void
    68→    68→  replaceHtml: (html: string) => void
    69→    69→  focus: () => void
    70→    70→  /** Returns the current number of A4 pages based on actual content height */
    71→    71→  getPageCount: () => number
    72→    72→}
    73→    73→
    74→    74→interface A4EditorProps {
    75→    75→  value: string // HTML
    76→    76→  onChange: (html: string) => void
    77→    77→  editable?: boolean
    78→    78→  showToolbar?: boolean
    79→    79→  /** Optional callback invoked whenever the page count changes (1, 2, 3, …). */
    80→    80→  onPageCountChange?: (count: number) => void
    81→    81→}

   110→   110→  function A4Editor({ value, onChange, editable = true, showToolbar = true, onPageCountChange }, ref) {
   111→   111→    const editorRef = React.useRef<HTMLDivElement>(null)
   112→   112→    const lastValueRef = React.useRef<string>(value)
   113→   113→    const lastPageCountRef = React.useRef<number>(1)
   114→   114→    const [pageCountState, setPageCountState] = React.useState<number>(1)
   115→   115→
   116→   116→    // Active formatting state — refreshed on selectionchange and on input.
   117→   117→    const [activeState, setActiveState] = React.useState({
   118→   118→      bold: false,
   119→   119→      italic: false,
   120→   120→      underline: false,
   121→   121→      strikeThrough: false,
   122→   122→      subscript: false,
   123→   123→      superscript: false,
   124→   124→      insertUnorderedList: false,
   125→   125→      insertOrderedList: false,
   126→   126→      justifyLeft: false,
   127→   127→      justifyCenter: false,
   128→   128→      justifyRight: false,
   129→   129→      justifyFull: true, // default for academic body text
   130→   130→      block: 'p', // 'p' | 'h1' | 'h2' | 'h3' | 'blockquote'
   131→   131→    })
   132→   132→
   133→   133→    // Watermark state — applied to the A4 page background.
   134→   134→    const [watermark, setWatermark] = React.useState<{
   135→   135→      enabled: boolean
   136→   136→      text: string
   137→   137→      opacity: number
   138→   138→      angle: number
   139→   139→    }>({
   140→   140→      enabled: false,
   141→   141→      text: 'CONFIDENTIEL',
   142→   142→      opacity: 0.08,
   143→   143→      angle: -30,
   144→   144→    })
   145→   145→
   146→   146→    // ----------------------------------------------------------------------
   147→   147→    // Pagination : measure content height, derive page count, notify parent
   148→   148→    // ----------------------------------------------------------------------
   149→   149→    function recomputePageCount() {
   150→   150→      const el = editorRef.current
   151→   151→      if (!el) return
   152→   152→      const px = el.scrollHeight
   153→   153→      const count = Math.max(1, Math.ceil(px / A4_PAGE_HEIGHT_PX))
   154→   154→      if (count !== lastPageCountRef.current) {
   155→   155→        lastPageCountRef.current = count
   156→   156→        setPageCountState(count)
   157→   157→        onPageCountChange?.(count)
   158→   158→      }
   159→   159→    }
   160→   160→
   161→   161→    // ----------------------------------------------------------------------
   162→   162→    // Refresh active formatting state from the current selection.
   163→   163→    // Uses document.queryCommandState for the boolean commands and a manual
   164→   164→    // check on the closest block ancestor for formatBlock.
   165→   165→    // ----------------------------------------------------------------------
   166→   166→    function refreshActiveState() {
   167→   167→      if (!editorRef.current) return
   168→   168→      // Only refresh when the selection is inside our editor
   169→   169→      const sel = window.getSelection()
   170→   170→      if (!sel || sel.rangeCount === 0) return
   171→   171→      const range = sel.getRangeAt(0)
   172→   172→      if (!editorRef.current.contains(range.commonAncestorContainer)) return
   173→   173→
   174→   174→      const safe = (cmd: string) => {
   175→   175→        try {
   176→   176→          return document.queryCommandState(cmd)
   177→   177→        } catch {
   178→   178→          return false
   179→   179→        }
   180→   180→      }
   181→   181→
   182→   182→      // Determine the block tag of the current selection
   183→   183→      let node: Node | null = range.commonAncestorContainer
   184→   184→      let blockTag = 'p'
   185→   185→      while (node && node !== editorRef.current) {
   186→   186→        if (node.nodeType === 1) {
   187→   187→          const tag = (node as HTMLElement).tagName.toLowerCase()
   188→   188→          if (['h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'blockquote', 'p', 'div', 'li'].includes(tag)) {
   189→   189→            blockTag = tag === 'div' || tag === 'li' ? 'p' : tag
   190→   190→            break
   191→   191→          }
   192→   192→        }
   193→   193→        node = node.parentNode
   194→   194→      }
   195→   195→
   196→   196→      setActiveState({
   197→   197→        bold: safe('bold'),
   198→   198→        italic: safe('italic'),
   199→   199→        underline: safe('underline'),
   200→   200→        strikeThrough: safe('strikeThrough'),
   201→   201→        subscript: safe('subscript'),
   202→   202→        superscript: safe('superscript'),
   203→   203→        insertUnorderedList: safe('insertUnorderedList'),
   204→   204→        insertOrderedList: safe('insertOrderedList'),
   205→   205→        justifyLeft: safe('justifyLeft'),
   206→   206→        justifyCenter: safe('justifyCenter'),
   207→   207→        justifyRight: safe('justifyRight'),
   208→   208→        justifyFull: safe('justifyFull'),
   209→   209→        block: blockTag as any,
   210→   210→      })
   211→   211→    }
   212→   212→
   213→   213→    // Expose imperative API
   214→   214→    React.useImperativeHandle(ref, () => ({
   215→   215→      insertHtml: (html: string) => {
   216→   216→        const el = editorRef.current
   217→   217→        if (!el) return
   218→   218→        el.focus()
   219→   219→        const sel = window.getSelection()
   220→   220→        const hasSelection = sel && sel.rangeCount > 0 && !sel.isCollapsed
   221→   221→        if (hasSelection) {
   222→   222→          const range = sel.getRangeAt(0)
   223→   223→          if (el.contains(range.commonAncestorContainer)) {
   224→   224→            range.deleteContents()
   225→   225→            const fragment = range.createContextualFragment(html)
   226→   226→            const lastNode = fragment.lastChild
   227→   227→            range.insertNode(fragment)
   228→   228→            if (lastNode) {
   229→   229→              range.setStartAfter(lastNode)
   230→   230→              range.setEndAfter(lastNode)
   231→   231→              sel.removeAllRanges()
   232→   232→              sel.addRange(range)
   233→   233→            }
   234→   234→          } else {
   235→   235→            appendHtml(el, html)
   236→   236→          }
   237→   237→        } else {
   238→   238→          appendHtml(el, html)
   239→   239→        }
   240→   240→        const newHtml = el.innerHTML
   241→   241→        lastValueRef.current = newHtml
   242→   242→        onChange(newHtml)
   243→   243→      },
   244→   244→      replaceHtml: (html: string) => {
   245→   245→        const el = editorRef.current
   246→   246→        if (!el) return
   247→   247→        el.innerHTML = html || ''
   248→   248→        lastValueRef.current = html || ''
   249→   249→        onChange(html || '')
   250→   250→        el.focus()
   251→   251→        requestAnimationFrame(() => recomputePageCount())
   252→   252→      },
   253→   253→      focus: () => editorRef.current?.focus(),
   254→   254→      getPageCount: () => lastPageCountRef.current,
   255→   255→    }))
   256→   256→
   257→   257→    // ----------------------------------------------------------------------
   258→   258→    // Recompute page count when value changes (external updates, e.g., AI draft)
   259→   259→    // ----------------------------------------------------------------------
   260→   260→    React.useEffect(() => {
   261→   261→      recomputePageCount()
   262→   262→    }, [value])
   263→   263→
   264→   264→    React.useEffect(() => {
   265→   265→      const onResize = () => recomputePageCount()
   266→   266→      window.addEventListener('resize', onResize)
   267→   267→      // Listen for selection changes to update active state indicators
   268→   268→      const onSelChange = () => refreshActiveState()
   269→   269→      document.addEventListener('selectionchange', onSelChange)
   270→   270→      const el = editorRef.current
   271→   271→      let ro: ResizeObserver | null = null
   272→   272→      if (el && typeof ResizeObserver !== 'undefined') {
   273→   273→        ro = new ResizeObserver(() => recomputePageCount())
   274→   274→        ro.observe(el)
   275→   275→      }
   276→   276→      requestAnimationFrame(() => recomputePageCount())
   277→   277→      return () => {
   278→   278→        window.removeEventListener('resize', onResize)
   279→   279→        document.removeEventListener('selectionchange', onSelChange)
   280→   280→        ro?.disconnect()
   281→   281→      }
   282→   282→      // eslint-disable-next-line react-hooks/exhaustive-deps
   283→   283→    }, [])
   284→   284→
   285→   285→    // Sync external value -> editor (only when it really changed externally)
   286→   286→    React.useEffect(() => {
   287→   287→      const el = editorRef.current
   288→   288→      if (!el) return
   289→   289→      if (value !== lastValueRef.current && value !== el.innerHTML) {
   290→   290→        el.innerHTML = value || ''
   291→   291→        lastValueRef.current = value
   292→   292→      }
   293→   293→    }, [value])
   294→   294→
   295→   295→    // Initialize once
   296→   296→    React.useEffect(() => {
   297→   297→      const el = editorRef.current
   298→   298→      if (!el) return
   299→   299→      if (el.innerHTML !== (value || '')) {
   300→   300→        el.innerHTML = value || ''
   301→   301→        lastValueRef.current = value || ''
   302→   302→      }
   303→   303→      // eslint-disable-next-line react-hooks/exhaustive-deps
   304→   304→    }, [])
   305→   305→
   306→   306→    function handleInput() {
   307→   307→      const el = editorRef.current
   308→   308→      if (!el) return
   309→   309→      const html = el.innerHTML
   310→   310→      lastValueRef.current = html
   311→   311→      onChange(html)
   312→   312→      recomputePageCount()
   313→   313→      refreshActiveState()
   314→   314→    }
   315→   315→
   316→   316→    function exec(command: string, val?: string) {
   317→   317→      editorRef.current?.focus()
   318→   318→      document.execCommand(command, false, val)
   319→   319→      handleInput()
   320→   320→    }
   321→   321→
   322→   322→    function setBlock(tag: 'h1' | 'h2' | 'h3' | 'p' | 'blockquote') {
   323→   323→      editorRef.current?.focus()
   324→   324→      try {
   325→   325→        document.execCommand('formatBlock', false, `<${tag}>`)
   326→   326→      } catch {
   327→   327→        // fallback noop
   328→   328→      }
   329→   329→      handleInput()
   330→   330→    }
   331→   331→
   332→   332→    function clearFormatting() {
   333→   333→      editorRef.current?.focus()
   334→   334→      document.execCommand('formatBlock', false, '<p>')
   335→   335→      document.execCommand('removeFormat')
   336→   336→      // Also clear subscript/superscript and lists
   337→   337→      document.execCommand('unlink')
   338→   338→      handleInput()
   339→   339→    }
   340→   340→
   341→   341→    // ----------------------------------------------------------------------
   342→   342→    // Case conversion — operates on the current selection.
   343→   343→    // If the selection is collapsed, no-op.
   344→   344→    // ----------------------------------------------------------------------
   345→   345→    function convertCase(mode: 'upper' | 'lower' | 'title') {
   346→   346→      const sel = window.getSelection()
   347→   347→      if (!sel || sel.rangeCount === 0 || sel.isCollapsed) {
   348→   348→        toast_editor('Sélectionnez d\'abord le texte à transformer.')
   349→   349→        return
   350→   350→      }
   351→   351→      const range = sel.getRangeAt(0)
   352→   352→      const text = range.toString()
   353→   353→      if (!text) return
   354→   354→      let out = text
   355→   355→      if (mode === 'upper') out = text.toUpperCase()
   356→   356→      else if (mode === 'lower') out = text.toLowerCase()
   357→   357→      else if (mode === 'title') {
   358→   358→        out = text
   359→   359→          .toLowerCase()
   360→   360→          .replace(/(^|\s|[\.\!\?\;\:])([a-zà-ÿ])/g, (_, p1, p2) => p1 + p2.toUpperCase())
   361→   361→      }
   362→   362→      // Insert as a text node (preserves any inline styles of the wrapping element)
   363→   363→      const textNode = document.createTextNode(out)
   364→   364→      range.deleteContents()
   365→   365→      range.insertNode(textNode)
   366→   366→      // Reselect the new text so the user can continue working on it
   367→   367→      const newRange = document.createRange()
   368→   368→      newRange.selectNodeContents(textNode)
   369→   369→      sel.removeAllRanges()
   370→   370→      sel.addRange(newRange)
   371→   371→      handleInput()
   372→   372→    }
   373→   373→
   374→   374→    // Apply font family to the current selection (or future typing at caret)
   375→   375→    function applyFontFamily(fontValue: string) {
   376→   376→      const ff = FONT_FAMILIES.find((f) => f.value === fontValue)
   377→   377→      if (!ff) return
   378→   378→      exec('fontName', ff.css)
   379→   379→    }
   380→   380→
   381→   381→    // Apply font size to the current selection.
   382→   382→    // execCommand('fontSize', false, n) only accepts 1..7, so we use the
   383→   383→    // FONT_SIZES mapping above. To get true pt sizes we additionally wrap the
   384→   384→    // selection in a <span style="font-size:Npt"> — but execCommand is simpler
   385→   385→    // and matches what most users expect from a basic WYSIWYG toolbar.
   386→   386→    function applyFontSize(sizeValue: string) {
   387→   387→      exec('fontSize', sizeValue)
   388→   388→    }
   389→   389→
   390→   390→    // Apply text color via execCommand('foreColor')
   391→   391→    function applyTextColor(color: string) {
   392→   392→      exec('foreColor', color)
   393→   393→    }
   394→   394→
   395→   395→    // Apply highlight color via execCommand('hiliteColor') (fallback 'backColor')
   396→   396→    function applyHighlight(color: string) {
   397→   397→      editorRef.current?.focus()
   398→   398→      try {
   399→   399→        if (!document.execCommand('hiliteColor', false, color)) {
   400→   400→          document.execCommand('backColor', false, color)
   401→   401→        }
   402→   402→      } catch {
   403→   403→        document.execCommand('backColor', false, color)
   404→   404→      }
   405→   405→      handleInput()
   406→   406→    }
   407→   407→
   408→   408→    // Select all content inside the editor (object selection)
   409→   409→    function selectAllInEditor() {
   410→   410→      const el = editorRef.current
   411→   411→      if (!el) return
   412→   412→      el.focus()
   413→   413→      const range = document.createRange()
   414→   414→      range.selectNodeContents(el)
   415→   415→      const sel = window.getSelection()
   416→   416→      sel?.removeAllRanges()
   417→   417→      sel?.addRange(range)
   418→   418→      refreshActiveState()
   419→   419→    }
   420→   420→
   421→   421→    // Clear selection (collapse caret)
   422→   422→    function clearSelection() {
   423→   423→      const sel = window.getSelection()
   424→   424→      sel?.removeAllRanges()
   425→   425→      editorRef.current?.focus()
   426→   426→      refreshActiveState()
   427→   427→    }
   428→   428→
   429→   429→    function handleKeyDown(e: React.KeyboardEvent) {
   430→   430→      const meta = e.metaKey || e.ctrlKey
   431→   431→      if (meta && e.key.toLowerCase() === 'b') {
   432→   432→        e.preventDefault()
   433→   433→        exec('bold')
   434→   434→      } else if (meta && e.key.toLowerCase() === 'i') {
   435→   435→        e.preventDefault()
   436→   436→        exec('italic')
   437→   437→      } else if (meta && e.key.toLowerCase() === 'u') {
   438→   438→        e.preventDefault()
   439→   439→        exec('underline')
   440→   440→      } else if (meta && e.key.toLowerCase() === 's') {
   441→   441→        e.preventDefault()
   442→   442→        exec('strikethrough')
   443→   443→      } else if (meta && e.key.toLowerCase() === 'e') {
   444→   444→        // Ctrl+E = center alignment (custom shortcut)
   445→   445→        e.preventDefault()
   446→   446→        exec('justifyCenter')
   447→   447→      } else if (meta && e.shiftKey && e.key.toLowerCase() === 'a') {
   448→   448→        // Ctrl+Shift+A = select all in editor
   449→   449→        e.preventDefault()
   450→   450→        selectAllInEditor()
   451→   451→      }
   452→   452→    }
   453→   453→
   454→   454→    function handlePaste(e: React.ClipboardEvent) {
   455→   455→      e.preventDefault()
   456→   456→      const text = e.clipboardData.getData('text/plain')
   457→   457→      const html = text
   458→   458→        .split(/\n\s*\n+/)
   459→   459→        .map((p) => `<p>${escapeHtml(p).replace(/\n/g, '<br/>')}</p>`)
   460→   460→        .join('')
   461→   461→      document.execCommand('insertHTML', false, html)
   462→   462→      handleInput()
   463→   463→    }
   464→   464→
   465→   465→    return (
   466→   466→      <div className="flex flex-col h-full">
   467→   467→        {showToolbar && editable && (
   468→   468→          <Toolbar
   469→   469→            active={activeState}
   470→   470→            onBold={() => exec('bold')}
   471→   471→            onItalic={() => exec('italic')}
   472→   472→            onUnderline={() => exec('underline')}
   473→   473→            onStrike={() => exec('strikeThrough')}
   474→   474→            onSubscript={() => exec('subscript')}
   475→   475→            onSuperscript={() => exec('superscript')}
   476→   476→            onH1={() => setBlock('h1')}
