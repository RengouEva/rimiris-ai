   210→   210→function SectionItem({
   211→   211→  section,
   212→   212→  index,
   213→   213→  total,
   214→   214→  active,
   215→   215→  onSelect,
   216→   216→  onRename,
   217→   217→  onDelete,
   218→   218→  onDuplicate,
   219→   219→  onMoveUp,
   220→   220→  onMoveDown,
   221→   221→  onMarkCompleted,
   222→   222→}: {
   223→   223→  section: Section
   224→   224→  index: number
   225→   225→  total: number
   226→   226→  active: boolean
   227→   227→  onSelect: () => void
   228→   228→  onRename: (title: string) => void
   229→   229→  onDelete: () => void
   230→   230→  onDuplicate: () => void
   231→   231→  onMoveUp: () => void
   232→   232→  onMoveDown: () => void
   233→   233→  onMarkCompleted: () => void
   234→   234→}) {
   235→   235→  const [editing, setEditing] = React.useState(false)
   236→   236→  const [draftTitle, setDraftTitle] = React.useState(section.title)
   237→   237→  const status = STATUS_CONFIG[section.status]
   238→   238→
   239→   239→  React.useEffect(() => {
   240→   240→    setDraftTitle(section.title)
   241→   241→  }, [section.title])
   242→   242→
   243→   243→  function commitRename() {
   244→   244→    const t = draftTitle.trim()
   245→   245→    if (t && t !== section.title) {
   246→   246→      onRename(t)
   247→   247→    } else {
   248→   248→      setDraftTitle(section.title)
   249→   249→    }
   250→   250→    setEditing(false)
   251→   251→  }
   252→   252→
   253→   253→  return (
   254→   254→    <motion.div
   255→   255→      layout
   256→   256→      initial={{ opacity: 0, x: -10 }}
   257→   257→      animate={{ opacity: 1, x: 0 }}
   258→   258→      exit={{ opacity: 0, x: -10 }}
   259→   259→      className={cn(
   260→   260→        'group relative rounded-lg transition-all',
   261→   261→        active ? 'bg-card shadow-sm border border-primary/30' : 'hover:bg-card/60 border border-transparent'
   262→   262→      )}
   263→   263→    >
   264→   264→      <div
   265→   265→        onClick={onSelect}
   266→   266→        className="w-full text-left p-2.5 flex items-start gap-2 cursor-pointer"
   267→   267→      >
   268→   268→        <div className="flex flex-col items-center gap-1 pt-0.5">
   269→   269→          <span className="text-[10px] font-mono text-muted-foreground">{index + 1}</span>
   270→   270→          <span className={cn('w-1.5 h-1.5 rounded-full', status.dot)} />
   271→   271→        </div>
   272→   272→        <div className="flex-1 min-w-0">
   273→   273→          {editing ? (
   274→   274→            <input
   275→   275→              autoFocus
   276→   276→              value={draftTitle}
   277→   277→              onChange={(e) => setDraftTitle(e.target.value)}
   278→   278→              onBlur={commitRename}
   279→   279→              onKeyDown={(e) => {
   280→   280→                if (e.key === 'Enter') commitRename()
   281→   281→                if (e.key === 'Escape') {
   282→   282→                  setDraftTitle(section.title)
   283→   283→                  setEditing(false)
   284→   284→                }
   285→   285→              }}
   286→   286→              onClick={(e) => e.stopPropagation()}
   287→   287→              className="w-full text-sm font-medium bg-transparent border-b border-primary outline-none"
   288→   288→            />
   289→   289→          ) : (
   290→   290→            <p className="text-sm font-medium truncate">{section.title}</p>
   291→   291→          )}
   292→   292→          <div className="flex items-center gap-2 mt-0.5">
   293→   293→            <span className={cn('text-[10px]', status.color)}>{status.label}</span>
   294→   294→            {section.wordCount > 0 && (
   295→   295→              <span className="text-[10px] text-muted-foreground">{section.wordCount} mots</span>
   296→   296→            )}
   297→   297→          </div>
   298→   298→        </div>
   299→   299→
   300→   300→        <DropdownMenu>
   301→   301→          <DropdownMenuTrigger asChild>
   302→   302→            <button
   303→   303→              onClick={(e) => e.stopPropagation()}
   304→   304→              className="opacity-0 group-hover:opacity-100 transition-opacity p-1 rounded hover:bg-muted"
   305→   305→            >
   306→   306→              <MoreVertical className="h-3.5 w-3.5 text-muted-foreground" />
   307→   307→            </button>
   308→   308→          </DropdownMenuTrigger>
   309→   309→          <DropdownMenuContent align="end" className="w-44">
   310→   310→            <DropdownMenuItem onClick={() => setEditing(true)}>
   311→   311→              <Edit3 className="h-3.5 w-3.5 mr-2" />
   312→   312→              Renommer
   313→   313→            </DropdownMenuItem>
   314→   314→            <DropdownMenuItem onClick={onDuplicate}>
   315→   315→              <Copy className="h-3.5 w-3.5 mr-2" />
   316→   316→              Dupliquer
   317→   317→            </DropdownMenuItem>
   318→   318→            <DropdownMenuItem onClick={onMarkCompleted}>
   319→   319→              <CheckCircle2 className="h-3.5 w-3.5 mr-2" />
   320→   320→              {section.status === 'completed' ? 'Marquer brouillon' : 'Marquer terminé'}
   321→   321→            </DropdownMenuItem>
   322→   322→            <DropdownMenuSeparator />
   323→   323→            <DropdownMenuItem
   324→   324→              onClick={onMoveUp}
   325→   325→              disabled={index === 0}
   326→   326→            >
   327→   327→              <ChevronUp className="h-3.5 w-3.5 mr-2" />
   328→   328→              Monter
   329→   329→            </DropdownMenuItem>
   330→   330→            <DropdownMenuItem
   331→   331→              onClick={onMoveDown}
   332→   332→              disabled={index === total - 1}
   333→   333→            >
   334→   334→              <ChevronDown className="h-3.5 w-3.5 mr-2" />
   335→   335→              Descendre
   336→   336→            </DropdownMenuItem>
   337→   337→            <DropdownMenuSeparator />
   338→   338→            <DropdownMenuItem onClick={onDelete} className="text-red-500">
   339→   339→              <Trash2 className="h-3.5 w-3.5 mr-2" />
   340→   340→              Supprimer
   341→   341→            </DropdownMenuItem>
   342→   342→          </DropdownMenuContent>
   343→   343→        </DropdownMenu>
   344→   344→      </div>
   345→   345→    </motion.div>
   346→   346→  )
   347→   347→}

   353→   353→function EditorView({

   355→   355→  project,
   356→   356→  onUpdateContent,
   357→   357→  onOpenAI,
   358→   358→  onOpenBlocked,
   359→   359→}: {

   361→   361→  project: any

   363→   363→  onOpenAI: () => void
   364→   364→  onOpenBlocked: () => void

   372→   372→      setShowAIHint(true)

   389→   389→              <span className="text-xs text-muted-foreground">
   390→   390→                {section.wordCount} mots

   392→   392→              </span>
   393→   393→            </div>
   394→   394→            <div className="flex items-center gap-1.5">

   396→   396→                size="sm"

   398→   398→                onClick={onOpenBlocked}
   399→   399→                className="rounded-full border-amber-500/40 text-amber-600 hover:bg-amber-500/10"
   400→   400→              >
   401→   401→                <HelpCircle className="h-3.5 w-3.5 mr-1" />

   403→   403→              </Button>

   405→   405→                size="sm"
   406→   406→                onClick={onOpenAI}
   407→   407→                className="rounded-full iris-gradient text-white"
   408→   408→              >
   409→   409→                <Sparkles className="h-3.5 w-3.5 mr-1" />
   410→   410→                Demander à IRIS
   411→   411→              </Button>

   413→   413→          </div>

   416→   416→

   420→   420→          <input
   421→   421→            value={section.title}
   422→   422→            onChange={(e) => {
   423→   423→              // Title editing directly in the editor
   424→   424→              const store = useIrisStore.getState()

   437→   437→          />
   438→   438→
   439→   439→          {/* AI hint */}
   440→   440→          <AnimatePresence>
   441→   441→            {showAIHint && (
   442→   442→              <motion.div
   443→   443→                initial={{ opacity: 0, y: 10 }}
   444→   444→                animate={{ opacity: 1, y: 0 }}
   445→   445→                exit={{ opacity: 0, y: 10 }}
   446→   446→                className="mt-6 p-4 rounded-xl border border-primary/30 bg-primary/5"
   447→   447→              >
   448→   448→                <div className="flex items-start gap-3">
   449→   449→                  <div className="w-9 h-9 rounded-full iris-gradient flex items-center justify-center flex-shrink-0">
   450→   450→                    <Sparkles className="h-4 w-4 text-white" />
   451→   451→                  </div>
   452→   452→                  <div className="flex-1">
   453→   453→                    <p className="text-sm font-medium mb-1">
   454→   454→                      Vous avez commencé à écrire. Voulez-vous un retour ?
   455→   455→                    </p>
   456→   456→                    <p className="text-xs text-muted-foreground mb-2">
   457→   457→                      IRIS peut relire votre brouillon et vous proposer des améliorations.
   458→   458→                    </p>
   459→   459→                    <Button size="sm" variant="outline" onClick={onOpenAI}>
   460→   460→                      Demander un retour
   461→   461→                    </Button>
   462→   462→                  </div>
   463→   463→                </div>
   464→   464→              </motion.div>
   465→   465→            )}
   466→   466→          </AnimatePresence>
   467→   467→
   468→   468→          {section.messages.length > 0 && (
   469→   469→            <div className="mt-8 pt-6 border-t">
   470→   470→              <p className="text-xs font-semibold text-muted-foreground mb-2 uppercase tracking-wide">
   471→   471→                Vos échanges avec IRIS sur cette section
   472→   472→              </p>
   473→   473→              <Button size="sm" variant="ghost" onClick={onOpenAI}>
   474→   474→                <PenLine className="h-3.5 w-3.5 mr-1" />
   475→   475→                Ouvrir la conversation ({section.messages.length})
   476→   476→              </Button>
   477→   477→            </div>
   478→   478→          )}
   479→   479→        </div>
   480→   480→      </div>
   481→   481→    </div>
   482→   482→  )
   483→   483→}

   526→   526→function AIPanel({
   527→   527→  section,
   528→   528→  project,
   529→   529→  allSections,
   530→   530→  onClose,
   531→   531→}: {
   532→   532→  section: Section
   533→   533→  project: any
   534→   534→  allSections: { title: string; content: string }[]
   535→   535→  onClose: () => void
   536→   536→}) {
   537→   537→  const { addMessage, blockedMode, setBlockedMode } = useIrisStore()
   538→   538→  const [input, setInput] = React.useState('')
   539→   539→  const [loading, setLoading] = React.useState(false)
   540→   540→  const scrollRef = React.useRef<HTMLDivElement>(null)
   541→   541→
   542→   542→  React.useEffect(() => {
   543→   543→    if (scrollRef.current) {
   544→   544→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight
   545→   545→    }
   546→   546→  }, [section.messages.length, loading])
   547→   547→
   548→   548→  async function send(text: string) {
   549→   549→    if (!text.trim() || loading) return
   550→   550→
   551→   551→    addMessage(section.id, { role: 'user', content: text.trim() })
   552→   552→    setInput('')
   553→   553→    setLoading(true)
   554→   554→
   555→   555→    try {
   556→   556→      const history = section.messages
   557→   557→        .filter((m) => m.role !== 'system')
   558→   558→        .slice(-6)
   559→   559→        .map((m) => ({ role: m.role as 'user' | 'assistant', content: m.content }))
   560→   560→
   561→   561→      const res = await fetch('/api/ai/chat', {
   562→   562→        method: 'POST',
   563→   563→        headers: { 'Content-Type': 'application/json' },
   564→   564→        body: JSON.stringify({
   565→   565→          sectionTitle: section.title,
   566→   566→          sectionContent: section.content,
   567→   567→          templateRef: section.templateRef,
   568→   568→          project,
   569→   569→          allSections,
   570→   570→          history,
   571→   571→          userMessage: text.trim(),
   572→   572→          blockedMode,
   573→   573→        }),
   574→   574→      })
   575→   575→      const data = await res.json()
   576→   576→
   577→   577→      addMessage(section.id, {
   578→   578→        role: 'assistant',
   579→   579→        content: data.reply || '...',
   580→   580→        agent: data.agent?.id,
   581→   581→      })
   582→   582→
   583→   583→      if (blockedMode) setBlockedMode(false)
   584→   584→    } catch {
   585→   585→      addMessage(section.id, {
   586→   586→        role: 'assistant',
   587→   587→        content: "Erreur de communication. Reformulez votre demande.",
   588→   588→      })
   589→   589→    } finally {
   590→   590→      setLoading(false)
   591→   591→    }
   592→   592→  }
   593→   593→
   594→   594→  function handleKeyDown(e: React.KeyboardEvent<HTMLTextAreaElement>) {
   595→   595→    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   596→   596→      e.preventDefault()
   597→   597→      send(input)
   598→   598→    }
   599→   599→  }
   600→   600→
   601→   601→  // Determine agent
   602→   602→  const agent = React.useMemo(() => {
   603→   603→    if (section.templateRef) {
   604→   604→      return getAgentForChapter(section.templateRef)
   605→   605→    }
   606→   606→    return AGENTS[0] // Pr. IRIS by default
   607→   607→  }, [section.templateRef])
   608→   608→
   609→   609→  const suggestions = blockedMode
   610→   610→    ? [
   611→   611→        "Je ne sais pas par où commencer",
   612→   612→        "Je n'arrive pas à formuler cette partie",
   613→   613→        "Aidez-moi à structurer mes idées",
   614→   614→      ]
   615→   615→    : [
   616→   616→        "Relis mon brouillon et donne-moi 3 conseils",
   617→   617→        "Qu'est-ce qui manque à cette section ?",
   618→   618→        "Aide-moi à formuler une phrase d'introduction",
   619→   619→        "Comment améliorer la cohérence avec le reste ?",
   620→   620→      ]
   621→   621→
   622→   622→  return (
   623→   623→    <motion.aside
   624→   624→      initial={{ x: '100%' }}
   625→   625→      animate={{ x: 0 }}
   626→   626→      exit={{ x: '100%' }}
   627→   627→      transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   628→   628→      className="w-full sm:w-96 border-l border-border bg-card flex flex-col flex-shrink-0 absolute right-0 top-0 bottom-0 sm:relative z-30"
   629→   629→    >
   630→   630→      {/* Header */}
   631→   631→      <div className="p-3 border-b border-border flex items-center gap-2">
   632→   632→        <div className="w-8 h-8 rounded-full iris-gradient flex items-center justify-center flex-shrink-0">
   633→   633→          {blockedMode ? (
   634→   634→            <Brain className="h-4 w-4 text-white" />
   635→   635→          ) : (
   636→   636→            <Sparkles className="h-4 w-4 text-white" />
   637→   637→          )}
   638→   638→        </div>
   639→   639→        <div className="flex-1 min-w-0">
   640→   640→          <p className="text-sm font-semibold">
   641→   641→            {blockedMode ? 'Mode débloqué' : agent.name}
   642→   642→          </p>
   643→   643→          <p className="text-xs text-muted-foreground truncate">
   644→   644→            {blockedMode ? 'IRIS vous aide à repartir' : agent.role}
   645→   645→          </p>
   646→   646→        </div>
   647→   647→        {blockedMode && (
   648→   648→          <Button
   649→   649→            size="sm"
   650→   650→            variant="ghost"
   651→   651→            onClick={() => setBlockedMode(false)}
   652→   652→            className="text-xs h-7"
   653→   653→          >
   654→   654→            Retour normal
   655→   655→          </Button>
   656→   656→        )}
   657→   657→        <button onClick={onClose} className="p-1.5 rounded hover:bg-muted">
   658→   658→          <X className="h-4 w-4" />
   659→   659→        </button>
   660→   660→      </div>
   661→   661→
   662→   662→      {/* Context banner */}
   663→   663→      <div className="px-3 py-2 bg-muted/40 border-b border-border">
   664→   664→        <p className="text-[10px] uppercase tracking-wide text-muted-foreground mb-0.5">
   665→   665→          Section en cours
   666→   666→        </p>
   667→   667→        <p className="text-xs font-medium truncate">{section.title}</p>
   668→   668→      </div>
   669→   669→
   670→   670→      {/* Messages */}
   671→   671→      <div ref={scrollRef} className="flex-1 overflow-y-auto p-3 space-y-3">
   672→   672→        {section.messages.length === 0 ? (
   673→   673→          <div className="text-center py-6">
   674→   674→            <p className="text-sm text-muted-foreground mb-4">
   675→   675→              {blockedMode
   676→   676→                ? "Décrivez votre blocage, ou choisissez une suggestion."
   677→   677→                : `Posez votre question à ${agent.name}, ou choisissez une suggestion.`}
   678→   678→            </p>
   679→   679→            <div className="space-y-1.5">
   680→   680→              {suggestions.map((s, idx) => (
   681→   681→                <button
   682→   682→                  key={idx}
   683→   683→                  onClick={() => send(s)}
   684→   684→                  className="w-full text-left text-xs p-2.5 rounded-lg border border-border bg-background hover:border-primary/40 hover:bg-primary/5 transition-colors"
   685→   685→                >
   686→   686→                  <span className="text-primary mr-1.5">→</span>
   687→   687→                  {s}
   688→   688→                </button>
   689→   689→              ))}
   690→   690→            </div>
   691→   691→          </div>
   692→   692→        ) : (
   693→   693→          section.messages.map((msg) => {
   694→   694→            const isUser = msg.role === 'user'
   695→   695→            const msgAgent = msg.agent ? AGENTS.find((a) => a.id === msg.agent) : null
   696→   696→            return (
   697→   697→              <motion.div
   698→   698→                key={msg.id}
   699→   699→                initial={{ opacity: 0, y: 5 }}
   700→   700→                animate={{ opacity: 1, y: 0 }}
   701→   701→                className={cn('flex flex-col gap-1', isUser ? 'items-end' : 'items-start')}
   702→   702→              >
   703→   703→                {msgAgent && !isUser && (
   704→   704→                  <span className="text-[10px] font-semibold text-primary px-1">
   705→   705→                    {msgAgent.name}
   706→   706→                  </span>
   707→   707→                )}
   708→   708→                <div
   709→   709→                  className={cn(
   710→   710→                    'rounded-2xl p-3 text-sm max-w-[90%]',
   711→   711→                    isUser
   712→   712→                      ? 'bg-primary text-primary-foreground rounded-tr-sm'
   713→   713→                      : 'bg-muted rounded-tl-sm'
   714→   714→                  )}
   715→   715→                >
   716→   716→                  <p className="whitespace-pre-wrap leading-relaxed">{msg.content}</p>
   717→   717→                </div>
   718→   718→              </motion.div>
   719→   719→            )
   720→   720→          })
   721→   721→        )}
   722→   722→
   723→   723→        {loading && (
   724→   724→          <div className="flex items-start gap-2">
   725→   725→            <div className="w-7 h-7 rounded-full iris-gradient flex items-center justify-center flex-shrink-0">
   726→   726→              <Sparkles className="h-3.5 w-3.5 text-white" />
   727→   727→            </div>
   728→   728→            <div className="bg-muted rounded-2xl rounded-tl-sm p-3">
   729→   729→              <div className="flex items-center gap-1">
   730→   730→                <span className="typing-dot w-1.5 h-1.5 rounded-full bg-muted-foreground" />
   731→   731→                <span className="typing-dot w-1.5 h-1.5 rounded-full bg-muted-foreground" />
   732→   732→                <span className="typing-dot w-1.5 h-1.5 rounded-full bg-muted-foreground" />
   733→   733→              </div>
   734→   734→            </div>
   735→   735→          </div>
   736→   736→        )}
   737→   737→      </div>
   738→   738→
   739→   739→      {/* Input */}
   740→   740→      <div className="border-t border-border p-3 space-y-2">
   741→   741→        <Textarea
   742→   742→          value={input}
   743→   743→          onChange={(e) => setInput(e.target.value)}
   744→   744→          onKeyDown={handleKeyDown}
   745→   745→          placeholder={blockedMode ? "Décrivez votre blocage..." : "Votre question..."}
   746→   746→          className="resize-none min-h-16 max-h-32 text-sm"
   747→   747→          disabled={loading}
   748→   748→        />
   749→   749→        <div className="flex items-center justify-between">
   750→   750→          <span className="text-[10px] text-muted-foreground">Cmd/Ctrl+Entrée</span>
   751→   751→          <Button
   752→   752→            onClick={() => send(input)}
   753→   753→            disabled={!input.trim() || loading}
   754→   754→            size="sm"
   755→   755→            className="rounded-full iris-gradient text-white"
   756→   756→          >
   757→   757→            {loading ? (
   758→   758→              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   759→   759→            ) : (
   760→   760→              <>
   761→   761→                <Send className="h-3.5 w-3.5 mr-1" />
   762→   762→                Envoyer
   763→   763→              </>
   764→   764→            )}
   765→   765→          </Button>
   766→   766→        </div>
   767→   767→      </div>
   768→   768→    </motion.aside>
   769→   769→  )
   770→   770→}

   776→   776→function timeAgo(ts: number): string {
   777→   777→  const diff = Date.now() - ts
   778→   778→  if (diff < 60_000) return 'à l\'instant'
   779→   779→  if (diff < 3_600_000) return `il y a ${Math.floor(diff / 60_000)} min`
   780→   780→  if (diff < 86_400_000) return `il y a ${Math.floor(diff / 3_600_000)} h`
   781→   781→  return `il y a ${Math.floor(diff / 86_400_000)} j`
   782→   782→}
