    39→    39→// ============================================================================
    40→    40→// SectionWorkflowPanel — the right-side sliding panel that runs the per-section
    41→    41→// mini-workflow: interview (4 modes) → validation → drafting → humanization.
    42→    42→// Replaces the old AIPanel.

    44→    44→
    45→    45→type PanelStep = 'interview' | 'validation' | 'draft' | 'humanization' | 'done'
    46→    46→
    47→    47→export function SectionWorkflowPanel({
    48→    48→  section,
    49→    49→  onClose,
    50→    50→  onInsertDraft,
    51→    51→  onReplaceDraft,
    52→    52→}: {
    53→    53→  section: Section
    54→    54→  onClose: () => void
    55→    55→  onInsertDraft: (html: string) => void
    56→    56→  onReplaceDraft: (html: string) => void
    57→    57→}) {
    58→    58→  const {
    59→    59→    project,
    60→    60→    sections,
    61→    61→    themeUnderstanding,
    62→    62→    problemContext,
    63→    63→    addSectionInterviewAnswer,
    64→    64→    setSectionInterviewAnswers,
    65→    65→    setSectionValidation,
    66→    66→    setSectionHumanization,
    67→    67→    setSectionStatus,
    68→    68→    updateSectionContent,
    69→    69→  } = useIrisStore()
    70→    70→
    71→    71→  // Determine initial step based on section state
    72→    72→  const initialStep: PanelStep = React.useMemo(() => {
    73→    73→    if (section.status === 'completed') return 'done'
    74→    74→    if (section.humanization) return 'done'
    75→    75→    if (section.wordCount > 100 && section.validation) return 'humanization'
    76→    76→    if (section.validation?.overallOk) return 'draft'
    77→    77→    if (section.interviewAnswers.length > 0) return 'validation'
    78→    78→    return 'interview'
    79→    79→  }, [section])
    80→    80→
    81→    81→  const [step, setStep] = React.useState<PanelStep>(initialStep)
    82→    82→
    83→    83→  const allSections = sections.map((s) => ({ title: s.title, content: s.content }))
    84→    84→
    85→    85→  return (
    86→    86→    <motion.aside
    87→    87→      initial={{ x: '100%' }}
    88→    88→      animate={{ x: 0 }}
    89→    89→      exit={{ x: '100%' }}
    90→    90→      transition={{ type: 'spring', damping: 25, stiffness: 200 }}
    91→    91→      className="w-full sm:w-[460px] flex-shrink-0 border-l border-border bg-background flex flex-col h-[calc(100vh-3.5rem)]"
    92→    92→    >
    93→    93→      <PanelHeader step={step} section={section} onClose={onClose} />
    94→    94→
    95→    95→      <div className="flex-1 overflow-y-auto">
    96→    96→        <AnimatePresence mode="wait">
    97→    97→          {step === 'interview' && (
    98→    98→            <motion.div
    99→    99→              key="interview"
   100→   100→              initial={{ opacity: 0, x: 10 }}
   101→   101→              animate={{ opacity: 1, x: 0 }}
   102→   102→              exit={{ opacity: 0, x: -10 }}
   103→   103→            >
   104→   104→              <InterviewStep
   105→   105→                section={section}
   106→   106→                project={project}
   107→   107→                themeUnderstanding={themeUnderstanding}
   108→   108→                problemContext={problemContext}
   109→   109→                onAnswer={(a) => addSectionInterviewAnswer(section.id, a)}
   110→   110→                onSetAnswers={(as) => setSectionInterviewAnswers(section.id, as)}
   111→   111→                onContinue={() => setStep('validation')}
   112→   112→              />
   113→   113→            </motion.div>
   114→   114→          )}
   115→   115→          {step === 'validation' && (
   116→   116→            <motion.div
   117→   117→              key="validation"
   118→   118→              initial={{ opacity: 0, x: 10 }}
   119→   119→              animate={{ opacity: 1, x: 0 }}
   120→   120→              exit={{ opacity: 0, x: -10 }}
   121→   121→            >
   122→   122→              <ValidationStep
   123→   123→                section={section}
   124→   124→                project={project}
   125→   125→                onValidated={(v) => {
   126→   126→                  setSectionValidation(section.id, v)
   127→   127→                  // Always allow the student to continue (validation is informative, not blocking).
   128→   128→                  setStep('draft')
   129→   129→                }}
   130→   130→                onBack={() => setStep('interview')}
   131→   131→              />
   132→   132→            </motion.div>
   133→   133→          )}
   134→   134→          {step === 'draft' && (
   135→   135→            <motion.div
   136→   136→              key="draft"
   137→   137→              initial={{ opacity: 0, x: 10 }}
   138→   138→              animate={{ opacity: 1, x: 0 }}
   139→   139→              exit={{ opacity: 0, x: -10 }}
   140→   140→            >
   141→   141→              <DraftStep
   142→   142→                section={section}
   143→   143→                project={project}
   144→   144→                themeUnderstanding={themeUnderstanding}
   145→   145→                problemContext={problemContext}
   146→   146→                allSections={allSections}
   147→   147→                onInsert={(html) => {
   148→   148→                  // Insert into the editor (which triggers onChange → updateSectionContent)
   149→   149→                  onInsertDraft(html)
   150→   150→                  setStep('humanization')
   151→   151→                }}
   152→   152→                onBack={() => setStep('validation')}
   153→   153→              />
   154→   154→            </motion.div>
   155→   155→          )}
   156→   156→          {step === 'humanization' && (
   157→   157→            <motion.div
   158→   158→              key="humanization"
   159→   159→              initial={{ opacity: 0, x: 10 }}
   160→   160→              animate={{ opacity: 1, x: 0 }}
   161→   161→              exit={{ opacity: 0, x: -10 }}
   162→   162→            >
   163→   163→              <HumanizationStep
   164→   164→                section={section}
   165→   165→                project={project}
   166→   166→                onHumanized={(h) => {
   167→   167→                  setSectionHumanization(section.id, h)
   168→   168→                  // Replace (not append) the editor content with the humanized version.
   169→   169→                  // replaceHtml triggers onChange → updateSectionContent automatically.
   170→   170→                  onReplaceDraft(h.finalHtml)
   171→   171→                  setStep('done')
   172→   172→                }}
   173→   173→                onSkip={() => setStep('done')}
   174→   174→                onBack={() => setStep('draft')}
   175→   175→              />
   176→   176→            </motion.div>
   177→   177→          )}
   178→   178→          {step === 'done' && (
   179→   179→            <motion.div
   180→   180→              key="done"
   181→   181→              initial={{ opacity: 0, x: 10 }}
   182→   182→              animate={{ opacity: 1, x: 0 }}
   183→   183→              exit={{ opacity: 0, x: -10 }}
   184→   184→            >
   185→   185→              <DoneStep
   186→   186→                section={section}
   187→   187→                onClose={onClose}
   188→   188→                onMarkCompleted={() => {
   189→   189→                  setSectionStatus(section.id, 'completed')
   190→   190→                  onClose()
   191→   191→                }}
   192→   192→              />
   193→   193→            </motion.div>
   194→   194→          )}
   195→   195→        </AnimatePresence>
   196→   196→      </div>
   197→   197→    </motion.aside>
   198→   198→  )
   199→   199→}

   205→   205→function PanelHeader({
   206→   206→  step,
   207→   207→  section,
   208→   208→  onClose,
   209→   209→}: {
   210→   210→  step: PanelStep
   211→   211→  section: Section
   212→   212→  onClose: () => void
   213→   213→}) {
   214→   214→  const steps: { id: PanelStep; label: string; icon: any }[] = [
   215→   215→    { id: 'interview', label: 'Entretien', icon: HelpCircle },
   216→   216→    { id: 'validation', label: 'Validation', icon: ShieldCheck },
   217→   217→    { id: 'draft', label: 'Rédaction', icon: PenLine },
   218→   218→    { id: 'humanization', label: 'Humanisation', icon: Wand2 },
   219→   219→    { id: 'done', label: 'Terminé', icon: CheckCircle2 },
   220→   220→  ]
   221→   221→  const currentIdx = steps.findIndex((s) => s.id === step)
   222→   222→
   223→   223→  return (
   224→   224→    <div className="border-b border-border p-3 space-y-2.5">
   225→   225→      <div className="flex items-center justify-between">
   226→   226→        <div className="flex items-center gap-2 min-w-0">
   227→   227→          <Sparkles className="h-4 w-4 text-primary flex-shrink-0" />
   228→   228→          <div className="min-w-0">
   229→   229→            <p className="text-xs font-semibold leading-none">Workflow de section</p>
   230→   230→            <p className="text-[11px] text-muted-foreground truncate mt-0.5">{section.title}</p>
   231→   231→          </div>
   232→   232→        </div>
   233→   233→        <Button variant="ghost" size="sm" onClick={onClose} className="h-7 w-7 p-0 rounded-full">
   234→   234→          <X className="h-4 w-4" />
   235→   235→        </Button>
   236→   236→      </div>
   237→   237→      <div className="flex items-center gap-0.5">
   238→   238→        {steps.map((s, idx) => (
   239→   239→          <React.Fragment key={s.id}>
   240→   240→            <div
   241→   241→              className={cn(
   242→   242→                'flex items-center gap-1 px-1.5 py-0.5 rounded text-[10px] font-medium transition-all',
   243→   243→                idx === currentIdx
   244→   244→                  ? 'bg-primary/10 text-primary'
   245→   245→                  : idx < currentIdx
   246→   246→                  ? 'text-emerald-600 dark:text-emerald-400'
   247→   247→                  : 'text-muted-foreground/40'
   248→   248→              )}
   249→   249→            >
   250→   250→              {idx < currentIdx ? <Check className="h-2.5 w-2.5" /> : <s.icon className="h-2.5 w-2.5" />}
   251→   251→            </div>
   252→   252→            {idx < steps.length - 1 && (
   253→   253→              <div
   254→   254→                className={cn(
   255→   255→                  'w-3 h-px',
   256→   256→                  idx < currentIdx ? 'bg-emerald-500/40' : 'bg-border'
   257→   257→                )}
   258→   258→              />
   259→   259→            )}
   260→   260→          </React.Fragment>
   261→   261→        ))}
   262→   262→      </div>
   263→   263→    </div>
   264→   264→  )
   265→   265→}

   268→   268→// STEP 1 — Interview (4 modes)
   269→   269→// ============================================================================
   270→   270→
   271→   271→type InterviewMode = 'know' | 'dont_know' | 'propose' | 'example'
   272→   272→
   273→   273→function InterviewStep({
   274→   274→  section,
   275→   275→  project,
   276→   276→  themeUnderstanding,
   277→   277→  problemContext,
   278→   278→  onAnswer,
   279→   279→  onSetAnswers,
   280→   280→  onContinue,
   281→   281→}: {
   282→   282→  section: Section
   283→   283→  project: any
   284→   284→  themeUnderstanding: any
   285→   285→  problemContext: any
   286→   286→  onAnswer: (a: InterviewAnswer) => void
   287→   287→  onSetAnswers: (a: InterviewAnswer[]) => void
   288→   288→  onContinue: () => void
   289→   289→}) {
   290→   290→  const [currentQuestion, setCurrentQuestion] = React.useState<{
   291→   291→    questionId: string
   292→   292→    question: string
   293→   293→    helper: string
   294→   294→  } | null>(null)
   295→   295→  const [mode, setMode] = React.useState<InterviewMode | null>(null)
   296→   296→  const [studentInput, setStudentInput] = React.useState('')
   297→   297→  const [loading, setLoading] = React.useState(true)
   298→   298→  const [explanation, setExplanation] = React.useState<string | null>(null)
   299→   299→  const [proposals, setProposals] = React.useState<
   300→   300→    { id: string; label: string; content: string; rationale: string }[]
   301→   301→  >([])
   302→   302→  const [example, setExample] = React.useState<string | null>(null)
   303→   303→  const [error, setError] = React.useState<string | null>(null)
   304→   304→  const [progress, setProgress] = React.useState({ current: 0, total: 5 })
   305→   305→  // Track the number of answers we've already loaded a question for
   306→   306→  const [lastLoadedCount, setLastLoadedCount] = React.useState(section.interviewAnswers.length)
   307→   307→
   308→   308→  async function fetchNextQuestion() {
   309→   309→    setLoading(true)
   310→   310→    setError(null)
   311→   311→    setMode(null)
   312→   312→    setStudentInput('')
   313→   313→    setExplanation(null)
   314→   314→    setProposals([])
   315→   315→    setExample(null)
   316→   316→    try {
   317→   317→      const res = await fetch('/api/ai/section-interview', {
   318→   318→        method: 'POST',
   319→   319→        headers: { 'Content-Type': 'application/json' },
   320→   320→        body: JSON.stringify({
   321→   321→          sectionTitle: section.title,
   322→   322→          sectionDescription: '',
   323→   323→          project,
   324→   324→          themeUnderstanding,
   325→   325→          problemContext,
   326→   326→          answers: section.interviewAnswers,
   327→   327→          action: 'next_question',
   328→   328→        }),
   329→   329→      })
   330→   330→      const data = await res.json()
   331→   331→      if (data.done) {
   332→   332→        // All questions answered — go to validation
   333→   333→        onContinue()
   334→   334→        return
   335→   335→      }
   336→   336→      setCurrentQuestion(data.nextQuestion)
   337→   337→      setProgress(data.progress || progress)
   338→   338→    } catch (e: any) {
   339→   339→      setError(e.message || 'Connexion impossible.')
   340→   340→    } finally {
   341→   341→      setLoading(false)
   342→   342→    }
   343→   343→  }
   344→   344→
   345→   345→  // Initial load
   346→   346→  React.useEffect(() => {
   347→   347→    fetchNextQuestion()
   348→   348→    // eslint-disable-next-line react-hooks/exhaustive-deps
   349→   349→  }, [])
   350→   350→
   351→   351→  // When the answers array grows (i.e. student just answered), fetch the next question.
   352→   352→  // Skip the initial mount (handled above).
   353→   353→  React.useEffect(() => {
   354→   354→    if (section.interviewAnswers.length > lastLoadedCount) {
   355→   355→      setLastLoadedCount(section.interviewAnswers.length)
   356→   356→      fetchNextQuestion()
   357→   357→    }
   358→   358→    // eslint-disable-next-line react-hooks/exhaustive-deps
   359→   359→  }, [section.interviewAnswers.length])
   360→   360→
   361→   361→  async function handleModeAction(selectedMode: InterviewMode) {
   362→   362→    if (!currentQuestion) return
   363→   363→    setMode(selectedMode)
   364→   364→    setLoading(true)
   365→   365→    setError(null)
   366→   366→    try {
   367→   367→      const res = await fetch('/api/ai/section-interview', {
   368→   368→        method: 'POST',
   369→   369→        headers: { 'Content-Type': 'application/json' },
   370→   370→        body: JSON.stringify({
   371→   371→          sectionTitle: section.title,
   372→   372→          project,
   373→   373→          themeUnderstanding,
   374→   374→          problemContext,
   375→   375→          answers: section.interviewAnswers,
   376→   376→          action: selectedMode,
   377→   377→          studentInput: selectedMode === 'know' ? studentInput : undefined,
   378→   378→        }),
   379→   379→      })
   380→   380→      const data = await res.json()
   381→   381→      if (data.error) throw new Error(data.error)
   382→   382→      if (selectedMode === 'know') {
   383→   383→        // Register answer and move on
   384→   384→        onAnswer({
   385→   385→          questionId: currentQuestion.questionId,
   386→   386→          question: currentQuestion.question,
   387→   387→          answer: studentInput.trim(),
   388→   388→        })
   389→   389→        setStudentInput('')
   390→   390→        setMode(null)
   391→   391→        fetchNextQuestion()
   392→   392→      } else if (selectedMode === 'dont_know') {
   393→   393→        setExplanation(data.explanation)
   394→   394→      } else if (selectedMode === 'propose') {
   395→   395→        setProposals(data.proposals || [])
   396→   396→      } else if (selectedMode === 'example') {
   397→   397→        setExample(data.example)
   398→   398→      }
   399→   399→    } catch (e: any) {
   400→   400→      setError(e.message || 'Connexion impossible.')
   401→   401→    } finally {
   402→   402→      setLoading(false)
   403→   403→    }
   404→   404→  }
   405→   405→
   406→   406→  function acceptProposal(p: { id: string; content: string }) {
   407→   407→    if (!currentQuestion) return
   408→   408→    onAnswer({
   409→   409→      questionId: currentQuestion.questionId,
   410→   410→      question: currentQuestion.question,
   411→   411→      answer: p.content,
   412→   412→    })
   413→   413→    setProposals([])
   414→   414→    setMode(null)
   415→   415→    fetchNextQuestion()
   416→   416→  }
   417→   417→
   418→   418→  function handleSkipQuestion() {
   419→   419→    if (!currentQuestion) return
   420→   420→    onAnswer({
   421→   421→      questionId: currentQuestion.questionId,
   422→   422→      question: currentQuestion.question,
   423→   423→      answer: '(non précisé)',
   424→   424→    })
   425→   425→    setMode(null)
   426→   426→    fetchNextQuestion()
   427→   427→  }
   428→   428→
   429→   429→  return (
   430→   430→    <div className="p-4 space-y-4">
   431→   431→      <div>
   432→   432→        <Badge variant="outline" className="mb-2 text-[10px]">
   433→   433→          Question {progress.current + 1} / {progress.total}
   434→   434→        </Badge>
   435→   435→        <p className="text-xs text-muted-foreground mb-1.5">Phase 3 · Entretien structuré</p>
   436→   436→      </div>
   437→   437→
   438→   438→      {loading && !currentQuestion && (
   439→   439→        <div className="flex items-center justify-center py-8">
   440→   440→          <Loader2 className="h-6 w-6 animate-spin text-primary" />
   441→   441→        </div>
   442→   442→      )}
   443→   443→
   444→   444→      {currentQuestion && (
   445→   445→        <motion.div
   446→   446→          initial={{ opacity: 0, y: 5 }}
   447→   447→          animate={{ opacity: 1, y: 0 }}
   448→   448→          className="rounded-xl bg-muted/40 p-3 space-y-1"
   449→   449→        >
   450→   450→          <p className="text-sm font-medium leading-relaxed">{currentQuestion.question}</p>
   451→   451→          {currentQuestion.helper && (
   452→   452→            <p className="text-[11px] text-muted-foreground italic">{currentQuestion.helper}</p>
   453→   453→          )}
   454→   454→        </motion.div>
   455→   455→      )}
   456→   456→
   457→   457→      {error && (
   458→   458→        <div className="rounded-lg border border-red-500/40 bg-red-500/5 p-2.5 text-xs text-red-700 dark:text-red-400 flex items-center gap-2">
   459→   459→          <AlertCircle className="h-3.5 w-3.5 flex-shrink-0" />
   460→   460→          <span className="flex-1">{error}</span>
   461→   461→        </div>
   462→   462→      )}
   463→   463→
   464→   464→      {/* Mode selection (4 modes) */}
   465→   465→      {currentQuestion && !mode && !loading && (
   466→   466→        <div className="grid grid-cols-2 gap-1.5">
   467→   467→          <ModeButton
   468→   468→            icon={<PenLine className="h-3.5 w-3.5" />}
   469→   469→            label="Je sais"
   470→   470→            desc="J'écris ma réponse"
   471→   471→            onClick={() => setMode('know')}
   472→   472→            color="primary"
   473→   473→          />
   474→   474→          <ModeButton
   475→   475→            icon={<HelpCircle className="h-3.5 w-3.5" />}
   476→   476→            label="Je ne sais pas"
   477→   477→            desc="IRIS m'explique"
   478→   478→            onClick={() => handleModeAction('dont_know')}
   479→   479→            color="amber"
   480→   480→          />
   481→   481→          <ModeButton
   482→   482→            icon={<ListChecks className="h-3.5 w-3.5" />}
   483→   483→            label="Propose-moi"
   484→   484→            desc="Plusieurs options"
   485→   485→            onClick={() => handleModeAction('propose')}
   486→   486→            color="violet"
   487→   487→          />
   488→   488→          <ModeButton
   489→   489→            icon={<Eye className="h-3.5 w-3.5" />}
   490→   490→            label="Montre-moi"
   491→   491→            desc="Un exemple concret"
   492→   492→            onClick={() => handleModeAction('example')}
   493→   493→            color="cyan"
   494→   494→          />
   495→   495→        </div>
   496→   496→      )}
   497→   497→
   498→   498→      {/* Mode know: textarea + submit */}
   499→   499→      {mode === 'know' && currentQuestion && (
   500→   500→        <div className="space-y-2">
   501→   501→          <Textarea
   502→   502→            value={studentInput}

   693→   693→  section,
   694→   694→  project,

   696→   696→  onBack,
   697→   697→}: {
   698→   698→  section: Section
   699→   699→  project: any
   700→   700→  onValidated: (v: SectionValidation) => void
   701→   701→  onBack: () => void
   702→   702→}) {
   703→   703→  const [loading, setLoading] = React.useState(false)
   704→   704→  const [validation, setValidation] = React.useState<SectionValidation | null>(section.validation || null)
   705→   705→  const [error, setError] = React.useState<string | null>(null)
   706→   706→
   707→   707→  async function runValidation() {
   708→   708→    setLoading(true)
   709→   709→    setError(null)
   710→   710→    try {
   711→   711→      const res = await fetch('/api/ai/section-interview', {
   712→   712→        method: 'POST',
   713→   713→        headers: { 'Content-Type': 'application/json' },
   714→   714→        body: JSON.stringify({
   715→   715→          sectionTitle: section.title,
   716→   716→          project,
   717→   717→          answers: section.interviewAnswers,
   718→   718→          action: 'validate',
   719→   719→        }),

   725→   725→        feasibility: data.feasibility || { ok: true, notes: '' },

   728→   728→        overallOk: Boolean(data.overallOk),

   740→   740→    <div className="p-4 space-y-4">

   744→   744→        </Badge>
   745→   745→        <p className="text-sm font-semibold">Vérification avant rédaction</p>
   746→   746→        <p className="text-xs text-muted-foreground mt-1">
   747→   747→          Le Contrôleur qualité vérifie la cohérence, la faisabilité, la précision et la logique
   748→   748→          de vos réponses collectées. La rédaction ne peut commencer que si tout est validé.
   749→   749→        </p>
   750→   750→      </div>
   751→   751→
   752→   752→      {!validation && !loading && (
   753→   753→        <Button
   754→   754→          onClick={runValidation}
   755→   755→          className="w-full iris-gradient text-white rounded-full"
   756→   756→        >
   757→   757→          <ShieldCheck className="h-4 w-4 mr-1.5" />
   758→   758→          Lancer la validation
   759→   759→        </Button>
   760→   760→      )}
   761→   761→
   762→   762→      {loading && (
   763→   763→        <div className="flex flex-col items-center justify-center py-8 gap-2">
   764→   764→          <Loader2 className="h-6 w-6 animate-spin text-primary" />
   765→   765→          <p className="text-xs text-muted-foreground italic">Vérification en cours...</p>
   766→   766→        </div>
   767→   767→      )}

   769→   769→      {error && (
   770→   770→        <div className="rounded-lg border border-red-500/40 bg-red-500/5 p-2.5 text-xs text-red-700 dark:text-red-400 flex items-center gap-2">
   771→   771→          <AlertCircle className="h-3.5 w-3.5 flex-shrink-0" />

   777→   777→      )}

   781→   781→          initial={{ opacity: 0, y: 5 }}
   782→   782→          animate={{ opacity: 1, y: 0 }}
   783→   783→          className="space-y-3"
   784→   784→        >
   785→   785→          <ValidationRow label="Cohérence" v={validation.coherence} />
   786→   786→          <ValidationRow label="Faisabilité" v={validation.feasibility} />
   787→   787→          <ValidationRow label="Précision" v={validation.precision} />
   788→   788→          <ValidationRow label="Logique" v={validation.logic} />
   789→   789→
   790→   790→          <div
   791→   791→            className={cn(
   792→   792→              'rounded-xl p-3 border-2',
   793→   793→              validation.overallOk
   794→   794→                ? 'border-emerald-500/40 bg-emerald-500/5'
   795→   795→                : 'border-amber-500/40 bg-amber-500/5'
   796→   796→            )}
   797→   797→          >
   798→   798→            <div className="flex items-start gap-2">
   799→   799→              {validation.overallOk ? (
   800→   800→                <CheckCircle2 className="h-5 w-5 text-emerald-500 flex-shrink-0 mt-0.5" />
   801→   801→              ) : (
   802→   802→                <AlertCircle className="h-5 w-5 text-amber-500 flex-shrink-0 mt-0.5" />
   803→   803→              )}
   804→   804→              <div className="flex-1">
   805→   805→                <p className="text-sm font-semibold">
   806→   806→                  {validation.overallOk ? 'Validation réussie' : 'Validation partielle'}
   807→   807→                </p>
   808→   808→                <p className="text-xs text-muted-foreground mt-0.5">
   809→   809→                  {validation.overallOk
   810→   810→                    ? 'Toutes les conditions sont réunies pour rédiger cette section.'
   811→   811→                    : 'Vous pouvez tout de même continuer, mais prévoyez de compléter les informations manquantes.'}
   812→   812→                </p>
   813→   813→              </div>
   814→   814→            </div>
   815→   815→          </div>
   816→   816→
   817→   817→          {/* Always allow the student to proceed */}
   818→   818→          <Button
   819→   819→            onClick={() => onValidated(validation)}
   820→   820→            className="w-full iris-gradient text-white rounded-full"
   821→   821→          >
   822→   822→            <PenLine className="h-4 w-4 mr-1.5" />
   823→   823→            Continuer vers la rédaction
   824→   824→          </Button>
   825→   825→        </motion.div>
   826→   826→      )}
   827→   827→
   828→   828→      <div className="flex gap-1.5 pt-2 border-t">
   829→   829→        <Button variant="ghost" size="sm" onClick={onBack} className="text-xs">
   830→   830→          ← Revenir à l'entretien
   831→   831→        </Button>
   832→   832→      </div>
   833→   833→    </div>
   834→   834→  )
