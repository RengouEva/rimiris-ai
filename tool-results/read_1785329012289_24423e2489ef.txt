   438→function FormatPanel({
   439→  format,
   440→  project,
   441→  sections,
   442→  draftedCount,
   443→}: {
   444→  format: 'pdf' | 'word' | 'html' | 'markdown'
   445→  project: any
   446→  sections: any[]
   447→  draftedCount: number
   448→}) {
   449→  const [exporting, setExporting] = React.useState(false)
   450→  const [previewOpen, setPreviewOpen] = React.useState(false)
   451→  const [options, setOptions] = React.useState({
   452→    includeTitlePage: true,
   453→    includeTableOfContents: false,
   454→    includePageNumbers: true,
   455→    embedCss: true,
   456→  })
   457→
   458→  const config = {
   459→    pdf: {
   460→      name: 'PDF',
   461→      icon: FileType2,
   462→      color: 'red',
   463→      desc: 'Format portable pour impression et partage. Généré via le moteur d\'impression du navigateur.',
   464→      ext: 'pdf',
   465→      tips: [
   466→        'Le PDF est généré via la fonction d\'impression de votre navigateur.',
   467→        'Dans la boîte d\'impression, choisissez "Enregistrer en PDF" comme destination.',
   468→        'Les marges A4 (25mm) et la pagination sont automatiquement appliquées.',
   469→      ],
   470→    },
   471→    word: {
   472→      name: 'Word',
   473→      icon: FileText,
   474→      color: 'blue',
   475→      desc: 'Document Microsoft Word éditable (.doc). Compatible avec Word, LibreOffice et Google Docs.',
   476→      ext: 'doc',
   477→      tips: [
   478→        'Le fichier .doc est un HTML que Word ouvre nativement comme un document éditable.',
   479→        'Toutes les mises en forme (titres, gras, listes) sont conservées.',
   480→        'Idéal pour remettre un brouillon à votre directeur de mémoire.',
   481→      ],
   482→    },
   483→    html: {
   484→      name: 'HTML',
   485→      icon: FileCode2,
   486→      color: 'orange',
   487→      desc: 'Page web autonome avec CSS inclus. Ouvrable dans tout navigateur.',
   488→      ext: 'html',
   489→      tips: [
   490→        'Le HTML inclut le CSS en ligne (aucun fichier externe nécessaire).',
   491→        'Idéal pour publier en ligne ou partager par email.',
   492→        'Le rendu visuel est identique à l\'aperçu dans IRIS.',
   493→      ],
   494→    },
   495→    markdown: {
   496→      name: 'Markdown',
   497→      icon: Hash,
   498→      color: 'gray',
   499→      desc: 'Texte structuré léger (.md). Compatible avec GitHub, Notion, Obsidian, etc.',
   500→      ext: 'md',
   501→      tips: [
   502→        'Le Markdown préserve la structure (titres, listes, citations) sans formatage visuel.',
   503→        'Idéal pour importer dans un autre outil d\'écriture.',
   504→        'Le HTML des sections est converti en Markdown basique.',
   505→      ],
   506→    },
   507→  }[format]
   508→
   509→  async function handleExport() {
   510→    if (draftedCount === 0) {
   511→      toast.error('Aucune section rédigée à exporter.')
   512→      return
   513→    }
   514→    setExporting(true)
   515→    try {
   516→      // Small delay for UX feedback
   517→      await new Promise((r) => setTimeout(r, 600))
   518→
   519→      let blob: Blob
   520→      let filename: string
   521→
   522→      if (format === 'pdf') {
   523→        // PDF = ouvre la boîte d'impression du navigateur (l'utilisateur choisit "Enregistrer en PDF")
   524→        toast.info('La boîte d\'impression va s\'ouvrir. Choisissez "Enregistrer en PDF".')
   525→        setTimeout(() => window.print(), 500)
   526→        setExporting(false)
   527→        return
   528→      } else if (format === 'word') {
   529→        const html = buildWordHtml(project, sections, options)
   530→        blob = new Blob([html], { type: 'application/msword;charset=utf-8' })
   531→        filename = `${slugify(project.title || 'memoire-iris')}.doc`
   532→      } else if (format === 'html') {
   533→        const html = buildStandaloneHtml(project, sections, options)
   534→        blob = new Blob([html], { type: 'text/html;charset=utf-8' })
   535→        filename = `${slugify(project.title || 'memoire-iris')}.html`
   536→      } else {
   537→        // markdown
   538→        const md = buildMarkdown(project, sections, options)
   539→        blob = new Blob([md], { type: 'text/markdown;charset=utf-8' })
   540→        filename = `${slugify(project.title || 'memoire-iris')}.md`
   541→      }
   542→
   543→      const url = URL.createObjectURL(blob)
   544→      const a = document.createElement('a')
   545→      a.href = url
   546→      a.download = filename
   547→      document.body.appendChild(a)
   548→      a.click()
   549→      document.body.removeChild(a)
   550→      URL.revokeObjectURL(url)
   551→      toast.success(`Export ${config.name} généré`)
   552→    } catch (e: any) {
   553→      toast.error(`Erreur : ${e?.message || 'inconnue'}`)
   554→    } finally {
   555→      setExporting(false)
   556→    }
   557→  }
   558→
   559→  const Icon = config.icon
   560→
   561→  return (
   562→    <div className="space-y-6">
   563→      {/* Header */}
   564→      <div className="flex items-start gap-4">
   565→        <div

   930→function slugify(s: string): string {
   931→  return s
   932→    .toLowerCase()
   933→    .normalize('NFD')
   934→    .replace(/[\u0300-\u036f]/g, '')
   935→    .replace(/[^a-z0-9]+/g, '-')
   936→    .replace(/^-+|-+$/g, '')
   937→    .slice(0, 60) || 'memoire-iris'
   938→}
   939→
   940→function buildSectionsHtml(sections: any[]): string {
   941→  return sections
   942→    .map((s, idx) => {
   943→      if (!s.content || !s.content.trim()) return ''
   944→      return `<h2>${idx + 1}. ${escapeHtml(s.title)}</h2>\n${s.content}\n`
   945→    })
   946→    .filter(Boolean)
   947→    .join('\n')
   948→}

  1120→  })
  1121→  return parts.join('\n')
