   210→function SectionItem({
   211→  section,
   212→  index,
   213→  total,
   214→  active,
   215→  onSelect,
   216→  onRename,
   217→  onDelete,
   218→  onDuplicate,
   219→  onMoveUp,
   220→  onMoveDown,
   221→  onMarkCompleted,
   222→}: {
   223→  section: Section
   224→  index: number
   225→  total: number
   226→  active: boolean
   227→  onSelect: () => void
   228→  onRename: (title: string) => void
   229→  onDelete: () => void
   230→  onDuplicate: () => void
   231→  onMoveUp: () => void
   232→  onMoveDown: () => void
   233→  onMarkCompleted: () => void
   234→}) {
   235→  const [editing, setEditing] = React.useState(false)
   236→  const [draftTitle, setDraftTitle] = React.useState(section.title)
   237→  const status = STATUS_CONFIG[section.status]
   238→
   239→  React.useEffect(() => {
   240→    setDraftTitle(section.title)
   241→  }, [section.title])
   242→
   243→  function commitRename() {
   244→    const t = draftTitle.trim()
   245→    if (t && t !== section.title) {
   246→      onRename(t)
   247→    } else {
   248→      setDraftTitle(section.title)
   249→    }
   250→    setEditing(false)
   251→  }
   252→
   253→  return (
   254→    <motion.div
   255→      layout
   256→      initial={{ opacity: 0, x: -10 }}
   257→      animate={{ opacity: 1, x: 0 }}
   258→      exit={{ opacity: 0, x: -10 }}
   259→      className={cn(
   260→        'group relative rounded-lg transition-all',
   261→        active ? 'bg-card shadow-sm border border-primary/30' : 'hover:bg-card/60 border border-transparent'
   262→      )}
   263→    >
   264→      <div
   265→        onClick={onSelect}
   266→        className="w-full text-left p-2.5 flex items-start gap-2 cursor-pointer"
   267→      >
   268→        <div className="flex flex-col items-center gap-1 pt-0.5">
   269→          <span className="text-[10px] font-mono text-muted-foreground">{index + 1}</span>
   270→          <span className={cn('w-1.5 h-1.5 rounded-full', status.dot)} />
   271→        </div>
   272→        <div className="flex-1 min-w-0">
   273→          {editing ? (
   274→            <input
   275→              autoFocus
   276→              value={draftTitle}
   277→              onChange={(e) => setDraftTitle(e.target.value)}
   278→              onBlur={commitRename}
   279→              onKeyDown={(e) => {
   280→                if (e.key === 'Enter') commitRename()
   281→                if (e.key === 'Escape') {
   282→                  setDraftTitle(section.title)
   283→                  setEditing(false)
   284→                }
   285→              }}
   286→              onClick={(e) => e.stopPropagation()}
   287→              className="w-full text-sm font-medium bg-transparent border-b border-primary outline-none"
   288→            />
   289→          ) : (
   290→            <p className="text-sm font-medium truncate">{section.title}</p>
   291→          )}
   292→          <div className="flex items-center gap-2 mt-0.5">
   293→            <span className={cn('text-[10px]', status.color)}>{status.label}</span>
   294→            {section.wordCount > 0 && (
   295→              <span className="text-[10px] text-muted-foreground">{section.wordCount} mots</span>
   296→            )}
   297→          </div>
   298→        </div>
   299→
   300→        <DropdownMenu>
   301→          <DropdownMenuTrigger asChild>
   302→            <button
   303→              onClick={(e) => e.stopPropagation()}
   304→              className="opacity-0 group-hover:opacity-100 transition-opacity p-1 rounded hover:bg-muted"
   305→            >
   306→              <MoreVertical className="h-3.5 w-3.5 text-muted-foreground" />
   307→            </button>
   308→          </DropdownMenuTrigger>
   309→          <DropdownMenuContent align="end" className="w-44">
   310→            <DropdownMenuItem onClick={() => setEditing(true)}>
   311→              <Edit3 className="h-3.5 w-3.5 mr-2" />
   312→              Renommer
   313→            </DropdownMenuItem>
   314→            <DropdownMenuItem onClick={onDuplicate}>
   315→              <Copy className="h-3.5 w-3.5 mr-2" />
   316→              Dupliquer
   317→            </DropdownMenuItem>
   318→            <DropdownMenuItem onClick={onMarkCompleted}>
   319→              <CheckCircle2 className="h-3.5 w-3.5 mr-2" />
   320→              {section.status === 'completed' ? 'Marquer brouillon' : 'Marquer terminé'}
   321→            </DropdownMenuItem>
   322→            <DropdownMenuSeparator />
   323→            <DropdownMenuItem
   324→              onClick={onMoveUp}
   325→              disabled={index === 0}
   326→            >
   327→              <ChevronUp className="h-3.5 w-3.5 mr-2" />
   328→              Monter
   329→            </DropdownMenuItem>
   330→            <DropdownMenuItem
   331→              onClick={onMoveDown}
   332→              disabled={index === total - 1}
   333→            >
   334→              <ChevronDown className="h-3.5 w-3.5 mr-2" />
   335→              Descendre
   336→            </DropdownMenuItem>
   337→            <DropdownMenuSeparator />
   338→            <DropdownMenuItem onClick={onDelete} className="text-red-500">
   339→              <Trash2 className="h-3.5 w-3.5 mr-2" />
   340→              Supprimer
   341→            </DropdownMenuItem>
   342→          </DropdownMenuContent>
   343→        </DropdownMenu>
   344→      </div>
   345→    </motion.div>
   346→  )
   347→}

   526→function AIPanel({
   527→  section,
   528→  project,
   529→  allSections,
   530→  onClose,
   531→}: {
   532→  section: Section
   533→  project: any
   534→  allSections: { title: string; content: string }[]
   535→  onClose: () => void
   536→}) {
   537→  const { addMessage, blockedMode, setBlockedMode } = useIrisStore()
   538→  const [input, setInput] = React.useState('')
   539→  const [loading, setLoading] = React.useState(false)
   540→  const scrollRef = React.useRef<HTMLDivElement>(null)
   541→
   542→  React.useEffect(() => {
   543→    if (scrollRef.current) {
   544→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight
   545→    }
   546→  }, [section.messages.length, loading])
   547→
   548→  async function send(text: string) {
   549→    if (!text.trim() || loading) return
   550→
   551→    addMessage(section.id, { role: 'user', content: text.trim() })
   552→    setInput('')
   553→    setLoading(true)
   554→
   555→    try {
   556→      const history = section.messages
   557→        .filter((m) => m.role !== 'system')
   558→        .slice(-6)
   559→        .map((m) => ({ role: m.role as 'user' | 'assistant', content: m.content }))
   560→
   561→      const res = await fetch('/api/ai/chat', {
   562→        method: 'POST',
   563→        headers: { 'Content-Type': 'application/json' },
   564→        body: JSON.stringify({
   565→          sectionTitle: section.title,
   566→          sectionContent: section.content,
   567→          templateRef: section.templateRef,
   568→          project,
   569→          allSections,
   570→          history,
   571→          userMessage: text.trim(),
   572→          blockedMode,
   573→        }),
   574→      })
   575→      const data = await res.json()
   576→
   577→      addMessage(section.id, {
   578→        role: 'assistant',
   579→        content: data.reply || '...',
   580→        agent: data.agent?.id,
   581→      })
   582→
   583→      if (blockedMode) setBlockedMode(false)
   584→    } catch {
   585→      addMessage(section.id, {
   586→        role: 'assistant',
   587→        content: "Erreur de communication. Reformulez votre demande.",
   588→      })
   589→    } finally {
   590→      setLoading(false)
   591→    }
   592→  }
   593→
   594→  function handleKeyDown(e: React.KeyboardEvent<HTMLTextAreaElement>) {
   595→    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   596→      e.preventDefault()
   597→      send(input)
   598→    }
   599→  }
   600→
   601→  // Determine agent
   602→  const agent = React.useMemo(() => {
   603→    if (section.templateRef) {
   604→      return getAgentForChapter(section.templateRef)
   605→    }
   606→    return AGENTS[0] // Pr. IRIS by default
   607→  }, [section.templateRef])
   608→
   609→  const suggestions = blockedMode
   610→    ? [
   611→        "Je ne sais pas par où commencer",
   612→        "Je n'arrive pas à formuler cette partie",
   613→        "Aidez-moi à structurer mes idées",
   614→      ]
   615→    : [
   616→        "Relis mon brouillon et donne-moi 3 conseils",
   617→        "Qu'est-ce qui manque à cette section ?",
   618→        "Aide-moi à formuler une phrase d'introduction",
   619→        "Comment améliorer la cohérence avec le reste ?",
   620→      ]
   621→
   622→  return (
   623→    <motion.aside
   624→      initial={{ x: '100%' }}
   625→      animate={{ x: 0 }}
   626→      exit={{ x: '100%' }}
   627→      transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   628→      className="w-full sm:w-96 border-l border-border bg-card flex flex-col flex-shrink-0 absolute right-0 top-0 bottom-0 sm:relative z-30"
   629→    >
   630→      {/* Header */}
   631→      <div className="p-3 border-b border-border flex items-center gap-2">
   632→        <div className="w-8 h-8 rounded-full iris-gradient flex items-center justify-center flex-shrink-0">
   633→          {blockedMode ? (
   634→            <Brain className="h-4 w-4 text-white" />
   635→          ) : (
   636→            <Sparkles className="h-4 w-4 text-white" />
   637→          )}
   638→        </div>
   639→        <div className="flex-1 min-w-0">
   640→          <p className="text-sm font-semibold">
   641→            {blockedMode ? 'Mode débloqué' : agent.name}
   642→          </p>
   643→          <p className="text-xs text-muted-foreground truncate">
   644→            {blockedMode ? 'IRIS vous aide à repartir' : agent.role}
   645→          </p>
   646→        </div>
   647→        {blockedMode && (
   648→          <Button
   649→            size="sm"
   650→            variant="ghost"
   651→            onClick={() => setBlockedMode(false)}
   652→            className="text-xs h-7"
   653→          >
   654→            Retour normal
   655→          </Button>
   656→        )}
   657→        <button onClick={onClose} className="p-1.5 rounded hover:bg-muted">
   658→          <X className="h-4 w-4" />
   659→        </button>
   660→      </div>
   661→
   662→      {/* Context banner */}
   663→      <div className="px-3 py-2 bg-muted/40 border-b border-border">
   664→        <p className="text-[10px] uppercase tracking-wide text-muted-foreground mb-0.5">
   665→          Section en cours
   666→        </p>
   667→        <p className="text-xs font-medium truncate">{section.title}</p>
   668→      </div>
   669→
   670→      {/* Messages */}
   671→      <div ref={scrollRef} className="flex-1 overflow-y-auto p-3 space-y-3">
   672→        {section.messages.length === 0 ? (
   673→          <div className="text-center py-6">
   674→            <p className="text-sm text-muted-foreground mb-4">
   675→              {blockedMode
   676→                ? "Décrivez votre blocage, ou choisissez une suggestion."
   677→                : `Posez votre question à ${agent.name}, ou choisissez une suggestion.`}
   678→            </p>
   679→            <div className="space-y-1.5">
   680→              {suggestions.map((s, idx) => (
   681→                <button
   682→                  key={idx}
   683→                  onClick={() => send(s)}
   684→                  className="w-full text-left text-xs p-2.5 rounded-lg border border-border bg-background hover:border-primary/40 hover:bg-primary/5 transition-colors"
   685→                >
   686→                  <span className="text-primary mr-1.5">→</span>
   687→                  {s}
   688→                </button>
   689→              ))}
   690→            </div>
   691→          </div>
   692→        ) : (
   693→          section.messages.map((msg) => {
   694→            const isUser = msg.role === 'user'
   695→            const msgAgent = msg.agent ? AGENTS.find((a) => a.id === msg.agent) : null
   696→            return (
   697→              <motion.div
   698→                key={msg.id}
   699→                initial={{ opacity: 0, y: 5 }}
   700→                animate={{ opacity: 1, y: 0 }}
   701→                className={cn('flex flex-col gap-1', isUser ? 'items-end' : 'items-start')}
   702→              >
   703→                {msgAgent && !isUser && (
   704→                  <span className="text-[10px] font-semibold text-primary px-1">
   705→                    {msgAgent.name}
   706→                  </span>
   707→                )}
   708→                <div
   709→                  className={cn(
   710→                    'rounded-2xl p-3 text-sm max-w-[90%]',
   711→                    isUser
   712→                      ? 'bg-primary text-primary-foreground rounded-tr-sm'
   713→                      : 'bg-muted rounded-tl-sm'
   714→                  )}
   715→                >
   716→                  <p className="whitespace-pre-wrap leading-relaxed">{msg.content}</p>
   717→                </div>
   718→              </motion.div>
   719→            )
   720→          })
   721→        )}
   722→
   723→        {loading && (
   724→          <div className="flex items-start gap-2">
   725→            <div className="w-7 h-7 rounded-full iris-gradient flex items-center justify-center flex-shrink-0">
   726→              <Sparkles className="h-3.5 w-3.5 text-white" />
   727→            </div>
   728→            <div className="bg-muted rounded-2xl rounded-tl-sm p-3">
   729→              <div className="flex items-center gap-1">
   730→                <span className="typing-dot w-1.5 h-1.5 rounded-full bg-muted-foreground" />
   731→                <span className="typing-dot w-1.5 h-1.5 rounded-full bg-muted-foreground" />
   732→                <span className="typing-dot w-1.5 h-1.5 rounded-full bg-muted-foreground" />
   733→              </div>
   734→            </div>
   735→          </div>
   736→        )}
   737→      </div>
   738→
   739→      {/* Input */}
   740→      <div className="border-t border-border p-3 space-y-2">
   741→        <Textarea
   742→          value={input}
   743→          onChange={(e) => setInput(e.target.value)}
   744→          onKeyDown={handleKeyDown}
   745→          placeholder={blockedMode ? "Décrivez votre blocage..." : "Votre question..."}
   746→          className="resize-none min-h-16 max-h-32 text-sm"
   747→          disabled={loading}
   748→        />
   749→        <div className="flex items-center justify-between">
   750→          <span className="text-[10px] text-muted-foreground">Cmd/Ctrl+Entrée</span>
   751→          <Button
   752→            onClick={() => send(input)}
   753→            disabled={!input.trim() || loading}
   754→            size="sm"
   755→            className="rounded-full iris-gradient text-white"
   756→          >
   757→            {loading ? (
   758→              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   759→            ) : (
   760→              <>
   761→                <Send className="h-3.5 w-3.5 mr-1" />
   762→                Envoyer
   763→              </>
   764→            )}
   765→          </Button>
   766→        </div>
   767→      </div>
   768→    </motion.aside>
   769→  )
   770→}

   776→function timeAgo(ts: number): string {
   777→  const diff = Date.now() - ts
   778→  if (diff < 60_000) return 'à l\'instant'
   779→  if (diff < 3_600_000) return `il y a ${Math.floor(diff / 60_000)} min`
   780→  if (diff < 86_400_000) return `il y a ${Math.floor(diff / 3_600_000)} h`
   781→  return `il y a ${Math.floor(diff / 86_400_000)} j`
   782→}
