   231→function PrintableMemoire({ project, sections }: { project: any; sections: any[] }) {
   232→  return (
   233→    <div className="printable-memoire hidden print:block">
   234→      {/* Page de titre */}

   236→        <h1>{project.title || 'Mémoire sans titre'}</h1>
   237→        <p className="subtitle">{project.university || ''}</p>
   238→        <p>{project.faculty || ''}</p>
   239→        <p>
   240→          {project.level || ''}
   241→          {project.filiere ? ` · ${project.filiere}` : ''}
   242→        </p>
   243→        <p className="norm">Norme : {project.norme || 'APA'}</p>
   244→      </div>
   245→
   246→      {/* Sections */}
   247→      {sections.map((s, idx) => {
   248→        if (!s.content || !s.content.trim()) return null
   249→        return (
   250→          <div key={s.id} className="printable-section">
   251→            <h2>
   252→              {idx + 1}. {s.title}
   253→            </h2>
   254→            <div dangerouslySetInnerHTML={{ __html: s.content }} />
   255→          </div>
   256→        )
   257→      })}
   258→    </div>
   259→  )
   260→}

   262→// ============================================================================
   263→// PreviewPanel — Aperçu A4 paginé avec contrôle de zoom (défaut 100%)
   264→// Affiche le mémoire tel qu'il sera exporté : page de titre + sections
   265→// dans des "pages A4" visuelles, avec barre de zoom collée en haut.
   266→// ============================================================================
   267→
   268→function PreviewPanel({ project, sections }: { project: any; sections: any[] }) {
   269→  // Zoom par défaut : 100%. L'utilisateur peut ajuster entre 50% et 200%.
   270→  const [zoom, setZoom] = React.useState<number>(100)
   271→  const drafted = sections.filter((s) => s.content && s.content.trim())
   272→
   273→  function changeZoom(delta: number) {
   274→    setZoom((z) => Math.min(200, Math.max(50, z + delta)))
   275→  }
   276→
   277→  function resetZoom() {
   278→    setZoom(100)
   279→  }
   280→
   281→  return (
   282→    <div className="flex flex-col h-full">
   283→      {/* Barre de zoom — collée en haut, ne défile pas */}
   284→      <div className="sticky top-0 z-10 border-b border-border bg-background/95 backdrop-blur-sm px-4 py-2 flex items-center justify-between gap-3">
   285→        <div className="flex items-center gap-2 min-w-0">
   286→          <Eye className="h-4 w-4 text-primary flex-shrink-0" />
   287→          <div className="min-w-0">
   288→            <p className="text-sm font-semibold leading-none truncate">
   289→              Aperçu — {project.title || 'Mémoire sans titre'}
   290→            </p>
   291→            <p className="text-[11px] text-muted-foreground mt-0.5 leading-none">
   292→              {drafted.length} section{drafted.length > 1 ? 's' : ''} · {sections.reduce((sum, s) => sum + s.wordCount, 0)} mots
   293→            </p>
   294→          </div>
   295→        </div>
   296→
   297→        {/* Contrôles de zoom */}
   298→        <div className="flex items-center gap-1.5 flex-shrink-0">
   299→          <Button
   300→            variant="outline"
   301→            size="sm"
   302→            onClick={() => changeZoom(-10)}
   303→            disabled={zoom <= 50}
   304→            className="h-8 w-8 p-0"
   305→            title="Dézoomer"
   306→          >
   307→            <ZoomOut className="h-3.5 w-3.5" />
   308→          </Button>
   309→          <button
   310→            onClick={resetZoom}
   311→            className="h-8 min-w-[60px] px-2 rounded-md border border-border bg-background text-xs font-medium tabular-nums hover:bg-muted transition-colors"
   312→            title="Réinitialiser le zoom à 100%"
   313→          >
   314→            {zoom}%
   315→          </button>
   316→          <Button
   317→            variant="outline"
   318→            size="sm"
   319→            onClick={() => changeZoom(10)}
   320→            disabled={zoom >= 200}
   321→            className="h-8 w-8 p-0"
   322→            title="Zoomer"
   323→          >
   324→            <ZoomIn className="h-3.5 w-3.5" />
   325→          </Button>
   326→          {/* Slider de zoom */}
   327→          <input
   328→            type="range"
   329→            min={50}
   330→            max={200}
   331→            step={10}
   332→            value={zoom}
   333→            onChange={(e) => setZoom(parseInt(e.target.value))}
   334→            className="w-24 accent-primary hidden sm:block"
   335→            title={`Zoom : ${zoom}%`}
   336→          />
   337→          <Button
   338→            variant="ghost"
   339→            size="sm"
   340→            onClick={() => setZoom(100)}
   341→            className="h-8 px-2 hidden sm:flex"
   342→            title="Réinitialiser"
   343→          >
   344→            <Maximize2 className="h-3.5 w-3.5" />
   345→          </Button>
   346→        </div>
   347→      </div>
   348→
   349→      {/* Zone défilante contenant les pages A4 */}
   350→      <div className="flex-1 overflow-y-auto bg-muted/40 p-4 sm:p-8">
   351→        {drafted.length === 0 ? (
   352→          <div className="flex flex-col items-center justify-center h-full text-center max-w-md mx-auto">
   353→            <div className="w-14 h-14 rounded-2xl bg-muted flex items-center justify-center mb-3">
   354→              <Eye className="h-7 w-7 text-muted-foreground" />
   355→            </div>
   356→            <p className="font-semibold text-base mb-1">Rien à afficher</p>
   357→            <p className="text-sm text-muted-foreground">
   358→              Aucune section n'a encore été rédigée. Retournez au mémoire pour
   359→              commencer l'écriture, puis revenez ici pour voir l'aperçu.
   360→            </p>
   361→          </div>
   362→        ) : (
   363→          <div
   364→            className="flex flex-col items-center gap-6 mx-auto"
   365→            style={{
   366→              transform: `scale(${zoom / 100})`,
   367→              transformOrigin: 'top center',
   368→              transition: 'transform 0.2s ease',
   369→            }}
   370→          >
   371→            {/* Page de titre */}
   372→            <PreviewPage>
   373→              <div className="flex flex-col items-center justify-center text-center py-16">
   374→                <h1 className="font-bold text-3xl mb-6" style={{ fontFamily: 'Inter, sans-serif' }}>
   375→                  {project.title || 'Mémoire sans titre'}
   376→                </h1>
   377→                <p className="text-lg text-muted-foreground mb-1">
   378→                  {project.university || ''}
   379→                </p>
   380→                <p className="text-base text-muted-foreground mb-1">
   381→                  {project.faculty || ''}
   382→                </p>
   383→                <p className="text-base text-muted-foreground mb-6">
   384→                  {project.level || ''}
   385→                  {project.filiere ? ` · ${project.filiere}` : ''}
   386→                </p>
   387→                <p className="text-sm text-muted-foreground italic">
   388→                  Norme : {project.norme || 'APA'}
   389→                </p>
   390→              </div>
   391→            </PreviewPage>
   392→
   393→            {/* Pages de contenu */}
   394→            {drafted.map((s, idx) => (
   395→              <PreviewPage key={s.id}>
   396→                <h2
   397→                  className="font-bold text-xl mb-3"
   398→                  style={{ fontFamily: 'Inter, sans-serif' }}
   399→                >
   400→                  {idx + 1}. {s.title}
   401→                </h2>
   402→                <div
   403→                  className="prose-iris text-sm leading-relaxed"
   404→                  dangerouslySetInnerHTML={{ __html: s.content }}
   405→                />
   406→              </PreviewPage>
   407→            ))}
   408→          </div>
   409→        )}
   410→      </div>
   411→    </div>
   412→  )
   413→}

  1302→function buildMarkdown(
  1303→  project: any,
  1304→  sections: any[],
  1305→  options: { includeTitlePage: boolean; includeTableOfContents: boolean }
  1306→): string {
  1307→  const parts: string[] = []
  1308→  if (options.includeTitlePage) {
  1309→    parts.push(`# ${project.title || 'Mémoire sans titre'}`)
  1310→    parts.push('')
  1311→    parts.push(`**${project.university || ''}**  `)
  1312→    parts.push(`**${project.faculty || ''}**  `)
  1313→    parts.push(`**${project.level || ''} · ${project.filiere || ''}**  `)
  1314→    parts.push('')
  1315→    parts.push(`*Norme : ${project.norme || 'APA'}*`)
  1316→    parts.push('')
  1317→    parts.push('---')
  1318→    parts.push('')
  1319→  }
  1320→  if (options.includeTableOfContents) {
  1321→    parts.push('## Table des matières')
  1322→    parts.push('')
  1323→    sections.forEach((s, idx) => {
  1324→      parts.push(`${idx + 1}. ${s.title}`)
  1325→    })
  1326→    parts.push('')
  1327→    parts.push('---')
  1328→    parts.push('')
  1329→  }
  1330→  sections.forEach((s, idx) => {
  1331→    if (!s.content || !s.content.trim()) return
  1332→    parts.push(`## ${idx + 1}. ${s.title}`)
  1333→    parts.push('')
  1334→    // Convert HTML → Markdown (basic)
  1335→    let md = s.content
  1336→    md = md.replace(/<h1[^>]*>(.*?)<\/h1>/gi, '# $1\n')
  1337→    md = md.replace(/<h2[^>]*>(.*?)<\/h2>/gi, '## $1\n')
  1338→    md = md.replace(/<h3[^>]*>(.*?)<\/h3>/gi, '### $1\n')
  1339→    md = md.replace(/<strong[^>]*>(.*?)<\/strong>/gi, '**$1**')
  1340→    md = md.replace(/<b[^>]*>(.*?)<\/b>/gi, '**$1**')
  1341→    md = md.replace(/<em[^>]*>(.*?)<\/em>/gi, '*$1*')
  1342→    md = md.replace(/<i[^>]*>(.*?)<\/i>/gi, '*$1*')
  1343→    md = md.replace(/<u[^>]*>(.*?)<\/u>/gi, '$1')
  1344→    md = md.replace(/<blockquote[^>]*>([\s\S]*?)<\/blockquote>/gi, '> $1\n')
  1345→    md = md.replace(/<li[^>]*>(.*?)<\/li>/gi, '- $1\n')
  1346→    md = md.replace(/<\/?(ul|ol)[^>]*>/gi, '\n')
  1347→    md = md.replace(/<p[^>]*>(.*?)<\/p>/gi, '$1\n\n')
  1348→    md = md.replace(/<br\s*\/?>/gi, '\n')
  1349→    md = md.replace(/<[^>]+>/g, '') // strip remaining tags
  1350→    md = md.replace(/\n{3,}/g, '\n\n')
  1351→    parts.push(md.trim())
  1352→    parts.push('')
  1353→  })
  1354→  return parts.join('\n')
  1355→}

  1357→function escapeHtml(s: string): string {
  1358→  return (s || '')
  1359→    .replace(/&/g, '&amp;')
  1360→    .replace(/</g, '&lt;')
  1361→    .replace(/>/g, '&gt;')
  1362→    .replace(/"/g, '&quot;')
  1363→}
