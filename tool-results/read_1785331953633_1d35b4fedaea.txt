    75→// ============================================================================
    76→// A4Editor — WYSIWYG editor that mimics an A4 page
    77→// - 210mm × 297mm aspect ratio
    78→// - Standard academic margins (25mm top/sides, 30mm bottom — clearly visualized)
    79→// - Justified serif body text, sans-serif headings
    80→// - Two-row rich toolbar (NO scroll, all tools visible):
    81→//     Row 1: undo/redo | font family | font size | bold/italic/underline/strike |
    82→//            subscript/superscript | case (upper/lower/title) | highlight
    83→//     Row 2: paragraph/H1/H2/H3 | lists/quote | alignment | watermark | select all/clear | erase
    84→// - Per-button active state indicators (voyant lumineux) tracked via
    85→//   document.queryCommandState + selectionchange event.
    86→// - Optional watermark (filigrane) rendered behind the page content.
    87→// - Persists HTML to the store.
    88→// ============================================================================
    89→

    95→  getPageCount: () => number
    96→}

   194→  function A4Editor({ value, onChange, editable = true, showToolbar = true, onPageCountChange }, ref) {
   195→    const editorRef = React.useRef<HTMLDivElement>(null)
   196→    const lastValueRef = React.useRef<string>(value)
   197→    const lastPageCountRef = React.useRef<number>(1)
   198→    const [pageCountState, setPageCountState] = React.useState<number>(1)
   199→
   200→    // Active formatting state — refreshed on selectionchange and on input.
   201→    const [activeState, setActiveState] = React.useState({
   202→      bold: false,
   203→      italic: false,
   204→      underline: false,
   205→      strikeThrough: false,
   206→      subscript: false,
   207→      superscript: false,
   208→      insertUnorderedList: false,
   209→      insertOrderedList: false,
   210→      justifyLeft: false,
   211→      justifyCenter: false,
   212→      justifyRight: false,
   213→      justifyFull: true, // default for academic body text
   214→      block: 'p', // 'p' | 'h1' | 'h2' | 'h3' | 'blockquote'
   215→    })
   216→
   217→    // Watermark state — applied to the A4 page background.
   218→    const [watermark, setWatermark] = React.useState<{
   219→      enabled: boolean
   220→      text: string
   221→      opacity: number
   222→      angle: number
   223→    }>({
   224→      enabled: false,
   225→      text: 'CONFIDENTIEL',
   226→      opacity: 0.08,
   227→      angle: -30,
   228→    })
   229→
   230→    // Find & Replace state — drives the FindReplaceButton popover.
   231→    const [findState, setFindState] = React.useState<{
   232→      query: string
   233→      replacement: string
   234→      count: number
   235→      current: number
   236→    }>({ query: '', replacement: '', count: 0, current: 0 })
   237→
   238→    // Ref to the last match found — used by replace() to know which text node to update.
   239→    const lastMatchRef = React.useRef<Range | null>(null)

   241→    // ----------------------------------------------------------------------
   242→    // Pagination : measure content height, derive page count, notify parent
   243→    // ----------------------------------------------------------------------
   244→    function recomputePageCount() {
   245→      const el = editorRef.current
   246→      if (!el) return
   247→      const px = el.scrollHeight
   248→      const count = Math.max(1, Math.ceil(px / A4_PAGE_HEIGHT_PX))
   249→      if (count !== lastPageCountRef.current) {
   250→        lastPageCountRef.current = count
   251→        setPageCountState(count)
   252→        onPageCountChange?.(count)
   253→      }
   254→    }
   255→
   256→    // ----------------------------------------------------------------------
   257→    // Refresh active formatting state from the current selection.
   258→    // Uses document.queryCommandState for the boolean commands and a manual
   259→    // check on the closest block ancestor for formatBlock.
   260→    // ----------------------------------------------------------------------
   261→    function refreshActiveState() {
   262→      if (!editorRef.current) return
   263→      // Only refresh when the selection is inside our editor
   264→      const sel = window.getSelection()
   265→      if (!sel || sel.rangeCount === 0) return
   266→      const range = sel.getRangeAt(0)
   267→      if (!editorRef.current.contains(range.commonAncestorContainer)) return
   268→
   269→      const safe = (cmd: string) => {
   270→        try {
   271→          return document.queryCommandState(cmd)
   272→        } catch {
   273→          return false
   274→        }
   275→      }
   276→
   277→      // Determine the block tag of the current selection
   278→      let node: Node | null = range.commonAncestorContainer
   279→      let blockTag = 'p'
   280→      while (node && node !== editorRef.current) {
   281→        if (node.nodeType === 1) {
   282→          const tag = (node as HTMLElement).tagName.toLowerCase()
   283→          if (['h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'blockquote', 'p', 'div', 'li'].includes(tag)) {
   284→            blockTag = tag === 'div' || tag === 'li' ? 'p' : tag
   285→            break
   286→          }
   287→        }
   288→        node = node.parentNode
   289→      }
   290→
   291→      setActiveState({
   292→        bold: safe('bold'),
   293→        italic: safe('italic'),
   294→        underline: safe('underline'),
   295→        strikeThrough: safe('strikeThrough'),
   296→        subscript: safe('subscript'),
   297→        superscript: safe('superscript'),
   298→        insertUnorderedList: safe('insertUnorderedList'),
   299→        insertOrderedList: safe('insertOrderedList'),
   300→        justifyLeft: safe('justifyLeft'),
   301→        justifyCenter: safe('justifyCenter'),
   302→        justifyRight: safe('justifyRight'),
   303→        justifyFull: safe('justifyFull'),
   304→        block: blockTag as any,
   305→      })
   306→    }
   307→
   308→    // Expose imperative API
   309→    React.useImperativeHandle(ref, () => ({
   310→      insertHtml: (html: string) => {
   311→        const el = editorRef.current
   312→        if (!el) return
   313→        el.focus()
   314→        const sel = window.getSelection()
   315→        const hasSelection = sel && sel.rangeCount > 0 && !sel.isCollapsed
   316→        if (hasSelection) {
   317→          const range = sel.getRangeAt(0)
   318→          if (el.contains(range.commonAncestorContainer)) {
   319→            range.deleteContents()
   320→            const fragment = range.createContextualFragment(html)
   321→            const lastNode = fragment.lastChild
   322→            range.insertNode(fragment)
   323→            if (lastNode) {
   324→              range.setStartAfter(lastNode)
   325→              range.setEndAfter(lastNode)
   326→              sel.removeAllRanges()
   327→              sel.addRange(range)
   328→            }
   329→          } else {
   330→            appendHtml(el, html)
   331→          }
   332→        } else {
   333→          appendHtml(el, html)
   334→        }
   335→        const newHtml = el.innerHTML
   336→        lastValueRef.current = newHtml
   337→        onChange(newHtml)
   338→      },
   339→      replaceHtml: (html: string) => {
   340→        const el = editorRef.current
   341→        if (!el) return
   342→        el.innerHTML = html || ''
   343→        lastValueRef.current = html || ''
   344→        onChange(html || '')
   345→        el.focus()
   346→        requestAnimationFrame(() => recomputePageCount())
   347→      },
   348→      focus: () => editorRef.current?.focus(),
   349→      getPageCount: () => lastPageCountRef.current,
   350→    }))
   351→
   352→    // ----------------------------------------------------------------------
   353→    // Recompute page count when value changes (external updates, e.g., AI draft)
   354→    // ----------------------------------------------------------------------
   355→    React.useEffect(() => {
   356→      recomputePageCount()
   357→    }, [value])
   358→
   359→    React.useEffect(() => {
   360→      const onResize = () => recomputePageCount()
   361→      window.addEventListener('resize', onResize)
   362→      // Listen for selection changes to update active state indicators
   363→      const onSelChange = () => refreshActiveState()
   364→      document.addEventListener('selectionchange', onSelChange)
   365→      const el = editorRef.current
   366→      let ro: ResizeObserver | null = null
   367→      if (el && typeof ResizeObserver !== 'undefined') {
   368→        ro = new ResizeObserver(() => recomputePageCount())
   369→        ro.observe(el)
   370→      }
   371→      requestAnimationFrame(() => recomputePageCount())
   372→      return () => {
   373→        window.removeEventListener('resize', onResize)
   374→        document.removeEventListener('selectionchange', onSelChange)
   375→        ro?.disconnect()
   376→      }
   377→      // eslint-disable-next-line react-hooks/exhaustive-deps
   378→    }, [])
   379→
   380→    // Sync external value -> editor (only when it really changed externally)
   381→    React.useEffect(() => {
   382→      const el = editorRef.current
   383→      if (!el) return
   384→      if (value !== lastValueRef.current && value !== el.innerHTML) {
   385→        el.innerHTML = value || ''
   386→        lastValueRef.current = value
   387→      }
   388→    }, [value])
   389→
   390→    // Initialize once
   391→    React.useEffect(() => {
   392→      const el = editorRef.current
   393→      if (!el) return
   394→      if (el.innerHTML !== (value || '')) {
   395→        el.innerHTML = value || ''
   396→        lastValueRef.current = value || ''
   397→      }
   398→      // eslint-disable-next-line react-hooks/exhaustive-deps
   399→    }, [])
   400→
   401→    function handleInput() {
   402→      const el = editorRef.current
   403→      if (!el) return
   404→      const html = el.innerHTML
   405→      lastValueRef.current = html
   406→      onChange(html)
   407→      recomputePageCount()
   408→      refreshActiveState()
   409→    }
   410→
   411→    function exec(command: string, val?: string) {
   412→      editorRef.current?.focus()
   413→      document.execCommand(command, false, val)
   414→      handleInput()
   415→    }
   416→
   417→    function setBlock(tag: 'h1' | 'h2' | 'h3' | 'p' | 'blockquote') {
   418→      editorRef.current?.focus()
   419→      try {
   420→        document.execCommand('formatBlock', false, `<${tag}>`)
   421→      } catch {
   422→        // fallback noop
   423→      }
   424→      handleInput()
   425→    }
   426→
   427→    function clearFormatting() {
   428→      editorRef.current?.focus()
   429→      document.execCommand('formatBlock', false, '<p>')
   430→      document.execCommand('removeFormat')
   431→      // Also clear subscript/superscript and lists
   432→      document.execCommand('unlink')
   433→      handleInput()
   434→    }

   436→    // ----------------------------------------------------------------------
   437→    // Case conversion — operates on the current selection.
   438→    // If the selection is collapsed, no-op.
   439→    // ----------------------------------------------------------------------
   440→    function convertCase(mode: 'upper' | 'lower' | 'title') {
   441→      const sel = window.getSelection()
   442→      if (!sel || sel.rangeCount === 0 || sel.isCollapsed) {
   443→        toast_editor('Sélectionnez d\'abord le texte à transformer.')
   444→        return
   445→      }
   446→      const range = sel.getRangeAt(0)
   447→      const text = range.toString()
   448→      if (!text) return
   449→      let out = text
   450→      if (mode === 'upper') out = text.toUpperCase()
   451→      else if (mode === 'lower') out = text.toLowerCase()
   452→      else if (mode === 'title') {
   453→        out = text
   454→          .toLowerCase()
   455→          .replace(/(^|\s|[\.\!\?\;\:])([a-zà-ÿ])/g, (_, p1, p2) => p1 + p2.toUpperCase())
   456→      }
   457→      // Insert as a text node (preserves any inline styles of the wrapping element)
   458→      const textNode = document.createTextNode(out)
   459→      range.deleteContents()
   460→      range.insertNode(textNode)
   461→      // Reselect the new text so the user can continue working on it
   462→      const newRange = document.createRange()
   463→      newRange.selectNodeContents(textNode)
   464→      sel.removeAllRanges()
   465→      sel.addRange(newRange)
   466→      handleInput()
   467→    }
   468→
   469→    // Apply font family to the current selection (or future typing at caret)
   470→    function applyFontFamily(fontValue: string) {
   471→      const ff = FONT_FAMILIES.find((f) => f.value === fontValue)
   472→      if (!ff) return
   473→      exec('fontName', ff.css)
   474→    }

   476→    // Apply font size to the current selection.
   477→    // execCommand('fontSize', false, n) only accepts 1..7, so we use the
   478→    // FONT_SIZES mapping above. To get true pt sizes we additionally wrap the
   479→    // selection in a <span style="font-size:Npt"> — but execCommand is simpler
   480→    // and matches what most users expect from a basic WYSIWYG toolbar.
   481→    function applyFontSize(sizeValue: string) {
   482→      exec('fontSize', sizeValue)
   483→    }

   486→    function applyTextColor(color: string) {
   487→      exec('foreColor', color)
   488→    }
   489→
   490→    // Apply highlight color via execCommand('hiliteColor') (fallback 'backColor')
   491→    function applyHighlight(color: string) {
   492→      editorRef.current?.focus()
   493→      try {
   494→        if (!document.execCommand('hiliteColor', false, color)) {
   495→          document.execCommand('backColor', false, color)
   496→        }
   497→      } catch {
   498→        document.execCommand('backColor', false, color)
   499→      }
   500→      handleInput()
   501→    }
   502→
   503→    // Select all content inside the editor (object selection)
   504→    function selectAllInEditor() {
   505→      const el = editorRef.current
   506→      if (!el) return
   507→      el.focus()
   508→      const range = document.createRange()
   509→      range.selectNodeContents(el)
   510→      const sel = window.getSelection()
   511→      sel?.removeAllRanges()
   512→      sel?.addRange(range)
   513→      refreshActiveState()
   514→    }
   515→
   516→    // Clear selection (collapse caret)
   517→    function clearSelection() {
   518→      const sel = window.getSelection()
   519→      sel?.removeAllRanges()
   520→      editorRef.current?.focus()
   521→      refreshActiveState()
   522→    }

   524→    // ----------------------------------------------------------------------
   525→    // Deselect highlight — clears the highlight (surlignage) from the current
   526→    // selection by applying a transparent background color.
   527→    // ----------------------------------------------------------------------
   528→    function clearHighlight() {
   529→      editorRef.current?.focus()
   530→      // hiliteColor with 'transparent' clears the highlight in modern browsers.
   531→      try {
   532→        document.execCommand('hiliteColor', false, 'transparent')
   533→      } catch {
   534→        document.execCommand('backColor', false, 'transparent')
   535→      }
   536→      handleInput()
   537→    }

   539→    // ----------------------------------------------------------------------
   540→    // Insert a table at the caret position.
   541→    // Generates an HTML <table> with <thead> (1 row) + <tbody> (rows-1 rows).
   542→    // ----------------------------------------------------------------------
   543→    function insertTable(rows: number, cols: number) {

   547→        : ''
   548→      const bodyRows = Array.from(
   549→        { length: Math.max(1, rows - (hasHeader ? 1 : 0)) },
   550→        (_, r) =>
   551→          `<tr>${Array.from({ length: cols }, () => `<td>&nbsp;</td>`).join('')}</tr>`
   552→      ).join('')
   553→      const html = `<table class="iris-table"><caption>Légende du tableau</caption>${head}<tbody>${bodyRows}</tbody></table><p>&nbsp;</p>`
   554→      editorRef.current?.focus()
   555→      document.execCommand('insertHTML', false, html)

   723→    // top-bottom/left-right/shadow) handled separately.
   724→    // ----------------------------------------------------------------------
   725→    function applyBorder(preset: { label: string; value: string | null }) {
   726→      const block = getCurrentBlockElement()
   727→      if (!block) return
   728→      // Reset all borders first
   729→      block.style.border = ''
   730→      block.style.borderTop = ''
   731→      block.style.borderBottom = ''
   732→      block.style.borderLeft = ''
   733→      block.style.borderRight = ''
   734→      block.style.boxShadow = ''
   735→      block.style.padding = ''
   736→      if (preset.value == null) {
   737→        handleInput()
   738→        return
   739→      }
   740→      if (preset.value === 'shadow') {
   741→        block.style.boxShadow = '0 2px 8px rgba(124, 58, 237, 0.18)'
   742→        block.style.padding = '8pt 12pt'
   743→      } else if (['top', 'bottom', 'left', 'right'].includes(preset.value)) {
   744→        const side = preset.value as 'top' | 'bottom' | 'left' | 'right'
   745→        const cap = side.charAt(0).toUpperCase() + side.slice(1)
   746→        ;(block.style as any)[`border${cap}`] = '1.5px solid #111'
   747→        block.style.padding = '6pt 8pt'
   748→      } else if (preset.value === 'top-bottom') {
   749→        block.style.borderTop = '1.5px solid #111'
   750→        block.style.borderBottom = '1.5px solid #111'
   751→        block.style.padding = '6pt 8pt'
   752→      } else if (preset.value === 'left-right') {
   753→        block.style.borderLeft = '1.5px solid #111'
   754→        block.style.borderRight = '1.5px solid #111'
   755→        block.style.padding = '6pt 8pt'
   756→      } else {
   757→        // Full shorthand
   758→        block.style.border = preset.value
   759→        block.style.padding = '6pt 8pt'
   760→      }
   761→      handleInput()
   762→    }

   781→    // Returns the closest p/h1/h2/h3/blockquote/li/div (with class) ancestor.
   782→    // ----------------------------------------------------------------------
   783→    function getCurrentBlockElement(): HTMLElement | null {
   784→      const editor = editorRef.current
   785→      if (!editor) return null
   786→      const sel = window.getSelection()
   787→      if (!sel || sel.rangeCount === 0) return null
   788→      let node: Node | null = sel.getRangeAt(0).commonAncestorContainer
   789→      while (node && node !== editor) {
   790→        if (node.nodeType === 1) {
   791→          const tag = (node as HTMLElement).tagName.toLowerCase()
   792→          if (['p', 'h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'blockquote', 'li', 'pre', 'div'].includes(tag)) {
   793→            return node as HTMLElement
   794→          }
   795→        }
   796→        node = node.parentNode
   797→      }
   798→      return editor
   799→    }

   801→    // ----------------------------------------------------------------------
   802→    // Find & Replace — search the editor content for `query` and select the
   803→    // next match. Uses TreeWalker to traverse text nodes.
   804→    // ----------------------------------------------------------------------
   805→    function findInDocument(query: string) {
   806→      const editor = editorRef.current
   807→      if (!editor || !query) {
   808→        setFindState((s) => ({ ...s, query, count: 0, current: 0 }))
   809→        lastMatchRef.current = null
   810→        return
   811→      }
   812→      const walker = document.createTreeWalker(editor, NodeFilter.SHOW_TEXT, {
   813→        acceptNode(node) {
   814→          if (!node.nodeValue || !node.nodeValue.includes(query)) {
   815→            return NodeFilter.FILTER_REJECT
   816→          }
   817→          // Skip text inside script/style
   818→          const parent = node.parentElement
   819→          if (parent && ['SCRIPT', 'STYLE'].includes(parent.tagName)) {
   820→            return NodeFilter.FILTER_REJECT
   821→          }
   822→          return NodeFilter.FILTER_ACCEPT
   823→        },
   824→      })
   825→      const matches: { node: Text; offset: number }[] = []
   826→      let cur: Node | null
   827→      while ((cur = walker.nextNode())) {
   828→        const text = cur as Text
   829→        const idx = text.nodeValue!.indexOf(query)
   830→        if (idx >= 0) matches.push({ node: text, offset: idx })
   831→      }
   832→      if (matches.length === 0) {
   833→        setFindState((s) => ({ ...s, query, count: 0, current: 0 }))
   834→        lastMatchRef.current = null
   835→        toast_editor(`Aucune occurrence de « ${query} ».`)
   836→        return
   837→      }
   838→      // Find next match after current selection
   839→      const sel = window.getSelection()
   840→      let startIdx = 0
   841→      if (sel && sel.rangeCount > 0) {
   842→        const range = sel.getRangeAt(0)
   843→        for (let i = 0; i < matches.length; i++) {
   844→          const m = matches[i]
   845→          // Skip matches that are entirely before the current selection
   846→          if (
   847→            m.node === range.startContainer &&
   848→            m.offset + query.length <= range.startOffset
   849→          ) continue
   850→          startIdx = i
   851→          break
   852→        }
   853→      }
   854→      const m = matches[startIdx]
   855→      const range = document.createRange()
   856→      range.setStart(m.node, m.offset)
   857→      range.setEnd(m.node, m.offset + query.length)
   858→      sel?.removeAllRanges()
   859→      sel?.addRange(range)
   860→      lastMatchRef.current = range.cloneRange()
   861→      // Scroll the match into view
   862→      const rect = range.getBoundingClientRect()
   863→      if (rect.top < 80 || rect.top > window.innerHeight - 80) {
   864→        window.scrollBy({ top: rect.top - 200, behavior: 'smooth' })
   865→      }
   866→      setFindState((s) => ({ ...s, query, count: matches.length, current: startIdx + 1 }))
   867→    }

   869→    function replaceInDocument(query: string, replacement: string, all = false) {
   870→      const editor = editorRef.current
   871→      if (!editor || !query) return
   872→      let count = 0
   873→      const walker = document.createTreeWalker(editor, NodeFilter.SHOW_TEXT, {
   874→        acceptNode(node) {
   875→          if (!node.nodeValue || !node.nodeValue.includes(query)) {
   876→            return NodeFilter.FILTER_REJECT
   877→          }
   878→          const parent = node.parentElement
   879→          if (parent && ['SCRIPT', 'STYLE'].includes(parent.tagName)) {
   880→            return NodeFilter.FILTER_REJECT
   881→          }
   882→          return NodeFilter.FILTER_ACCEPT
   883→        },
   884→      })
   885→      const nodes: Text[] = []
   886→      let cur: Node | null
   887→      while ((cur = walker.nextNode())) nodes.push(cur as Text)
   888→      for (const text of nodes) {
   889→        if (!text.nodeValue) continue
   890→        if (all) {
   891→          const newValue = text.nodeValue.split(query).join(replacement)
   892→          if (newValue !== text.nodeValue) {
   893→            text.nodeValue = newValue
   894→            count += (text.nodeValue.match(new RegExp(escapeRegExp(replacement), 'g')) || []).length
   895→          }
   896→        } else {
   897→          // Replace only the next occurrence after the last match
   898→          const idx = text.nodeValue.indexOf(query)
   899→          if (idx >= 0) {
   900→            text.nodeValue =
   901→              text.nodeValue.slice(0, idx) +
   902→              replacement +
   903→              text.nodeValue.slice(idx + query.length)
   904→            count = 1
   905→            // Select the replacement so find-next continues from here
   906→            const range = document.createRange()
   907→            range.setStart(text, idx)
   908→            range.setEnd(text, idx + replacement.length)
   909→            const sel = window.getSelection()
   910→            sel?.removeAllRanges()
   911→            sel?.addRange(range)
   912→            lastMatchRef.current = range.cloneRange()
   913→            break
   914→          }
   915→        }
   916→      }
   917→      if (count > 0) handleInput()
   918→      setFindState((s) => ({ ...s, replacement, current: all ? 0 : s.current, count: all ? 0 : s.count }))
   919→      toast_editor(all ? `${count} remplacement(s) effectué(s).` : (count ? 'Remplacement effectué.' : 'Aucune occurrence trouvée.'))
   920→    }

   922→    function handleKeyDown(e: React.KeyboardEvent) {
   923→      const meta = e.metaKey || e.ctrlKey
   924→      if (meta && e.key.toLowerCase() === 'b') {
   925→        e.preventDefault()
   926→        exec('bold')
   927→      } else if (meta && e.key.toLowerCase() === 'i') {
   928→        e.preventDefault()
   929→        exec('italic')
   930→      } else if (meta && e.key.toLowerCase() === 'u') {
   931→        e.preventDefault()
   932→        exec('underline')
   933→      } else if (meta && e.key.toLowerCase() === 's') {
   934→        e.preventDefault()
   935→        exec('strikethrough')
   936→      } else if (meta && e.key.toLowerCase() === 'e') {
   937→        // Ctrl+E = center alignment (custom shortcut)
   938→        e.preventDefault()
   939→        exec('justifyCenter')
   940→      } else if (meta && e.shiftKey && e.key.toLowerCase() === 'a') {
   941→        // Ctrl+Shift+A = select all in editor
   942→        e.preventDefault()
   943→        selectAllInEditor()
   944→      } else if (meta && e.key.toLowerCase() === 'f') {
   945→        // Ctrl+F = find (we mark it handled so the browser's native find doesn't fire)
   946→        // The FindReplaceButton popover is the actual UI; Ctrl+F just dispatches
   947→        // a custom event the toolbar can listen for. For now we focus the editor
   948→        // and let the user click the Recherche button — this avoids hijacking
   949→        // the browser native find which some users may prefer.
   950→        // No preventDefault() — let the browser handle it if it wants to.
   951→      } else if (meta && e.key.toLowerCase() === 'k') {
   952→        // Ctrl+K = insert hyperlink (Google Docs / Notion convention)
   953→        e.preventDefault()
   954→        const url = window.prompt('URL du lien :', 'https://')
   955→        if (url) insertHyperlink(url)
   956→      } else if (meta && e.shiftKey && e.key === 'Enter') {
   957→        // Ctrl+Shift+Enter = insert manual page break
   958→        e.preventDefault()
   959→        insertPageBreak()
   960→      } else if (meta && e.shiftKey && (e.key === '7' || e.key === '8')) {
   961→        // Ctrl+Shift+7 = numbered list, Ctrl+Shift+8 = bulleted list (Google Docs convention)
   962→        e.preventDefault()
   963→        if (e.key === '7') exec('insertOrderedList')
   964→        else exec('insertUnorderedList')
   965→      }
   966→    }

   974→        .join('')
   975→      document.execCommand('insertHTML', false, html)
   976→      handleInput()
   977→    }
