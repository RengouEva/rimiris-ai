    75→// ============================================================================
    76→// A4Editor — WYSIWYG editor that mimics an A4 page
    77→// - 210mm × 297mm aspect ratio
    78→// - Standard academic margins (25mm top/sides, 30mm bottom — clearly visualized)
    79→// - Justified serif body text, sans-serif headings
    80→// - Two-row rich toolbar (NO scroll, all tools visible):
    81→//     Row 1: undo/redo | font family | font size | bold/italic/underline/strike |
    82→//            subscript/superscript | case (upper/lower/title) | highlight
    83→//     Row 2: paragraph/H1/H2/H3 | lists/quote | alignment | watermark | select all/clear | erase
    84→// - Per-button active state indicators (voyant lumineux) tracked via
    85→//   document.queryCommandState + selectionchange event.
    86→// - Optional watermark (filigrane) rendered behind the page content.
    87→// - Persists HTML to the store.
    88→// ============================================================================

   203→  function A4Editor({ value, onChange, editable = true, showToolbar = true, onPageCountChange }, ref) {
   204→    const editorRef = React.useRef<HTMLDivElement>(null)
   205→    const lastValueRef = React.useRef<string>(value)
   206→    const lastPageCountRef = React.useRef<number>(1)
   207→    const [pageCountState, setPageCountState] = React.useState<number>(1)
   208→    // Debounce timer for the auto-pagination pass — we don't want to reflow
   209→    // pages on every keystroke (would cause cursor jumps), but we DO want
   210→    // to react quickly when the user pauses or pastes/inserts content.
   211→    const paginationTimerRef = React.useRef<number | null>(null)
   212→    // Tracks whether we are currently inside an enforceAutoPagination pass,
   213→    // to avoid recursion (handleInput calls enforce, which mutates DOM, which
   214→    // could re-trigger input events).
   215→    const isPaginatingRef = React.useRef<boolean>(false)
   216→
   217→    // Active formatting state — refreshed on selectionchange and on input.
   218→    const [activeState, setActiveState] = React.useState({

   220→      italic: false,
   221→      underline: false,
   222→      strikeThrough: false,

   232→    })

   258→    // ----------------------------------------------------------------------
   259→    // Pagination : measure content height, derive page count, notify parent
   260→    // ----------------------------------------------------------------------
   261→    function recomputePageCount() {
   262→      const el = editorRef.current
   263→      if (!el) return
   264→      const px = el.scrollHeight
   265→      const count = Math.max(1, Math.ceil(px / A4_PAGE_HEIGHT_PX))
   266→      if (count !== lastPageCountRef.current) {
   267→        lastPageCountRef.current = count
   268→        setPageCountState(count)
   269→        onPageCountChange?.(count)
   270→      }
   271→    }
   272→
   273→    // ----------------------------------------------------------------------
   274→    // AUTO-PAGINATION — the core of "le texte ne doit jamais déborder dans la marge".
   275→    //
   276→    // Strategy: the contentEditable is a single continuous flow. A4 pages are
   277→    // purely visual (background-image every 297mm). To PREVENT text from
   278→    // entering the bottom margin (30mm) of any page, we auto-insert
   279→    // <div class="iris-page-break iris-page-break-auto" contenteditable="false">
   280→    // elements BEFORE any block that would otherwise cross into the bottom
   281→    // margin. The pusher's min-height is computed so that the next block
   282→    // starts exactly at the top of the next A4 page's text zone.
   283→    //
   284→    // Steps:
   285→    //   1. Remove ALL previously auto-inserted page breaks (data-auto="true").
   286→    //      This is the "reset" pass — we rebuild the pagination from scratch.
   287→    //   2. Walk top-level children of the editor (paragraphs, headings, lists,
   288→    //      tables, footnotes, manual page breaks, …).
   289→    //   3. For each child, compute its top/bottom position relative to the
   290→    //      top of the editor (offsetTop + cumulative).
   291→    //   4. Determine which A4 page the child STARTS on:
   292→    //        pageIdx = floor(topRelative / 297mm)
   293→    //      The bottom of that page's text zone is at:
   294→    //        pageTextZoneBottom = (pageIdx + 1) * 297mm − 30mm
   295→    //   5. If the child's bottom EXTENDS past pageTextZoneBottom AND the
   296→    //      child's top is BEFORE pageTextZoneBottom (i.e., it really starts
   297→    //      on this page, not already pushed to the next), insert an auto
   298→    //      page-break div BEFORE the child with min-height =
   299→    //        (pageIdx + 1) * 297mm − childTop
   300→    //      This pushes the child to start exactly at the next page's top,
   301→    //      and its text-zone top will be (pageIdx + 1) * 297mm + 25mm
   302→    //      (the next page's top margin), so it never enters the bottom
   303→    //      margin of the current page.
   304→    //   6. After each insertion, positions shift — we re-evaluate from the
   305→    //      top. Limit to 30 iterations as a safety net (a 30-page document
   306→    //      is the practical max for in-browser editing).
   307→    //   7. Special case: if a single block is TALLER than the entire text
   308→    //      zone (242mm — e.g., a very long table or a heading followed by
   309→    //      many lines with no <p> split), we cannot split it safely. We
   310→    //      still push it to a fresh page (so it starts at the top of a new
   311→    //      page) but we leave it overflowing into the next page's bottom
   312→    //      margin. This is a known limitation; the user should split long
   313→    //      blocks manually with Enter.
   314→    //
   315→    // The pusher element is non-editable (contenteditable="false") and is
   316→    // excluded from manual page-break operations. It is silently removed
   317→    // on every recompute, so it never duplicates or stacks.
   318→    // ----------------------------------------------------------------------
   319→    function enforceAutoPagination() {
   320→      const editor = editorRef.current
   321→      if (!editor) return
   322→      if (isPaginatingRef.current) return
   323→      isPaginatingRef.current = true
   324→
   325→      try {
   326→        // --- Step 1: remove all previously auto-inserted page breaks ---
   327→        const oldAuto = editor.querySelectorAll('.iris-page-break-auto[data-auto="true"]')
   328→        oldAuto.forEach((el) => el.remove())
   329→
   330→        // --- Step 2-6: walk children, insert pushers where needed ---
   331→        // We re-evaluate positions after each insertion because inserting a
   332→        // pusher shifts the positions of all subsequent siblings.
   333→        const MAX_ITERATIONS = 50
   334→        for (let iter = 0; iter < MAX_ITERATIONS; iter++) {
   335→          const children = Array.from(editor.children) as HTMLElement[]
   336→          let inserted = false
   337→
   338→          for (let i = 0; i < children.length; i++) {
   339→            const child = children[i]
   340→            // Skip our own auto page-breaks (shouldn't exist after reset, but
   341→            // safety net).
   342→            if (child.classList.contains('iris-page-break-auto')) continue
   343→            // Skip the footnotes section — it stays at the very end.
   344→            if (child.classList.contains('iris-footnotes')) continue
   345→            // Manual page breaks already force a real break — skip and let
   346→            // them do their job.
   347→            if (
   348→              child.classList.contains('iris-page-break') &&
   349→              !child.classList.contains('iris-page-break-auto')
   350→            ) {
   351→              continue
   352→            }
   353→
   354→            // Position of the child's TOP relative to the editor's content box.
   355→            // offsetTop is relative to offsetParent; for a contentEditable div
   356→            // directly containing block children, offsetParent is the editor
   357→            // itself (or the closest positioned ancestor). We use getBoundingClientRect
   358→            // + the editor's rect to be robust against any positioning.
   359→            const editorRect = editor.getBoundingClientRect()
   360→            const childRect = child.getBoundingClientRect()
   361→            const childTop = childRect.top - editorRect.top + editor.scrollTop
   362→            const childBottom = childRect.bottom - editorRect.top + editor.scrollTop
   363→            const childHeight = childRect.height
   364→
   365→            // If the child is collapsed/empty, skip
   366→            if (childHeight < 1) continue
   367→
   368→            // Safety: if a single block is TALLER than the entire text zone
   369→            // (242mm), it can never fit on a single page — pushing it would
   370→            // loop forever. We skip it (leave it where it is, even if it
   371→            // overflows). The user must split long blocks manually.
   372→            if (childHeight > A4_TEXT_ZONE_HEIGHT_PX - 5) continue
   373→
   374→            // Which A4 page does the child's TOP fall on? (0-indexed)
   375→            const pageIdx = Math.floor(childTop / A4_PAGE_HEIGHT_PX)
   376→            // The bottom of this page's TEXT ZONE (where text is allowed).
   377→            const pageTextZoneBottom =
   378→              (pageIdx + 1) * A4_PAGE_HEIGHT_PX - A4_MARGIN_BOTTOM_PX
   379→            // The bottom edge of this A4 page (= top edge of the next page).
   380→            const pageBottom = (pageIdx + 1) * A4_PAGE_HEIGHT_PX
   381→
   382→            // The child violates the bottom-margin invariant if EITHER:
   383→            //   (a) it crosses the text-zone boundary (starts before, ends after), OR
   384→            //   (b) it starts inside the bottom margin (already past the text zone).
   385→            const crossesTextZone =
   386→              childTop < pageTextZoneBottom - 5 && childBottom > pageTextZoneBottom + 5
   387→            const startsInMargin =
   388→              childTop >= pageTextZoneBottom - 5 && childTop < pageBottom - 5
   389→
   390→            if (crossesTextZone || startsInMargin) {
   391→              // We need to push this child to the NEXT PAGE'S TEXT ZONE.
   392→              // The next page's content-box top is `pageBottom`, but the
   393→              // actual text-zone top of the next page is `pageBottom + 25mm`
   394→              // (the next page's top margin). The pusher occupies space from
   395→              // childTop to nextPageTextZoneTop, which includes:
   396→              //   - the rest of the current page's bottom margin, AND
   397→              //   - the next page's entire top margin.
   398→              // After insertion, the child's new TOP = nextPageTextZoneTop,
   399→              // so the first line of text on the next page sits below the
   400→              // top margin — exactly where it should be.
   401→              const nextPageTextZoneTop = pageBottom + A4_MARGIN_TOP_PX
   402→              const pushHeight = nextPageTextZoneTop - childTop
   403→
   404→              // Safety: if pushHeight is invalid or zero, skip
   405→              if (pushHeight < 1) continue
   406→
   407→              // Build the pusher element
   408→              const pusher = document.createElement('div')
   409→              pusher.className = 'iris-page-break iris-page-break-auto'
   410→              pusher.setAttribute('contenteditable', 'false')
   411→              pusher.setAttribute('data-auto', 'true')
   412→              pusher.setAttribute('data-iris', 'page-break')
   413→              pusher.setAttribute('aria-hidden', 'true')
   414→              pusher.style.minHeight = `${pushHeight}px`
   415→              pusher.innerHTML = '<span>Page suivante</span>'
   416→
   417→              // Insert BEFORE the offending child
   418→              editor.insertBefore(pusher, child)
   419→              inserted = true
   420→              break // restart the loop with fresh positions
   421→            }
   422→          }
   423→
   424→          if (!inserted) break
   425→        }
   426→
   427→        // --- Final: recompute page count + persist HTML ---
   428→        recomputePageCount()
   429→        // Persist the new HTML (with auto page-breaks inserted) to the store
   430→        // so a subsequent React re-render doesn't wipe out the pushers.
   431→        const newHtml = editor.innerHTML
   432→        if (newHtml !== lastValueRef.current) {
   433→          lastValueRef.current = newHtml
   434→          onChange(newHtml)
   435→        }
   436→      } finally {
   437→        isPaginatingRef.current = false
   438→      }
   439→    }
   440→
   441→    // Debounced wrapper — schedules a single pagination pass after the user
   442→    // stops typing. Cancels any previously-scheduled pass so we don't pile
   443→    // up work during rapid typing.
   444→    function scheduleAutoPagination() {
   445→      if (paginationTimerRef.current !== null) {
   446→        window.clearTimeout(paginationTimerRef.current)
   447→      }
   448→      paginationTimerRef.current = window.setTimeout(() => {
   449→        paginationTimerRef.current = null
   450→        enforceAutoPagination()
   451→      }, PAGINATION_DEBOUNCE_MS)
   452→    }
   453→
   454→    // ----------------------------------------------------------------------
   455→    // Refresh active formatting state from the current selection.
   456→    // Uses document.queryCommandState for the boolean commands and a manual
   457→    // check on the closest block ancestor for formatBlock.
   458→    // ----------------------------------------------------------------------
   459→    function refreshActiveState() {
   460→      if (!editorRef.current) return
   461→      // Only refresh when the selection is inside our editor
   462→      const sel = window.getSelection()
   463→      if (!sel || sel.rangeCount === 0) return
   464→      const range = sel.getRangeAt(0)
   465→      if (!editorRef.current.contains(range.commonAncestorContainer)) return
   466→
   467→      const safe = (cmd: string) => {
   468→        try {
   469→          return document.queryCommandState(cmd)
   470→        } catch {
   471→          return false
   472→        }
   473→      }
   474→
   475→      // Determine the block tag of the current selection
   476→      let node: Node | null = range.commonAncestorContainer

   595→        document.removeEventListener('selectionchange', onSelChange)

   674→    // If the selection is collapsed, no-op.

   676→    function convertCase(mode: 'upper' | 'lower' | 'title') {
   677→      const sel = window.getSelection()
   678→      if (!sel || sel.rangeCount === 0 || sel.isCollapsed) {
   679→        toast_editor('Sélectionnez d\'abord le texte à transformer.')
   680→        return
   681→      }
   682→      const range = sel.getRangeAt(0)
   683→      const text = range.toString()
   684→      if (!text) return
   685→      let out = text
   686→      if (mode === 'upper') out = text.toUpperCase()
   687→      else if (mode === 'lower') out = text.toLowerCase()
   688→      else if (mode === 'title') {
   689→        out = text
   690→          .toLowerCase()
   691→          .replace(/(^|\s|[\.\!\?\;\:])([a-zà-ÿ])/g, (_, p1, p2) => p1 + p2.toUpperCase())
   692→      }
   693→      // Insert as a text node (preserves any inline styles of the wrapping element)
   694→      const textNode = document.createTextNode(out)
   695→      range.deleteContents()
   696→      range.insertNode(textNode)
   697→      // Reselect the new text so the user can continue working on it
   698→      const newRange = document.createRange()
   699→      newRange.selectNodeContents(textNode)
   700→      sel.removeAllRanges()
   701→      sel.addRange(newRange)
   702→      handleInput()
   703→    }

   959→    // top-bottom/left-right/shadow) handled separately.
   960→    // ----------------------------------------------------------------------
   961→    function applyBorder(preset: { label: string; value: string | null }) {
   962→      const block = getCurrentBlockElement()
   963→      if (!block) return
   964→      // Reset all borders first
   965→      block.style.border = ''
   966→      block.style.borderTop = ''
   967→      block.style.borderBottom = ''
   968→      block.style.borderLeft = ''
   969→      block.style.borderRight = ''
   970→      block.style.boxShadow = ''
   971→      block.style.padding = ''
   972→      if (preset.value == null) {
   973→        handleInput()
   974→        return
   975→      }
   976→      if (preset.value === 'shadow') {
   977→        block.style.boxShadow = '0 2px 8px rgba(124, 58, 237, 0.18)'
   978→        block.style.padding = '8pt 12pt'
   979→      } else if (['top', 'bottom', 'left', 'right'].includes(preset.value)) {
   980→        const side = preset.value as 'top' | 'bottom' | 'left' | 'right'
   981→        const cap = side.charAt(0).toUpperCase() + side.slice(1)
   982→        ;(block.style as any)[`border${cap}`] = '1.5px solid #111'
   983→        block.style.padding = '6pt 8pt'
   984→      } else if (preset.value === 'top-bottom') {
   985→        block.style.borderTop = '1.5px solid #111'
   986→        block.style.borderBottom = '1.5px solid #111'
   987→        block.style.padding = '6pt 8pt'
   988→      } else if (preset.value === 'left-right') {
   989→        block.style.borderLeft = '1.5px solid #111'
   990→        block.style.borderRight = '1.5px solid #111'
   991→        block.style.padding = '6pt 8pt'
   992→      } else {
   993→        // Full shorthand
   994→        block.style.border = preset.value
   995→        block.style.padding = '6pt 8pt'
   996→      }
   997→      handleInput()
   998→    }

  1015→    // ----------------------------------------------------------------------
  1016→    // Helper — find the block element containing the current selection.
  1017→    // Returns the closest p/h1/h2/h3/blockquote/li/div (with class) ancestor.
  1018→    // ----------------------------------------------------------------------
  1019→    function getCurrentBlockElement(): HTMLElement | null {
  1020→      const editor = editorRef.current
  1021→      if (!editor) return null
  1022→      const sel = window.getSelection()
  1023→      if (!sel || sel.rangeCount === 0) return null
  1024→      let node: Node | null = sel.getRangeAt(0).commonAncestorContainer
  1025→      while (node && node !== editor) {
  1026→        if (node.nodeType === 1) {
  1027→          const tag = (node as HTMLElement).tagName.toLowerCase()
  1028→          if (['p', 'h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'blockquote', 'li', 'pre', 'div'].includes(tag)) {
  1029→            return node as HTMLElement
  1030→          }
  1031→        }
  1032→        node = node.parentNode
  1033→      }
  1034→      return editor
  1035→    }

  1037→    // ----------------------------------------------------------------------
  1038→    // Find & Replace — search the editor content for `query` and select the
  1039→    // next match. Uses TreeWalker to traverse text nodes.
  1040→    // ----------------------------------------------------------------------
  1041→    function findInDocument(query: string) {
  1042→      const editor = editorRef.current
  1043→      if (!editor || !query) {
  1044→        setFindState((s) => ({ ...s, query, count: 0, current: 0 }))
  1045→        lastMatchRef.current = null
  1046→        return
  1047→      }
  1048→      const walker = document.createTreeWalker(editor, NodeFilter.SHOW_TEXT, {
  1049→        acceptNode(node) {
  1050→          if (!node.nodeValue || !node.nodeValue.includes(query)) {
  1051→            return NodeFilter.FILTER_REJECT
  1052→          }
  1053→          // Skip text inside script/style
  1054→          const parent = node.parentElement
  1055→          if (parent && ['SCRIPT', 'STYLE'].includes(parent.tagName)) {
  1056→            return NodeFilter.FILTER_REJECT
  1057→          }
  1058→          return NodeFilter.FILTER_ACCEPT
  1059→        },
  1060→      })
  1061→      const matches: { node: Text; offset: number }[] = []
  1062→      let cur: Node | null
  1063→      while ((cur = walker.nextNode())) {
  1064→        const text = cur as Text
  1065→        const idx = text.nodeValue!.indexOf(query)
  1066→        if (idx >= 0) matches.push({ node: text, offset: idx })
  1067→      }
  1068→      if (matches.length === 0) {
  1069→        setFindState((s) => ({ ...s, query, count: 0, current: 0 }))
  1070→        lastMatchRef.current = null
  1071→        toast_editor(`Aucune occurrence de « ${query} ».`)
  1072→        return
  1073→      }
  1074→      // Find next match after current selection
  1075→      const sel = window.getSelection()
  1076→      let startIdx = 0
  1077→      if (sel && sel.rangeCount > 0) {
  1078→        const range = sel.getRangeAt(0)
  1079→        for (let i = 0; i < matches.length; i++) {
  1080→          const m = matches[i]
  1081→          // Skip matches that are entirely before the current selection
  1082→          if (
  1083→            m.node === range.startContainer &&
  1084→            m.offset + query.length <= range.startOffset
  1085→          ) continue
  1086→          startIdx = i
  1087→          break
  1088→        }
  1089→      }
  1090→      const m = matches[startIdx]
  1091→      const range = document.createRange()
  1092→      range.setStart(m.node, m.offset)
  1093→      range.setEnd(m.node, m.offset + query.length)
  1094→      sel?.removeAllRanges()
  1095→      sel?.addRange(range)
  1096→      lastMatchRef.current = range.cloneRange()
  1097→      // Scroll the match into view
  1098→      const rect = range.getBoundingClientRect()
  1099→      if (rect.top < 80 || rect.top > window.innerHeight - 80) {
  1100→        window.scrollBy({ top: rect.top - 200, behavior: 'smooth' })
  1101→      }
  1102→      setFindState((s) => ({ ...s, query, count: matches.length, current: startIdx + 1 }))
  1103→    }
  1104→
  1105→    function replaceInDocument(query: string, replacement: string, all = false) {
  1106→      const editor = editorRef.current
  1107→      if (!editor || !query) return
  1108→      let count = 0
  1109→      const walker = document.createTreeWalker(editor, NodeFilter.SHOW_TEXT, {
  1110→        acceptNode(node) {
  1111→          if (!node.nodeValue || !node.nodeValue.includes(query)) {
  1112→            return NodeFilter.FILTER_REJECT
  1113→          }
  1114→          const parent = node.parentElement
  1115→          if (parent && ['SCRIPT', 'STYLE'].includes(parent.tagName)) {
  1116→            return NodeFilter.FILTER_REJECT
  1117→          }
  1118→          return NodeFilter.FILTER_ACCEPT
  1119→        },
  1120→      })
  1121→      const nodes: Text[] = []
  1122→      let cur: Node | null
  1123→      while ((cur = walker.nextNode())) nodes.push(cur as Text)
  1124→      for (const text of nodes) {
  1125→        if (!text.nodeValue) continue
  1126→        if (all) {
  1127→          const newValue = text.nodeValue.split(query).join(replacement)
  1128→          if (newValue !== text.nodeValue) {
  1129→            text.nodeValue = newValue
  1130→            count += (text.nodeValue.match(new RegExp(escapeRegExp(replacement), 'g')) || []).length
  1131→          }
  1132→        } else {
  1133→          // Replace only the next occurrence after the last match
  1134→          const idx = text.nodeValue.indexOf(query)
  1135→          if (idx >= 0) {
  1136→            text.nodeValue =
  1137→              text.nodeValue.slice(0, idx) +
  1138→              replacement +
  1139→              text.nodeValue.slice(idx + query.length)
  1140→            count = 1
  1141→            // Select the replacement so find-next continues from here
  1142→            const range = document.createRange()
  1143→            range.setStart(text, idx)
  1144→            range.setEnd(text, idx + replacement.length)
  1145→            const sel = window.getSelection()
  1146→            sel?.removeAllRanges()
  1147→            sel?.addRange(range)
  1148→            lastMatchRef.current = range.cloneRange()
  1149→            break
  1150→          }
  1151→        }
  1152→      }
  1153→      if (count > 0) handleInput()
  1154→      setFindState((s) => ({ ...s, replacement, current: all ? 0 : s.current, count: all ? 0 : s.count }))
  1155→      toast_editor(all ? `${count} remplacement(s) effectué(s).` : (count ? 'Remplacement effectué.' : 'Aucune occurrence trouvée.'))
  1156→    }

  1158→    function handleKeyDown(e: React.KeyboardEvent) {
  1159→      const meta = e.metaKey || e.ctrlKey
  1160→      if (meta && e.key.toLowerCase() === 'b') {
  1161→        e.preventDefault()
  1162→        exec('bold')
  1163→      } else if (meta && e.key.toLowerCase() === 'i') {
  1164→        e.preventDefault()
  1165→        exec('italic')
  1166→      } else if (meta && e.key.toLowerCase() === 'u') {
  1167→        e.preventDefault()
  1168→        exec('underline')
  1169→      } else if (meta && e.key.toLowerCase() === 's') {
  1170→        e.preventDefault()
  1171→        exec('strikethrough')
  1172→      } else if (meta && e.key.toLowerCase() === 'e') {
  1173→        // Ctrl+E = center alignment (custom shortcut)
  1174→        e.preventDefault()
  1175→        exec('justifyCenter')
  1176→      } else if (meta && e.shiftKey && e.key.toLowerCase() === 'a') {
  1177→        // Ctrl+Shift+A = select all in editor
  1178→        e.preventDefault()
  1179→        selectAllInEditor()
  1180→      } else if (meta && e.key.toLowerCase() === 'f') {
  1181→        // Ctrl+F = find (we mark it handled so the browser's native find doesn't fire)
  1182→        // The FindReplaceButton popover is the actual UI; Ctrl+F just dispatches
  1183→        // a custom event the toolbar can listen for. For now we focus the editor
  1184→        // and let the user click the Recherche button — this avoids hijacking
  1185→        // the browser native find which some users may prefer.
  1186→        // No preventDefault() — let the browser handle it if it wants to.
  1187→      } else if (meta && e.key.toLowerCase() === 'k') {
  1188→        // Ctrl+K = insert hyperlink (Google Docs / Notion convention)
  1189→        e.preventDefault()
  1190→        const url = window.prompt('URL du lien :', 'https://')

  1192→      } else if (meta && e.shiftKey && e.key === 'Enter') {
  1193→        // Ctrl+Shift+Enter = insert manual page break
  1194→        e.preventDefault()
  1195→        insertPageBreak()
  1196→      } else if (meta && e.shiftKey && (e.key === '7' || e.key === '8')) {

  1198→        e.preventDefault()

  1200→        else exec('insertUnorderedList')
