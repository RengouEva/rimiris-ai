    39→// ============================================================================
    40→// SectionWorkflowPanel — the right-side sliding panel that runs the per-section
    41→// mini-workflow: interview (4 modes) → validation → drafting → humanization.
    42→// Replaces the old AIPanel.
    43→// ============================================================================
    44→
    45→type PanelStep = 'interview' | 'validation' | 'draft' | 'humanization' | 'done'
    46→
    47→export function SectionWorkflowPanel({
    48→  section,
    49→  onClose,
    50→  onInsertDraft,
    51→  onReplaceDraft,
    52→}: {
    53→  section: Section
    54→  onClose: () => void
    55→  onInsertDraft: (html: string) => void
    56→  onReplaceDraft: (html: string) => void
    57→}) {
    58→  const {
    59→    project,
    60→    sections,
    61→    themeUnderstanding,
    62→    problemContext,
    63→    addSectionInterviewAnswer,
    64→    setSectionInterviewAnswers,
    65→    setSectionValidation,
    66→    setSectionHumanization,
    67→    setSectionStatus,
    68→    updateSectionContent,
    69→  } = useIrisStore()
    70→
    71→  // Determine initial step based on section state
    72→  const initialStep: PanelStep = React.useMemo(() => {
    73→    if (section.status === 'completed') return 'done'
    74→    if (section.humanization) return 'done'
    75→    if (section.wordCount > 100 && section.validation) return 'humanization'
    76→    if (section.validation?.overallOk) return 'draft'
    77→    if (section.interviewAnswers.length > 0) return 'validation'
    78→    return 'interview'
    79→  }, [section])
    80→
    81→  const [step, setStep] = React.useState<PanelStep>(initialStep)
    82→
    83→  const allSections = sections.map((s) => ({ title: s.title, content: s.content }))
    84→
    85→  return (
    86→    <motion.aside
    87→      initial={{ x: '100%' }}
    88→      animate={{ x: 0 }}
    89→      exit={{ x: '100%' }}
    90→      transition={{ type: 'spring', damping: 25, stiffness: 200 }}
    91→      className="w-full sm:w-[460px] flex-shrink-0 border-l border-border bg-background flex flex-col h-[calc(100vh-3.5rem)]"
    92→    >
    93→      <PanelHeader step={step} section={section} onClose={onClose} />
    94→
    95→      <div className="flex-1 overflow-y-auto">
    96→        <AnimatePresence mode="wait">
    97→          {step === 'interview' && (
    98→            <motion.div
    99→              key="interview"
   100→              initial={{ opacity: 0, x: 10 }}
   101→              animate={{ opacity: 1, x: 0 }}
   102→              exit={{ opacity: 0, x: -10 }}
   103→            >
   104→              <InterviewStep
   105→                section={section}
   106→                project={project}
   107→                themeUnderstanding={themeUnderstanding}
   108→                problemContext={problemContext}
   109→                onAnswer={(a) => addSectionInterviewAnswer(section.id, a)}
   110→                onSetAnswers={(as) => setSectionInterviewAnswers(section.id, as)}
   111→                onContinue={() => setStep('validation')}
   112→              />
   113→            </motion.div>
   114→          )}
   115→          {step === 'validation' && (
   116→            <motion.div
   117→              key="validation"
   118→              initial={{ opacity: 0, x: 10 }}
   119→              animate={{ opacity: 1, x: 0 }}
   120→              exit={{ opacity: 0, x: -10 }}
   121→            >
   122→              <ValidationStep
   123→                section={section}
   124→                project={project}
   125→                onValidated={(v) => {
   126→                  setSectionValidation(section.id, v)
   127→                  // Always allow the student to continue (validation is informative, not blocking).
   128→                  setStep('draft')
   129→                }}
   130→                onBack={() => setStep('interview')}
   131→              />
   132→            </motion.div>
   133→          )}
   134→          {step === 'draft' && (
   135→            <motion.div
   136→              key="draft"
   137→              initial={{ opacity: 0, x: 10 }}
   138→              animate={{ opacity: 1, x: 0 }}
   139→              exit={{ opacity: 0, x: -10 }}
   140→            >
   141→              <DraftStep
   142→                section={section}
   143→                project={project}
   144→                themeUnderstanding={themeUnderstanding}
   145→                problemContext={problemContext}
   146→                allSections={allSections}
   147→                onInsert={(html) => {
   148→                  // Insert into the editor (which triggers onChange → updateSectionContent)
   149→                  onInsertDraft(html)
   150→                  setStep('humanization')
   151→                }}
   152→                onBack={() => setStep('validation')}
   153→              />
   154→            </motion.div>
   155→          )}
   156→          {step === 'humanization' && (
   157→            <motion.div
   158→              key="humanization"
   159→              initial={{ opacity: 0, x: 10 }}
   160→              animate={{ opacity: 1, x: 0 }}
   161→              exit={{ opacity: 0, x: -10 }}
   162→            >
   163→              <HumanizationStep
   164→                section={section}
   165→                project={project}
   166→                onHumanized={(h) => {
   167→                  setSectionHumanization(section.id, h)
   168→                  // Replace (not append) the editor content with the humanized version.
   169→                  // replaceHtml triggers onChange → updateSectionContent automatically.
   170→                  onReplaceDraft(h.finalHtml)
   171→                  setStep('done')
   172→                }}
   173→                onSkip={() => setStep('done')}
   174→                onBack={() => setStep('draft')}
   175→              />
   176→            </motion.div>
   177→          )}
   178→          {step === 'done' && (
   179→            <motion.div
   180→              key="done"
   181→              initial={{ opacity: 0, x: 10 }}
   182→              animate={{ opacity: 1, x: 0 }}
   183→              exit={{ opacity: 0, x: -10 }}
   184→            >
   185→              <DoneStep
   186→                section={section}
   187→                onClose={onClose}
   188→                onMarkCompleted={() => {
   189→                  setSectionStatus(section.id, 'completed')
   190→                  onClose()
   191→                }}
   192→              />
   193→            </motion.div>
   194→          )}
   195→        </AnimatePresence>
   196→      </div>
   197→    </motion.aside>
   198→  )
   199→}

   201→// ============================================================================
   202→// Panel header with step indicator
   203→// ============================================================================
   204→
   205→function PanelHeader({
   206→  step,
   207→  section,
   208→  onClose,
   209→}: {
   210→  step: PanelStep
   211→  section: Section
   212→  onClose: () => void
   213→}) {
   214→  const steps: { id: PanelStep; label: string; icon: any }[] = [
   215→    { id: 'interview', label: 'Entretien', icon: HelpCircle },
   216→    { id: 'validation', label: 'Validation', icon: ShieldCheck },
   217→    { id: 'draft', label: 'Rédaction', icon: PenLine },
   218→    { id: 'humanization', label: 'Humanisation', icon: Wand2 },
   219→    { id: 'done', label: 'Terminé', icon: CheckCircle2 },
   220→  ]
   221→  const currentIdx = steps.findIndex((s) => s.id === step)
   222→
   223→  return (
   224→    <div className="border-b border-border p-3 space-y-2.5">
   225→      <div className="flex items-center justify-between">
   226→        <div className="flex items-center gap-2 min-w-0">
   227→          <Sparkles className="h-4 w-4 text-primary flex-shrink-0" />
   228→          <div className="min-w-0">
   229→            <p className="text-xs font-semibold leading-none">Workflow de section</p>
   230→            <p className="text-[11px] text-muted-foreground truncate mt-0.5">{section.title}</p>
   231→          </div>
   232→        </div>
   233→        <Button variant="ghost" size="sm" onClick={onClose} className="h-7 w-7 p-0 rounded-full">
   234→          <X className="h-4 w-4" />
   235→        </Button>
   236→      </div>
   237→      <div className="flex items-center gap-0.5">
   238→        {steps.map((s, idx) => (
   239→          <React.Fragment key={s.id}>
   240→            <div
   241→              className={cn(
   242→                'flex items-center gap-1 px-1.5 py-0.5 rounded text-[10px] font-medium transition-all',
   243→                idx === currentIdx
   244→                  ? 'bg-primary/10 text-primary'
   245→                  : idx < currentIdx
   246→                  ? 'text-emerald-600 dark:text-emerald-400'
   247→                  : 'text-muted-foreground/40'
   248→              )}
   249→            >
   250→              {idx < currentIdx ? <Check className="h-2.5 w-2.5" /> : <s.icon className="h-2.5 w-2.5" />}
   251→            </div>
   252→            {idx < steps.length - 1 && (
   253→              <div
   254→                className={cn(
   255→                  'w-3 h-px',
   256→                  idx < currentIdx ? 'bg-emerald-500/40' : 'bg-border'
   257→                )}
   258→              />
   259→            )}
   260→          </React.Fragment>
   261→        ))}
   262→      </div>
   263→    </div>
   264→  )
   265→}

   267→// ============================================================================
   268→// STEP 1 — Interview (4 modes)
   269→// ============================================================================
   270→
   271→type InterviewMode = 'know' | 'dont_know' | 'propose' | 'example'
   272→
   273→function InterviewStep({
   274→  section,
   275→  project,
   276→  themeUnderstanding,
   277→  problemContext,
   278→  onAnswer,
   279→  onSetAnswers,
   280→  onContinue,
   281→}: {
   282→  section: Section
   283→  project: any
   284→  themeUnderstanding: any
   285→  problemContext: any
   286→  onAnswer: (a: InterviewAnswer) => void
   287→  onSetAnswers: (a: InterviewAnswer[]) => void
   288→  onContinue: () => void
   289→}) {
   290→  const [currentQuestion, setCurrentQuestion] = React.useState<{
   291→    questionId: string
   292→    question: string
   293→    helper: string
   294→  } | null>(null)
   295→  const [mode, setMode] = React.useState<InterviewMode | null>(null)
   296→  const [studentInput, setStudentInput] = React.useState('')
   297→  const [loading, setLoading] = React.useState(true)
   298→  const [explanation, setExplanation] = React.useState<string | null>(null)
   299→  const [proposals, setProposals] = React.useState<
   300→    { id: string; label: string; content: string; rationale: string }[]
   301→  >([])
   302→  const [example, setExample] = React.useState<string | null>(null)
   303→  const [error, setError] = React.useState<string | null>(null)
   304→  const [progress, setProgress] = React.useState({ current: 0, total: 5 })
   305→  // Track the number of answers we've already loaded a question for
   306→  const [lastLoadedCount, setLastLoadedCount] = React.useState(section.interviewAnswers.length)
   307→
   308→  async function fetchNextQuestion() {
   309→    setLoading(true)
   310→    setError(null)
   311→    setMode(null)
   312→    setStudentInput('')
   313→    setExplanation(null)
   314→    setProposals([])
   315→    setExample(null)
   316→    try {
   317→      const res = await fetch('/api/ai/section-interview', {
   318→        method: 'POST',
   319→        headers: { 'Content-Type': 'application/json' },
   320→        body: JSON.stringify({
   321→          sectionTitle: section.title,
   322→          sectionDescription: '',
   323→          project,
   324→          themeUnderstanding,
   325→          problemContext,
   326→          answers: section.interviewAnswers,
   327→          action: 'next_question',
   328→        }),
   329→      })
   330→      const data = await res.json()
   331→      if (data.done) {
   332→        // All questions answered — go to validation
   333→        onContinue()
   334→        return
   335→      }
   336→      setCurrentQuestion(data.nextQuestion)
   337→      setProgress(data.progress || progress)
   338→    } catch (e: any) {
   339→      setError(e.message || 'Connexion impossible.')
   340→    } finally {
   341→      setLoading(false)
   342→    }
   343→  }
   344→
   345→  // Initial load
   346→  React.useEffect(() => {
   347→    fetchNextQuestion()
   348→    // eslint-disable-next-line react-hooks/exhaustive-deps
   349→  }, [])
   350→
   351→  // When the answers array grows (i.e. student just answered), fetch the next question.
   352→  // Skip the initial mount (handled above).
   353→  React.useEffect(() => {
   354→    if (section.interviewAnswers.length > lastLoadedCount) {
   355→      setLastLoadedCount(section.interviewAnswers.length)
   356→      fetchNextQuestion()
   357→    }
   358→    // eslint-disable-next-line react-hooks/exhaustive-deps
   359→  }, [section.interviewAnswers.length])
   360→
   361→  async function handleModeAction(selectedMode: InterviewMode) {
   362→    if (!currentQuestion) return
   363→    setMode(selectedMode)
   364→    setLoading(true)
   365→    setError(null)
   366→    try {
   367→      const res = await fetch('/api/ai/section-interview', {
   368→        method: 'POST',
   369→        headers: { 'Content-Type': 'application/json' },
   370→        body: JSON.stringify({
   371→          sectionTitle: section.title,
   372→          project,
   373→          themeUnderstanding,
   374→          problemContext,
   375→          answers: section.interviewAnswers,
   376→          action: selectedMode,
   377→          studentInput: selectedMode === 'know' ? studentInput : undefined,
   378→        }),
   379→      })
   380→      const data = await res.json()
   381→      if (data.error) throw new Error(data.error)
   382→      if (selectedMode === 'know') {
   383→        // Register answer and move on
   384→        onAnswer({
   385→          questionId: currentQuestion.questionId,
   386→          question: currentQuestion.question,
   387→          answer: studentInput.trim(),
   388→        })
   389→        setStudentInput('')
   390→        setMode(null)
   391→        fetchNextQuestion()
   392→      } else if (selectedMode === 'dont_know') {
   393→        setExplanation(data.explanation)
   394→      } else if (selectedMode === 'propose') {
   395→        setProposals(data.proposals || [])
   396→      } else if (selectedMode === 'example') {
   397→        setExample(data.example)
   398→      }
   399→    } catch (e: any) {
   400→      setError(e.message || 'Connexion impossible.')
   401→    } finally {
   402→      setLoading(false)
   403→    }
   404→  }
   405→
   406→  function acceptProposal(p: { id: string; content: string }) {
   407→    if (!currentQuestion) return
   408→    onAnswer({
   409→      questionId: currentQuestion.questionId,
   410→      question: currentQuestion.question,
   411→      answer: p.content,
   412→    })
   413→    setProposals([])
   414→    setMode(null)
   415→    fetchNextQuestion()
   416→  }
   417→
   418→  function handleSkipQuestion() {
   419→    if (!currentQuestion) return
   420→    onAnswer({
   421→      questionId: currentQuestion.questionId,
   422→      question: currentQuestion.question,
   423→      answer: '(non précisé)',
   424→    })
   425→    setMode(null)
   426→    fetchNextQuestion()
   427→  }
   428→
   429→  return (
   430→    <div className="p-4 space-y-4">
   431→      <div>
   432→        <Badge variant="outline" className="mb-2 text-[10px]">
   433→          Question {progress.current + 1} / {progress.total}
   434→        </Badge>
   435→        <p className="text-xs text-muted-foreground mb-1.5">Phase 3 · Entretien structuré</p>
   436→      </div>
   437→
   438→      {loading && !currentQuestion && (
   439→        <div className="flex items-center justify-center py-8">
   440→          <Loader2 className="h-6 w-6 animate-spin text-primary" />
   441→        </div>
   442→      )}
   443→
   444→      {currentQuestion && (
   445→        <motion.div
   446→          initial={{ opacity: 0, y: 5 }}
   447→          animate={{ opacity: 1, y: 0 }}
   448→          className="rounded-xl bg-muted/40 p-3 space-y-1"
   449→        >
   450→          <p className="text-sm font-medium leading-relaxed">{currentQuestion.question}</p>
   451→          {currentQuestion.helper && (
   452→            <p className="text-[11px] text-muted-foreground italic">{currentQuestion.helper}</p>
   453→          )}
   454→        </motion.div>
   455→      )}
   456→
   457→      {error && (
   458→        <div className="rounded-lg border border-red-500/40 bg-red-500/5 p-2.5 text-xs text-red-700 dark:text-red-400 flex items-center gap-2">
   459→          <AlertCircle className="h-3.5 w-3.5 flex-shrink-0" />
   460→          <span className="flex-1">{error}</span>
   461→        </div>
   462→      )}
   463→
   464→      {/* Mode selection (4 modes) */}
   465→      {currentQuestion && !mode && !loading && (
   466→        <div className="grid grid-cols-2 gap-1.5">
   467→          <ModeButton
   468→            icon={<PenLine className="h-3.5 w-3.5" />}
   469→            label="Je sais"
   470→            desc="J'écris ma réponse"
   471→            onClick={() => setMode('know')}
   472→            color="primary"
   473→          />
   474→          <ModeButton
   475→            icon={<HelpCircle className="h-3.5 w-3.5" />}
   476→            label="Je ne sais pas"
   477→            desc="IRIS m'explique"
   478→            onClick={() => handleModeAction('dont_know')}
   479→            color="amber"
   480→          />
   481→          <ModeButton
   482→            icon={<ListChecks className="h-3.5 w-3.5" />}
   483→            label="Propose-moi"
   484→            desc="Plusieurs options"
   485→            onClick={() => handleModeAction('propose')}
   486→            color="violet"
   487→          />
   488→          <ModeButton
   489→            icon={<Eye className="h-3.5 w-3.5" />}
   490→            label="Montre-moi"
   491→            desc="Un exemple concret"
   492→            onClick={() => handleModeAction('example')}
   493→            color="cyan"
   494→          />
   495→        </div>
   496→      )}
   497→
   498→      {/* Mode know: textarea + submit */}
   499→      {mode === 'know' && currentQuestion && (
   500→        <div className="space-y-2">
   501→          <Textarea
   502→            value={studentInput}
   503→            onChange={(e) => setStudentInput(e.target.value)}
   504→            placeholder="Votre réponse..."
   505→            autoFocus
   506→            className="resize-none min-h-[100px] text-sm"
   507→            disabled={loading}
   508→          />
   509→          <div className="flex gap-1.5">
   510→            <Button
   511→              variant="ghost"
   512→              size="sm"
   513→              onClick={() => setMode(null)}
   514→              className="text-xs"
   515→            >
   516→              Annuler
   517→            </Button>
   518→            <Button
   519→              size="sm"
   520→              onClick={() => handleModeAction('know')}
   521→              disabled={!studentInput.trim() || loading}
   522→              className="flex-1 iris-gradient text-white"
   523→            >
   524→              {loading ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Send className="h-3.5 w-3.5 mr-1" />}
   525→              Enregistrer
   526→            </Button>
   527→          </div>
   528→        </div>
   529→      )}
   530→
   531→      {/* Mode dont_know: explanation */}
   532→      {mode === 'dont_know' && explanation && (
   533→        <motion.div
   534→          initial={{ opacity: 0, y: 5 }}
   535→          animate={{ opacity: 1, y: 0 }}
   536→          className="space-y-2"
   537→        >
   538→          <div className="rounded-xl border border-amber-500/30 bg-amber-500/5 p-3">
   539→            <div className="flex items-start gap-2">
   540→              <Lightbulb className="h-4 w-4 text-amber-500 flex-shrink-0 mt-0.5" />
   541→              <p className="text-sm leading-relaxed">{explanation}</p>
   542→            </div>
   543→          </div>
   544→          <Button
   545→            variant="outline"
   546→            size="sm"
   547→            onClick={() => setMode(null)}
   548→            className="w-full text-xs"
   549→          >
   550→            Maintenant je veux répondre
   551→          </Button>
   552→        </motion.div>
   553→      )}
   554→
   555→      {/* Mode propose: 3 proposals */}
   556→      {mode === 'propose' && proposals.length > 0 && (
   557→        <motion.div
   558→          initial={{ opacity: 0, y: 5 }}
   559→          animate={{ opacity: 1, y: 0 }}
   560→          className="space-y-2"
   561→        >
   562→          <p className="text-xs text-muted-foreground italic">
   563→            Choisissez une proposition (vous pourrez l'éditer après) :
   564→          </p>
   565→          {proposals.map((p) => (
   566→            <button
   567→              key={p.id}
   568→              onClick={() => acceptProposal(p)}
   569→              className="w-full text-left rounded-xl border border-border bg-card p-3 hover:border-primary/40 hover:bg-primary/5 transition-colors"
   570→            >
   571→              {p.label && <p className="text-xs font-semibold mb-0.5">{p.label}</p>}
   572→              <p className="text-sm">{p.content}</p>
   573→              {p.rationale && (
   574→                <p className="text-[11px] text-muted-foreground italic mt-1">{p.rationale}</p>
   575→              )}
   576→            </button>
   577→          ))}
   578→          <Button
   579→            variant="ghost"
   580→            size="sm"
   581→            onClick={() => setMode(null)}
   582→            className="w-full text-xs"
   583→          >
   584→            Annuler
   585→          </Button>
   586→        </motion.div>
   587→      )}
   588→
   589→      {/* Mode example */}
   590→      {mode === 'example' && example && (
   591→        <motion.div
   592→          initial={{ opacity: 0, y: 5 }}
   593→          animate={{ opacity: 1, y: 0 }}
   594→          className="space-y-2"
   595→        >
   596→          <div className="rounded-xl border border-cyan-500/30 bg-cyan-500/5 p-3">
   597→            <div className="flex items-start gap-2">
   598→              <Eye className="h-4 w-4 text-cyan-500 flex-shrink-0 mt-0.5" />
   599→              <p className="text-sm leading-relaxed">{example}</p>
   600→            </div>
   601→          </div>
   602→          <Button
   603→            variant="outline"
   604→            size="sm"
   605→            onClick={() => setMode(null)}
   606→            className="w-full text-xs"
   607→          >
   608→            Maintenant je veux répondre
   609→          </Button>

  1015→// ============================================================================
  1016→// STEP 4 — Humanization (5-engine pipeline)
  1017→// ============================================================================
  1018→
  1019→type PassState = 'pending' | 'running' | 'done' | 'error'
  1020→
  1021→interface PassRuntime {
  1022→  state: PassState
  1023→  report?: string
  1024→  error?: string
  1025→}
  1026→
  1027→function HumanizationStep({
  1028→  section,
  1029→  project,
  1030→  onHumanized,
  1031→  onSkip,
  1032→  onBack,
  1033→}: {
  1034→  section: Section
  1035→  project: any
  1036→  onHumanized: (h: HumanizationResult) => void
  1037→  onSkip: () => void
  1038→  onBack: () => void
  1039→}) {
  1040→  // État par passe — chaque passe a son propre état indépendant
  1041→  // 'pending' → 'running' → 'done' (ou 'error' si échec)
  1042→  const [passesState, setPassesState] = React.useState<Record<string, PassRuntime>>(
  1043→    () => {
  1044→      // Si la section a déjà une humanisation complète, on marque les 5 passes comme 'done'
  1045→      if (section.humanization) {
  1046→        return {
  1047→          grammar: { state: 'done' as PassState, report: section.humanization.grammar },
  1048→          fluidity: { state: 'done' as PassState, report: section.humanization.fluidity },
  1049→          style: { state: 'done' as PassState, report: section.humanization.style },
  1050→          academic: { state: 'done' as PassState, report: section.humanization.academic },
  1051→          level: { state: 'done' as PassState, report: section.humanization.level },
  1052→        }
  1053→      }
  1054→      return {} as Record<string, PassRuntime>
  1055→    }
  1056→  )
  1057→
  1058→  // HTML courant au fil des passes (on enchaîne passe 1 → 2 → 3 → 4 → 5)
  1059→  const currentHtmlRef = React.useRef<string>(section.content)
  1060→  // HTML final (après passe 5) — utilisé pour mettre à jour l'éditeur
  1061→  const [finalHtml, setFinalHtml] = React.useState<string | null>(null)
  1062→  const [globalError, setGlobalError] = React.useState<string | null>(null)
  1063→  const [isRunning, setIsRunning] = React.useState(false)
  1064→
  1065→  const passes = [
  1066→    { name: 'Correction grammaticale', key: 'grammar', desc: 'Orthographe, accords, ponctuation' },
  1067→    { name: 'Fluidité', key: 'fluidity', desc: 'Transitions et connecteurs logiques' },
  1068→    { name: 'Variation du style', key: 'style', desc: 'Éviter les répétitions' },
  1069→    { name: 'Registre académique', key: 'academic', desc: 'Vocabulaire soutenu' },
  1070→    { name: 'Adaptation au niveau', key: 'level', desc: `${project.level || 'Master'}` },
  1071→  ]
  1072→
  1073→  function setPassState(key: string, state: PassState, extra?: Partial<PassRuntime>) {
  1074→    setPassesState((prev) => ({
  1075→      ...prev,
  1076→      [key]: { ...(prev[key] || { state: 'pending' }), state, ...extra },
  1077→    }))
  1078→  }
  1079→
  1080→  async function runPipeline() {
  1081→    if (isRunning) return
  1082→    setIsRunning(true)
  1083→    setGlobalError(null)
  1084→    currentHtmlRef.current = section.content
  1085→
  1086→    // Marquer toutes les passes non 'done' comme 'pending' (réinitialiser)
  1087→    const initial: Record<string, PassRuntime> = {}
  1088→    passes.forEach((p) => {
  1089→      const prev = passesState[p.key]
  1090→      initial[p.key] = prev && prev.state === 'done' ? prev : { state: 'pending' }
  1091→    })
  1092→    setPassesState(initial)
  1093→
  1094→    // Exécuter les passes une par une (sequenciellement)
  1095→    for (let i = 0; i < passes.length; i++) {
  1096→      const p = passes[i]
  1097→      // Si déjà fait (relance partielle), on garde le résultat
  1098→      if (passesState[p.key]?.state === 'done' && passesState[p.key]?.report) {
  1099→        // On ne relance pas — mais il faut quand même avancer le HTML courant
  1100→        // On ne peut pas reconstruire le HTML intermédiaire, donc on relance
  1101→        // réellement pour être sûr de l'enchaînement.
  1102→      }
  1103→      setPassState(p.key, 'running')
  1104→      try {
  1105→        const res = await fetch('/api/ai/humanize', {
  1106→          method: 'POST',
  1107→          headers: { 'Content-Type': 'application/json' },
  1108→          body: JSON.stringify({
  1109→            html: currentHtmlRef.current,
  1110→            level: project.level,
  1111→            filiere: project.filiere,
  1112→            norme: project.norme,
  1113→            language: project.language,
  1114→            mode: 'pass',
  1115→            passIndex: i + 1,
  1116→          }),
  1117→        })
  1118→        const data = await res.json()
  1119→        if (data.error) throw new Error(data.error)
  1120→
  1121→        currentHtmlRef.current = data.outputHtml
  1122→        setPassState(p.key, 'done', { report: data.report || '' })
  1123→
  1124→        // Petite pause pour laisser l'animation se voir
  1125→        await new Promise((r) => setTimeout(r, 250))
  1126→      } catch (e: any) {
  1127→        setPassState(p.key, 'error', { error: e.message || 'Échec' })
  1128→        setGlobalError(e.message || 'Connexion impossible.')
  1129→        setIsRunning(false)
  1130→        return
  1131→      }
  1132→    }
  1133→
  1134→    // Toutes les passes sont done — construire le HumanizationResult final
  1135→    const h: HumanizationResult = {
  1136→      grammar: passesState.grammar?.report || '',
  1137→      fluidity: passesState.fluidity?.report || '',
  1138→      style: passesState.style?.report || '',
  1139→      academic: passesState.academic?.report || '',
  1140→      level: passesState.level?.report || '',
  1141→      finalHtml: currentHtmlRef.current,
  1142→    }
  1143→    setFinalHtml(currentHtmlRef.current)
  1144→    onHumanized(h)
  1145→    setIsRunning(false)
  1146→  }
  1147→
  1148→  // Vrai état "terminé" : les 5 passes sont 'done'
  1149→  const allDone = passes.every((p) => passesState[p.key]?.state === 'done')

  1151→  return (
