  1069→  1069→function buildMarkdown(
  1070→  1070→  project: any,
  1071→  1071→  sections: any[],
  1072→  1072→  options: { includeTitlePage: boolean; includeTableOfContents: boolean }
  1073→  1073→): string {
  1074→  1074→  const parts: string[] = []
  1075→  1075→  if (options.includeTitlePage) {
  1076→  1076→    parts.push(`# ${project.title || 'Mémoire sans titre'}`)
  1077→  1077→    parts.push('')
  1078→  1078→    parts.push(`**${project.university || ''}**  `)
  1079→  1079→    parts.push(`**${project.faculty || ''}**  `)
  1080→  1080→    parts.push(`**${project.level || ''} · ${project.filiere || ''}**  `)
  1081→  1081→    parts.push('')
  1082→  1082→    parts.push(`*Norme : ${project.norme || 'APA'}*`)
  1083→  1083→    parts.push('')
  1084→  1084→    parts.push('---')
  1085→  1085→    parts.push('')
  1086→  1086→  }
  1087→  1087→  if (options.includeTableOfContents) {
  1088→  1088→    parts.push('## Table des matières')
  1089→  1089→    parts.push('')
  1090→  1090→    sections.forEach((s, idx) => {
  1091→  1091→      parts.push(`${idx + 1}. ${s.title}`)
  1092→  1092→    })
  1093→  1093→    parts.push('')
  1094→  1094→    parts.push('---')
  1095→  1095→    parts.push('')
  1096→  1096→  }
  1097→  1097→  sections.forEach((s, idx) => {
  1098→  1098→    if (!s.content || !s.content.trim()) return
  1099→  1099→    parts.push(`## ${idx + 1}. ${s.title}`)
  1100→  1100→    parts.push('')
  1101→  1101→    // Convert HTML → Markdown (basic)
  1102→  1102→    let md = s.content
  1103→  1103→    md = md.replace(/<h1[^>]*>(.*?)<\/h1>/gi, '# $1\n')
  1104→  1104→    md = md.replace(/<h2[^>]*>(.*?)<\/h2>/gi, '## $1\n')
  1105→  1105→    md = md.replace(/<h3[^>]*>(.*?)<\/h3>/gi, '### $1\n')
  1106→  1106→    md = md.replace(/<strong[^>]*>(.*?)<\/strong>/gi, '**$1**')
  1107→  1107→    md = md.replace(/<b[^>]*>(.*?)<\/b>/gi, '**$1**')
  1108→  1108→    md = md.replace(/<em[^>]*>(.*?)<\/em>/gi, '*$1*')
  1109→  1109→    md = md.replace(/<i[^>]*>(.*?)<\/i>/gi, '*$1*')
  1110→  1110→    md = md.replace(/<u[^>]*>(.*?)<\/u>/gi, '$1')
  1111→  1111→    md = md.replace(/<blockquote[^>]*>([\s\S]*?)<\/blockquote>/gi, '> $1\n')
  1112→  1112→    md = md.replace(/<li[^>]*>(.*?)<\/li>/gi, '- $1\n')
  1113→  1113→    md = md.replace(/<\/?(ul|ol)[^>]*>/gi, '\n')
  1114→  1114→    md = md.replace(/<p[^>]*>(.*?)<\/p>/gi, '$1\n\n')
  1115→  1115→    md = md.replace(/<br\s*\/?>/gi, '\n')
  1116→  1116→    md = md.replace(/<[^>]+>/g, '') // strip remaining tags
  1117→  1117→    md = md.replace(/\n{3,}/g, '\n\n')
  1118→  1118→    parts.push(md.trim())
  1119→  1119→    parts.push('')
  1120→  1120→  })
  1121→  1121→  return parts.join('\n')
  1122→  1122→}
