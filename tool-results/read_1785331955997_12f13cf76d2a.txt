    76→    76→// A4Editor — WYSIWYG editor that mimics an A4 page
    77→    77→// - 210mm × 297mm aspect ratio
    78→    78→// - Standard academic margins (25mm top/sides, 30mm bottom — clearly visualized)
    79→    79→// - Justified serif body text, sans-serif headings
    80→    80→// - Two-row rich toolbar (NO scroll, all tools visible):
    81→    81→//     Row 1: undo/redo | font family | font size | bold/italic/underline/strike |
    82→    82→//            subscript/superscript | case (upper/lower/title) | highlight
    83→    83→//     Row 2: paragraph/H1/H2/H3 | lists/quote | alignment | watermark | select all/clear | erase
    84→    84→// - Per-button active state indicators (voyant lumineux) tracked via
    85→    85→//   document.queryCommandState + selectionchange event.
    86→    86→// - Optional watermark (filigrane) rendered behind the page content.
    87→    87→// - Persists HTML to the store.

   214→   214→      block: 'p', // 'p' | 'h1' | 'h2' | 'h3' | 'blockquote'
   215→   215→    })

   237→   237→

   240→   240→
   241→   241→    // ----------------------------------------------------------------------
   242→   242→    // Pagination : measure content height, derive page count, notify parent
   243→   243→    // ----------------------------------------------------------------------
   244→   244→    function recomputePageCount() {
   245→   245→      const el = editorRef.current
   246→   246→      if (!el) return
   247→   247→      const px = el.scrollHeight
   248→   248→      const count = Math.max(1, Math.ceil(px / A4_PAGE_HEIGHT_PX))
   249→   249→      if (count !== lastPageCountRef.current) {
   250→   250→        lastPageCountRef.current = count
   251→   251→        setPageCountState(count)
   252→   252→        onPageCountChange?.(count)
   253→   253→      }
   254→   254→    }
   255→   255→
   256→   256→    // ----------------------------------------------------------------------
   257→   257→    // Refresh active formatting state from the current selection.
   258→   258→    // Uses document.queryCommandState for the boolean commands and a manual
   259→   259→    // check on the closest block ancestor for formatBlock.
   260→   260→    // ----------------------------------------------------------------------
   261→   261→    function refreshActiveState() {
   262→   262→      if (!editorRef.current) return
   263→   263→      // Only refresh when the selection is inside our editor
   264→   264→      const sel = window.getSelection()
   265→   265→      if (!sel || sel.rangeCount === 0) return
   266→   266→      const range = sel.getRangeAt(0)
   267→   267→      if (!editorRef.current.contains(range.commonAncestorContainer)) return
   268→   268→
   269→   269→      const safe = (cmd: string) => {
   270→   270→        try {
   271→   271→          return document.queryCommandState(cmd)
   272→   272→        } catch {
   273→   273→          return false
   274→   274→        }
   275→   275→      }
   276→   276→
   277→   277→      // Determine the block tag of the current selection
   278→   278→      let node: Node | null = range.commonAncestorContainer
   279→   279→      let blockTag = 'p'
   280→   280→      while (node && node !== editorRef.current) {
   281→   281→        if (node.nodeType === 1) {
   282→   282→          const tag = (node as HTMLElement).tagName.toLowerCase()
   283→   283→          if (['h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'blockquote', 'p', 'div', 'li'].includes(tag)) {
   284→   284→            blockTag = tag === 'div' || tag === 'li' ? 'p' : tag
   285→   285→            break
   286→   286→          }
   287→   287→        }
   288→   288→        node = node.parentNode
   289→   289→      }
   290→   290→
   291→   291→      setActiveState({
   292→   292→        bold: safe('bold'),
   293→   293→        italic: safe('italic'),
   294→   294→        underline: safe('underline'),
   295→   295→        strikeThrough: safe('strikeThrough'),
   296→   296→        subscript: safe('subscript'),
   297→   297→        superscript: safe('superscript'),
   298→   298→        insertUnorderedList: safe('insertUnorderedList'),
   299→   299→        insertOrderedList: safe('insertOrderedList'),
   300→   300→        justifyLeft: safe('justifyLeft'),
   301→   301→        justifyCenter: safe('justifyCenter'),
   302→   302→        justifyRight: safe('justifyRight'),
   303→   303→        justifyFull: safe('justifyFull'),
   304→   304→        block: blockTag as any,
   305→   305→      })
   306→   306→    }
   307→   307→
   308→   308→    // Expose imperative API
   309→   309→    React.useImperativeHandle(ref, () => ({
   310→   310→      insertHtml: (html: string) => {
   311→   311→        const el = editorRef.current
   312→   312→        if (!el) return
   313→   313→        el.focus()
   314→   314→        const sel = window.getSelection()
   315→   315→        const hasSelection = sel && sel.rangeCount > 0 && !sel.isCollapsed
   316→   316→        if (hasSelection) {
   317→   317→          const range = sel.getRangeAt(0)
   318→   318→          if (el.contains(range.commonAncestorContainer)) {
   319→   319→            range.deleteContents()
   320→   320→            const fragment = range.createContextualFragment(html)
   321→   321→            const lastNode = fragment.lastChild
   322→   322→            range.insertNode(fragment)
   323→   323→            if (lastNode) {
   324→   324→              range.setStartAfter(lastNode)
   325→   325→              range.setEndAfter(lastNode)
   326→   326→              sel.removeAllRanges()
   327→   327→              sel.addRange(range)
   328→   328→            }
   329→   329→          } else {
   330→   330→            appendHtml(el, html)
   331→   331→          }
   332→   332→        } else {
   333→   333→          appendHtml(el, html)
   334→   334→        }
   335→   335→        const newHtml = el.innerHTML
   336→   336→        lastValueRef.current = newHtml
   337→   337→        onChange(newHtml)
   338→   338→      },
   339→   339→      replaceHtml: (html: string) => {
   340→   340→        const el = editorRef.current
   341→   341→        if (!el) return
   342→   342→        el.innerHTML = html || ''
   343→   343→        lastValueRef.current = html || ''
   344→   344→        onChange(html || '')
   345→   345→        el.focus()
   346→   346→        requestAnimationFrame(() => recomputePageCount())
   347→   347→      },
   348→   348→      focus: () => editorRef.current?.focus(),
   349→   349→      getPageCount: () => lastPageCountRef.current,
   350→   350→    }))
   351→   351→
   352→   352→    // ----------------------------------------------------------------------
   353→   353→    // Recompute page count when value changes (external updates, e.g., AI draft)
   354→   354→    // ----------------------------------------------------------------------
   355→   355→    React.useEffect(() => {
   356→   356→      recomputePageCount()
   357→   357→    }, [value])
   358→   358→
   359→   359→    React.useEffect(() => {
   360→   360→      const onResize = () => recomputePageCount()
   361→   361→      window.addEventListener('resize', onResize)
   362→   362→      // Listen for selection changes to update active state indicators
   363→   363→      const onSelChange = () => refreshActiveState()
   364→   364→      document.addEventListener('selectionchange', onSelChange)
   365→   365→      const el = editorRef.current
   366→   366→      let ro: ResizeObserver | null = null
   367→   367→      if (el && typeof ResizeObserver !== 'undefined') {
   368→   368→        ro = new ResizeObserver(() => recomputePageCount())
   369→   369→        ro.observe(el)
   370→   370→      }
   371→   371→      requestAnimationFrame(() => recomputePageCount())
   372→   372→      return () => {
   373→   373→        window.removeEventListener('resize', onResize)
   374→   374→        document.removeEventListener('selectionchange', onSelChange)
   375→   375→        ro?.disconnect()
   376→   376→      }
   377→   377→      // eslint-disable-next-line react-hooks/exhaustive-deps
   378→   378→    }, [])
   379→   379→
   380→   380→    // Sync external value -> editor (only when it really changed externally)
   381→   381→    React.useEffect(() => {
   382→   382→      const el = editorRef.current
   383→   383→      if (!el) return
   384→   384→      if (value !== lastValueRef.current && value !== el.innerHTML) {
   385→   385→        el.innerHTML = value || ''
   386→   386→        lastValueRef.current = value
   387→   387→      }
   388→   388→    }, [value])
   389→   389→
   390→   390→    // Initialize once
   391→   391→    React.useEffect(() => {
   392→   392→      const el = editorRef.current
   393→   393→      if (!el) return
   394→   394→      if (el.innerHTML !== (value || '')) {
   395→   395→        el.innerHTML = value || ''
   396→   396→        lastValueRef.current = value || ''
   397→   397→      }
   398→   398→      // eslint-disable-next-line react-hooks/exhaustive-deps
   399→   399→    }, [])
   400→   400→
   401→   401→    function handleInput() {
   402→   402→      const el = editorRef.current
   403→   403→      if (!el) return
   404→   404→      const html = el.innerHTML
   405→   405→      lastValueRef.current = html
   406→   406→      onChange(html)
   407→   407→      recomputePageCount()
   408→   408→      refreshActiveState()
   409→   409→    }
   410→   410→
   411→   411→    function exec(command: string, val?: string) {
   412→   412→      editorRef.current?.focus()
   413→   413→      document.execCommand(command, false, val)
   414→   414→      handleInput()
   415→   415→    }
   416→   416→
   417→   417→    function setBlock(tag: 'h1' | 'h2' | 'h3' | 'p' | 'blockquote') {
   418→   418→      editorRef.current?.focus()
   419→   419→      try {
   420→   420→        document.execCommand('formatBlock', false, `<${tag}>`)
   421→   421→      } catch {
   422→   422→        // fallback noop
   423→   423→      }
   424→   424→      handleInput()
   425→   425→    }
   426→   426→
   427→   427→    function clearFormatting() {
   428→   428→      editorRef.current?.focus()
   429→   429→      document.execCommand('formatBlock', false, '<p>')
   430→   430→      document.execCommand('removeFormat')
   431→   431→      // Also clear subscript/superscript and lists
   432→   432→      document.execCommand('unlink')
   433→   433→      handleInput()
   434→   434→    }
   435→   435→
   436→   436→    // ----------------------------------------------------------------------
   437→   437→    // Case conversion — operates on the current selection.
   438→   438→    // If the selection is collapsed, no-op.
   439→   439→    // ----------------------------------------------------------------------
   440→   440→    function convertCase(mode: 'upper' | 'lower' | 'title') {
   441→   441→      const sel = window.getSelection()
   442→   442→      if (!sel || sel.rangeCount === 0 || sel.isCollapsed) {
   443→   443→        toast_editor('Sélectionnez d\'abord le texte à transformer.')
   444→   444→        return
   445→   445→      }
   446→   446→      const range = sel.getRangeAt(0)

   801→   801→    // ----------------------------------------------------------------------
   802→   802→    // Find & Replace — search the editor content for `query` and select the
   803→   803→    // next match. Uses TreeWalker to traverse text nodes.
   804→   804→    // ----------------------------------------------------------------------
   805→   805→    function findInDocument(query: string) {
   806→   806→      const editor = editorRef.current
   807→   807→      if (!editor || !query) {
   808→   808→        setFindState((s) => ({ ...s, query, count: 0, current: 0 }))
   809→   809→        lastMatchRef.current = null
   810→   810→        return
   811→   811→      }
   812→   812→      const walker = document.createTreeWalker(editor, NodeFilter.SHOW_TEXT, {
   813→   813→        acceptNode(node) {
   814→   814→          if (!node.nodeValue || !node.nodeValue.includes(query)) {
   815→   815→            return NodeFilter.FILTER_REJECT
   816→   816→          }
   817→   817→          // Skip text inside script/style
   818→   818→          const parent = node.parentElement
   819→   819→          if (parent && ['SCRIPT', 'STYLE'].includes(parent.tagName)) {
   820→   820→            return NodeFilter.FILTER_REJECT
   821→   821→          }
   822→   822→          return NodeFilter.FILTER_ACCEPT
   823→   823→        },
   824→   824→      })
   825→   825→      const matches: { node: Text; offset: number }[] = []
   826→   826→      let cur: Node | null
   827→   827→      while ((cur = walker.nextNode())) {
   828→   828→        const text = cur as Text
   829→   829→        const idx = text.nodeValue!.indexOf(query)
   830→   830→        if (idx >= 0) matches.push({ node: text, offset: idx })
   831→   831→      }
   832→   832→      if (matches.length === 0) {
   833→   833→        setFindState((s) => ({ ...s, query, count: 0, current: 0 }))
   834→   834→        lastMatchRef.current = null
   835→   835→        toast_editor(`Aucune occurrence de « ${query} ».`)
   836→   836→        return
   837→   837→      }
   838→   838→      // Find next match after current selection
   839→   839→      const sel = window.getSelection()
   840→   840→      let startIdx = 0
   841→   841→      if (sel && sel.rangeCount > 0) {
   842→   842→        const range = sel.getRangeAt(0)
   843→   843→        for (let i = 0; i < matches.length; i++) {
   844→   844→          const m = matches[i]
   845→   845→          // Skip matches that are entirely before the current selection
   846→   846→          if (
   847→   847→            m.node === range.startContainer &&
   848→   848→            m.offset + query.length <= range.startOffset
   849→   849→          ) continue
   850→   850→          startIdx = i
   851→   851→          break
   852→   852→        }
   853→   853→      }
   854→   854→      const m = matches[startIdx]
   855→   855→      const range = document.createRange()
   856→   856→      range.setStart(m.node, m.offset)
   857→   857→      range.setEnd(m.node, m.offset + query.length)
   858→   858→      sel?.removeAllRanges()
   859→   859→      sel?.addRange(range)
   860→   860→      lastMatchRef.current = range.cloneRange()
   861→   861→      // Scroll the match into view
   862→   862→      const rect = range.getBoundingClientRect()
   863→   863→      if (rect.top < 80 || rect.top > window.innerHeight - 80) {
   864→   864→        window.scrollBy({ top: rect.top - 200, behavior: 'smooth' })
   865→   865→      }
   866→   866→      setFindState((s) => ({ ...s, query, count: matches.length, current: startIdx + 1 }))
   867→   867→    }
   868→   868→
   869→   869→    function replaceInDocument(query: string, replacement: string, all = false) {
   870→   870→      const editor = editorRef.current
   871→   871→      if (!editor || !query) return
   872→   872→      let count = 0
   873→   873→      const walker = document.createTreeWalker(editor, NodeFilter.SHOW_TEXT, {
   874→   874→        acceptNode(node) {
   875→   875→          if (!node.nodeValue || !node.nodeValue.includes(query)) {
   876→   876→            return NodeFilter.FILTER_REJECT
   877→   877→          }
   878→   878→          const parent = node.parentElement
   879→   879→          if (parent && ['SCRIPT', 'STYLE'].includes(parent.tagName)) {
   880→   880→            return NodeFilter.FILTER_REJECT
   881→   881→          }
   882→   882→          return NodeFilter.FILTER_ACCEPT
   883→   883→        },
   884→   884→      })
   885→   885→      const nodes: Text[] = []
   886→   886→      let cur: Node | null
   887→   887→      while ((cur = walker.nextNode())) nodes.push(cur as Text)
   888→   888→      for (const text of nodes) {
   889→   889→        if (!text.nodeValue) continue
   890→   890→        if (all) {
   891→   891→          const newValue = text.nodeValue.split(query).join(replacement)

   902→   902→              replacement +
   903→   903→              text.nodeValue.slice(idx + query.length)

   905→   905→            // Select the replacement so find-next continues from here
   906→   906→            const range = document.createRange()

   908→   908→            range.setEnd(text, idx + replacement.length)
   909→   909→            const sel = window.getSelection()

   911→   911→            sel?.addRange(range)
   912→   912→            lastMatchRef.current = range.cloneRange()

   915→   915→        }

   917→   917→      if (count > 0) handleInput()
