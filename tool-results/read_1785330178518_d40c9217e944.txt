    51→// ============================================================================
    52→// A4Editor — WYSIWYG editor that mimics an A4 page
    53→// - 210mm × 297mm aspect ratio
    54→// - Standard academic margins (25mm top/sides, 30mm bottom — clearly visualized)
    55→// - Justified serif body text, sans-serif headings
    56→// - Two-row rich toolbar (NO scroll, all tools visible):
    57→//     Row 1: undo/redo | font family | font size | bold/italic/underline/strike |
    58→//            subscript/superscript | case (upper/lower/title) | highlight
    59→//     Row 2: paragraph/H1/H2/H3 | lists/quote | alignment | watermark | select all/clear | erase
    60→// - Per-button active state indicators (voyant lumineux) tracked via
    61→//   document.queryCommandState + selectionchange event.
    62→// - Optional watermark (filigrane) rendered behind the page content.
    63→// - Persists HTML to the store.
    64→// ============================================================================
    65→
    66→export interface A4EditorHandle {
    67→  insertHtml: (html: string) => void
    68→  replaceHtml: (html: string) => void
    69→  focus: () => void
    70→  /** Returns the current number of A4 pages based on actual content height */
    71→  getPageCount: () => number
    72→}
    73→
    74→interface A4EditorProps {
    75→  value: string // HTML
    76→  onChange: (html: string) => void
    77→  editable?: boolean
    78→  showToolbar?: boolean
    79→  /** Optional callback invoked whenever the page count changes (1, 2, 3, …). */
    80→  onPageCountChange?: (count: number) => void
    81→}
    82→
    83→// 297mm in CSS pixels at 96dpi: 297 * 96 / 25.4 ≈ 1122.52px
    84→const A4_PAGE_HEIGHT_PX = 1122.52
    85→
    86→// Font families offered in the toolbar.
    87→const FONT_FAMILIES: { label: string; value: string; css: string }[] = [
    88→  { label: 'Times New Roman', value: 'times', css: "'Times New Roman', 'Liberation Serif', Georgia, serif" },
    89→  { label: 'Arial', value: 'arial', css: "Arial, 'Helvetica Neue', sans-serif" },
    90→  { label: 'Calibri', value: 'calibri', css: "Calibri, 'Segoe UI', sans-serif" },
    91→  { label: 'Courier New', value: 'courier', css: "'Courier New', 'Liberation Mono', monospace" },
    92→  { label: 'Georgia', value: 'georgia', css: "Georgia, 'Times New Roman', serif" },
    93→  { label: 'Garamond', value: 'garamond', css: "Garamond, 'Times New Roman', serif" },
    94→  { label: 'Verdana', value: 'verdana', css: "Verdana, Geneva, sans-serif" },
    95→  { label: 'Inter', value: 'inter', css: "'Inter', 'Helvetica Neue', Arial, sans-serif" },
    96→]
    97→
    98→// Font sizes (in pt).
    99→const FONT_SIZES: { label: string; value: string }[] = [
   100→  { label: '8', value: '2' },   // 8pt ≈ size 2 in old execCommand fontSize
   101→  { label: '10', value: '1' },  // 10pt
   102→  { label: '12', value: '3' },  // 12pt
   103→  { label: '14', value: '4' },  // 14pt
   104→  { label: '16', value: '5' },  // 18pt
   105→  { label: '18', value: '6' },  // 24pt
   106→  { label: '24', value: '7' },  // 36pt
   107→]

   109→export const A4Editor = React.forwardRef<A4EditorHandle, A4EditorProps>(
   110→  function A4Editor({ value, onChange, editable = true, showToolbar = true, onPageCountChange }, ref) {
   111→    const editorRef = React.useRef<HTMLDivElement>(null)
   112→    const lastValueRef = React.useRef<string>(value)
   113→    const lastPageCountRef = React.useRef<number>(1)
   114→    const [pageCountState, setPageCountState] = React.useState<number>(1)
   115→
   116→    // Active formatting state — refreshed on selectionchange and on input.
   117→    const [activeState, setActiveState] = React.useState({
   118→      bold: false,
   119→      italic: false,
   120→      underline: false,
   121→      strikeThrough: false,
   122→      subscript: false,
   123→      superscript: false,
   124→      insertUnorderedList: false,
   125→      insertOrderedList: false,
   126→      justifyLeft: false,
   127→      justifyCenter: false,
   128→      justifyRight: false,
   129→      justifyFull: true, // default for academic body text
   130→      block: 'p', // 'p' | 'h1' | 'h2' | 'h3' | 'blockquote'
   131→    })
   132→
   133→    // Watermark state — applied to the A4 page background.
   134→    const [watermark, setWatermark] = React.useState<{
   135→      enabled: boolean
   136→      text: string
   137→      opacity: number
   138→      angle: number
   139→    }>({
   140→      enabled: false,
   141→      text: 'CONFIDENTIEL',
   142→      opacity: 0.08,
   143→      angle: -30,
   144→    })
   145→
   146→    // ----------------------------------------------------------------------
   147→    // Pagination : measure content height, derive page count, notify parent
   148→    // ----------------------------------------------------------------------
   149→    function recomputePageCount() {
   150→      const el = editorRef.current
   151→      if (!el) return
   152→      const px = el.scrollHeight
   153→      const count = Math.max(1, Math.ceil(px / A4_PAGE_HEIGHT_PX))
   154→      if (count !== lastPageCountRef.current) {
   155→        lastPageCountRef.current = count
   156→        setPageCountState(count)
   157→        onPageCountChange?.(count)
   158→      }
   159→    }
   160→
   161→    // ----------------------------------------------------------------------
   162→    // Refresh active formatting state from the current selection.
   163→    // Uses document.queryCommandState for the boolean commands and a manual
   164→    // check on the closest block ancestor for formatBlock.
   165→    // ----------------------------------------------------------------------
   166→    function refreshActiveState() {
   167→      if (!editorRef.current) return
   168→      // Only refresh when the selection is inside our editor
   169→      const sel = window.getSelection()
   170→      if (!sel || sel.rangeCount === 0) return
   171→      const range = sel.getRangeAt(0)
   172→      if (!editorRef.current.contains(range.commonAncestorContainer)) return
   173→
   174→      const safe = (cmd: string) => {
   175→        try {
   176→          return document.queryCommandState(cmd)
   177→        } catch {
   178→          return false
   179→        }
   180→      }
   181→
   182→      // Determine the block tag of the current selection
   183→      let node: Node | null = range.commonAncestorContainer
   184→      let blockTag = 'p'
   185→      while (node && node !== editorRef.current) {
   186→        if (node.nodeType === 1) {
   187→          const tag = (node as HTMLElement).tagName.toLowerCase()
   188→          if (['h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'blockquote', 'p', 'div', 'li'].includes(tag)) {
   189→            blockTag = tag === 'div' || tag === 'li' ? 'p' : tag
   190→            break
   191→          }
   192→        }
   193→        node = node.parentNode
   194→      }
   195→
   196→      setActiveState({
   197→        bold: safe('bold'),
   198→        italic: safe('italic'),
   199→        underline: safe('underline'),
   200→        strikeThrough: safe('strikeThrough'),
   201→        subscript: safe('subscript'),
   202→        superscript: safe('superscript'),
   203→        insertUnorderedList: safe('insertUnorderedList'),
   204→        insertOrderedList: safe('insertOrderedList'),
   205→        justifyLeft: safe('justifyLeft'),
   206→        justifyCenter: safe('justifyCenter'),
   207→        justifyRight: safe('justifyRight'),
   208→        justifyFull: safe('justifyFull'),
   209→        block: blockTag as any,
   210→      })
   211→    }
   212→
   213→    // Expose imperative API
   214→    React.useImperativeHandle(ref, () => ({
   215→      insertHtml: (html: string) => {
   216→        const el = editorRef.current
   217→        if (!el) return
   218→        el.focus()
   219→        const sel = window.getSelection()
   220→        const hasSelection = sel && sel.rangeCount > 0 && !sel.isCollapsed
   221→        if (hasSelection) {
   222→          const range = sel.getRangeAt(0)
   223→          if (el.contains(range.commonAncestorContainer)) {
   224→            range.deleteContents()
   225→            const fragment = range.createContextualFragment(html)
   226→            const lastNode = fragment.lastChild
   227→            range.insertNode(fragment)
   228→            if (lastNode) {
   229→              range.setStartAfter(lastNode)
   230→              range.setEndAfter(lastNode)
   231→              sel.removeAllRanges()
   232→              sel.addRange(range)
   233→            }
   234→          } else {
   235→            appendHtml(el, html)
   236→          }
   237→        } else {
   238→          appendHtml(el, html)
   239→        }
   240→        const newHtml = el.innerHTML
   241→        lastValueRef.current = newHtml
   242→        onChange(newHtml)
   243→      },
   244→      replaceHtml: (html: string) => {
   245→        const el = editorRef.current
   246→        if (!el) return
   247→        el.innerHTML = html || ''
   248→        lastValueRef.current = html || ''
   249→        onChange(html || '')
   250→        el.focus()
   251→        requestAnimationFrame(() => recomputePageCount())
   252→      },
   253→      focus: () => editorRef.current?.focus(),
   254→      getPageCount: () => lastPageCountRef.current,
   255→    }))
   256→
   257→    // ----------------------------------------------------------------------
   258→    // Recompute page count when value changes (external updates, e.g., AI draft)
   259→    // ----------------------------------------------------------------------
   260→    React.useEffect(() => {
   261→      recomputePageCount()
   262→    }, [value])
   263→
   264→    React.useEffect(() => {
   265→      const onResize = () => recomputePageCount()
   266→      window.addEventListener('resize', onResize)
   267→      // Listen for selection changes to update active state indicators
   268→      const onSelChange = () => refreshActiveState()
   269→      document.addEventListener('selectionchange', onSelChange)
   270→      const el = editorRef.current
   271→      let ro: ResizeObserver | null = null
   272→      if (el && typeof ResizeObserver !== 'undefined') {
   273→        ro = new ResizeObserver(() => recomputePageCount())
   274→        ro.observe(el)
   275→      }
   276→      requestAnimationFrame(() => recomputePageCount())
   277→      return () => {
   278→        window.removeEventListener('resize', onResize)
   279→        document.removeEventListener('selectionchange', onSelChange)
   280→        ro?.disconnect()
   281→      }
   282→      // eslint-disable-next-line react-hooks/exhaustive-deps
   283→    }, [])
   284→
   285→    // Sync external value -> editor (only when it really changed externally)
   286→    React.useEffect(() => {
   287→      const el = editorRef.current
   288→      if (!el) return
   289→      if (value !== lastValueRef.current && value !== el.innerHTML) {
   290→        el.innerHTML = value || ''
   291→        lastValueRef.current = value
   292→      }
   293→    }, [value])
   294→
   295→    // Initialize once
   296→    React.useEffect(() => {
   297→      const el = editorRef.current
   298→      if (!el) return
   299→      if (el.innerHTML !== (value || '')) {
   300→        el.innerHTML = value || ''
   301→        lastValueRef.current = value || ''
   302→      }
   303→      // eslint-disable-next-line react-hooks/exhaustive-deps
   304→    }, [])
   305→
   306→    function handleInput() {
   307→      const el = editorRef.current
   308→      if (!el) return
   309→      const html = el.innerHTML
   310→      lastValueRef.current = html
   311→      onChange(html)
   312→      recomputePageCount()
   313→      refreshActiveState()
   314→    }
   315→
   316→    function exec(command: string, val?: string) {
   317→      editorRef.current?.focus()
   318→      document.execCommand(command, false, val)
   319→      handleInput()
   320→    }
   321→
   322→    function setBlock(tag: 'h1' | 'h2' | 'h3' | 'p' | 'blockquote') {
   323→      editorRef.current?.focus()
   324→      try {
   325→        document.execCommand('formatBlock', false, `<${tag}>`)
   326→      } catch {
   327→        // fallback noop
   328→      }
   329→      handleInput()
   330→    }
   331→
   332→    function clearFormatting() {
   333→      editorRef.current?.focus()
   334→      document.execCommand('formatBlock', false, '<p>')
   335→      document.execCommand('removeFormat')
   336→      // Also clear subscript/superscript and lists
   337→      document.execCommand('unlink')
   338→      handleInput()
   339→    }
   340→
   341→    // ----------------------------------------------------------------------
   342→    // Case conversion — operates on the current selection.
   343→    // If the selection is collapsed, no-op.
   344→    // ----------------------------------------------------------------------
   345→    function convertCase(mode: 'upper' | 'lower' | 'title') {
   346→      const sel = window.getSelection()
   347→      if (!sel || sel.rangeCount === 0 || sel.isCollapsed) {
   348→        toast_editor('Sélectionnez d\'abord le texte à transformer.')
   349→        return
   350→      }
   351→      const range = sel.getRangeAt(0)
   352→      const text = range.toString()
   353→      if (!text) return
   354→      let out = text
   355→      if (mode === 'upper') out = text.toUpperCase()
   356→      else if (mode === 'lower') out = text.toLowerCase()
   357→      else if (mode === 'title') {
   358→        out = text
   359→          .toLowerCase()
   360→          .replace(/(^|\s|[\.\!\?\;\:])([a-zà-ÿ])/g, (_, p1, p2) => p1 + p2.toUpperCase())
   361→      }
   362→      // Insert as a text node (preserves any inline styles of the wrapping element)
   363→      const textNode = document.createTextNode(out)
   364→      range.deleteContents()
   365→      range.insertNode(textNode)
   366→      // Reselect the new text so the user can continue working on it
   367→      const newRange = document.createRange()
   368→      newRange.selectNodeContents(textNode)
   369→      sel.removeAllRanges()
   370→      sel.addRange(newRange)
   371→      handleInput()
   372→    }
   373→
   374→    // Apply font family to the current selection (or future typing at caret)
   375→    function applyFontFamily(fontValue: string) {
   376→      const ff = FONT_FAMILIES.find((f) => f.value === fontValue)
   377→      if (!ff) return
   378→      exec('fontName', ff.css)
   379→    }
   380→
   381→    // Apply font size to the current selection.
   382→    // execCommand('fontSize', false, n) only accepts 1..7, so we use the
   383→    // FONT_SIZES mapping above. To get true pt sizes we additionally wrap the
   384→    // selection in a <span style="font-size:Npt"> — but execCommand is simpler
   385→    // and matches what most users expect from a basic WYSIWYG toolbar.
   386→    function applyFontSize(sizeValue: string) {
   387→      exec('fontSize', sizeValue)
   388→    }
   389→
   390→    // Apply text color via execCommand('foreColor')
   391→    function applyTextColor(color: string) {
   392→      exec('foreColor', color)
   393→    }
   394→
   395→    // Apply highlight color via execCommand('hiliteColor') (fallback 'backColor')
   396→    function applyHighlight(color: string) {
   397→      editorRef.current?.focus()
   398→      try {
   399→        if (!document.execCommand('hiliteColor', false, color)) {
   400→          document.execCommand('backColor', false, color)
   401→        }
   402→      } catch {
   403→        document.execCommand('backColor', false, color)
   404→      }
   405→      handleInput()
   406→    }
   407→
   408→    // Select all content inside the editor (object selection)
   409→    function selectAllInEditor() {
   410→      const el = editorRef.current
   411→      if (!el) return
   412→      el.focus()
   413→      const range = document.createRange()
   414→      range.selectNodeContents(el)
   415→      const sel = window.getSelection()
   416→      sel?.removeAllRanges()
   417→      sel?.addRange(range)
   418→      refreshActiveState()
   419→    }
   420→
   421→    // Clear selection (collapse caret)
   422→    function clearSelection() {
   423→      const sel = window.getSelection()
   424→      sel?.removeAllRanges()
   425→      editorRef.current?.focus()
   426→      refreshActiveState()
   427→    }
   428→
   429→    function handleKeyDown(e: React.KeyboardEvent) {
   430→      const meta = e.metaKey || e.ctrlKey
   431→      if (meta && e.key.toLowerCase() === 'b') {
   432→        e.preventDefault()
   433→        exec('bold')
   434→      } else if (meta && e.key.toLowerCase() === 'i') {
   435→        e.preventDefault()
   436→        exec('italic')
   437→      } else if (meta && e.key.toLowerCase() === 'u') {
   438→        e.preventDefault()
   439→        exec('underline')
   440→      } else if (meta && e.key.toLowerCase() === 's') {
   441→        e.preventDefault()
   442→        exec('strikethrough')
   443→      } else if (meta && e.key.toLowerCase() === 'e') {
   444→        // Ctrl+E = center alignment (custom shortcut)
   445→        e.preventDefault()
   446→        exec('justifyCenter')
   447→      } else if (meta && e.shiftKey && e.key.toLowerCase() === 'a') {
   448→        // Ctrl+Shift+A = select all in editor
   449→        e.preventDefault()
   450→        selectAllInEditor()
   451→      }
   452→    }
   453→
   454→    function handlePaste(e: React.ClipboardEvent) {
   455→      e.preventDefault()
   456→      const text = e.clipboardData.getData('text/plain')
   457→      const html = text
   458→        .split(/\n\s*\n+/)
   459→        .map((p) => `<p>${escapeHtml(p).replace(/\n/g, '<br/>')}</p>`)
   460→        .join('')
   461→      document.execCommand('insertHTML', false, html)
   462→      handleInput()
   463→    }
   464→
   465→    return (
   466→      <div className="flex flex-col h-full">
   467→        {showToolbar && editable && (
   468→          <Toolbar
   469→            active={activeState}
   470→            onBold={() => exec('bold')}
   471→            onItalic={() => exec('italic')}
   472→            onUnderline={() => exec('underline')}
   473→            onStrike={() => exec('strikeThrough')}
   474→            onSubscript={() => exec('subscript')}
   475→            onSuperscript={() => exec('superscript')}
   476→            onH1={() => setBlock('h1')}
   477→            onH2={() => setBlock('h2')}
   478→            onH3={() => setBlock('h3')}
   479→            onP={() => setBlock('p')}
   480→            onUL={() => exec('insertUnorderedList')}
   481→            onOL={() => exec('insertOrderedList')}
   482→            onQuote={() => setBlock('blockquote')}
   483→            onUndo={() => exec('undo')}
   484→            onRedo={() => exec('redo')}
   485→            onClear={clearFormatting}
   486→            onAlignLeft={() => exec('justifyLeft')}
   487→            onAlignCenter={() => exec('justifyCenter')}
   488→            onAlignRight={() => exec('justifyRight')}
   489→            onAlignJustify={() => exec('justifyFull')}
   490→            onCaseUpper={() => convertCase('upper')}
   491→            onCaseLower={() => convertCase('lower')}
   492→            onCaseTitle={() => convertCase('title')}
   493→            onFontFamily={applyFontFamily}
   494→            onFontSize={applyFontSize}
   495→            onTextColor={applyTextColor}
   496→            onHighlight={applyHighlight}
   497→            onSelectAll={selectAllInEditor}
   498→            onClearSelection={clearSelection}
   499→            watermark={watermark}
   500→            onWatermarkChange={setWatermark}
   501→          />
   502→        )}
   503→
   504→        <div className="flex-1 overflow-y-auto bg-muted/30 p-4 sm:p-8 flex justify-center">
   505→          <A4Page
   506→            ref={editorRef}
   507→            editable={editable}
   508→            onInput={handleInput}
   509→            onKeyDown={handleKeyDown}
   510→            onPaste={handlePaste}
   511→            className="shadow-xl"
   512→            pageCount={pageCountState}
   513→            watermark={watermark.enabled ? watermark : null}
   514→          />
   515→        </div>
   516→      </div>
   517→    )
   518→  }
   519→)

   521→// ============================================================================
   522→// A4 page surface — the white sheet.
   523→// Now renders the watermark layer behind the content and uses the
   524→// `a4-paginated` class so the CSS background-image renders page boundaries
   525→// every 297mm (visual auto page-break indicator).
   526→// ============================================================================
   527→
   528→interface WatermarkConfig {
   529→  text: string
   530→  opacity: number
   531→  angle: number
   532→}
   533→
   534→const A4Page = React.forwardRef<
   535→  HTMLDivElement,
   536→  {
   537→    editable: boolean
   538→    onInput: () => void
   539→    onKeyDown: (e: React.KeyboardEvent) => void
   540→    onPaste: (e: React.ClipboardEvent) => void
   541→    className?: string
   542→    pageCount?: number
   543→    watermark?: WatermarkConfig | null
   544→  }
   545→>(function A4Page({ editable, onInput, onKeyDown, onPaste, className, pageCount = 1, watermark }, ref) {
   546→  return (
   547→    <div className="relative" style={{ width: '210mm' }}>
   548→      <div
   549→        ref={ref}
   550→        contentEditable={editable}
   551→        suppressContentEditableWarning
   552→        spellCheck
   553→        onInput={onInput}
   554→        onKeyDown={onKeyDown}
   555→        onPaste={onPaste}
   556→        className={cn(
   557→          'a4-page',
   558→          'a4-paginated',
   559→          'bg-white text-black',
   560→          'mx-auto',
   561→          'outline-none',
   562→          'prose-iris',
   563→          editable && 'cursor-text',
   564→          className
   565→        )}
   566→        style={{
   567→          width: '210mm',
   568→          minHeight: '297mm',
   569→          padding: '25mm 25mm 30mm 25mm',
   570→        }}
   571→        data-placeholder="Commencez à écrire ici, ou demandez à IRIS de générer un brouillon formaté…"
   572→      />
   573→      {/* Watermark layer — rendered behind the content, repeated on each A4 page.
   574→          Only visible when `watermark` prop is non-null. */}
   575→      {watermark && (
   576→        <div
   577→          className="a4-watermark-layer"
   578→          aria-hidden
   579→          style={{
   580→            // Position repeated watermark text every 297mm
   581→            // The CSS uses background-image with text rendered as SVG data URI
   582→            backgroundImage: `url("data:image/svg+xml;utf8,${encodeURIComponent(
